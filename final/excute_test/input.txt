local function AppendLogToFile(lineWithTime)
    pcall(function()
        local paths = {
            "/storage/emulated/0/Android/data/com.vnyh.gp/files/output.txt"
        }
        if CS and CS.UnityEngine and CS.UnityEngine.Application and CS.UnityEngine.Application.persistentDataPath then
            local p2 = CS.UnityEngine.Application.persistentDataPath .. "/output.txt"
            if p2 ~= paths[1] then table.insert(paths, p2) end
        end
        for _, p in ipairs(paths) do
            local f = io.open(p, "a")
            if f then
                f:write(lineWithTime .. "\r\n")
                f:close()
            end
        end
    end)
end

local function AddLog(msg)
    local lineWithTime = string.format("[%s] %s", os.date("%H:%M:%S"), tostring(msg))
    AppendLogToFile(lineWithTime)
end

pcall(function()
    AddLog("=========================================================================")
    AddLog(">>> [KÍCH HOẠT]: TÍNH TOÁN BÃI QUÁI TỐT NHẤT THEO PHÒNG THỦ ĐỂ FARM EXP <<<")

    -- 0. Kiểm tra thông tin hiện tại & Tính toán bãi quái tối ưu theo Thủ
    local curMap = (_G.SceneData and _G.SceneData.mapId) or 0
    local pMe = _G.RoleManager and _G.RoleManager.me
    local myPos = (pMe and pMe.cellPos) or (pMe and pMe.serverCoord)
    local curX = (myPos and myPos.x) or 0
    local curY = (myPos and myPos.y) or 0
    local myDef = _G.GetPlayerMonsterDefense and _G.GetPlayerMonsterDefense() or 0

    AddLog(string.format("-> Vị trí trước khi bay: Map %s (Tọa độ: %s#%s) | Phòng thủ: %d", tostring(curMap), tostring(curX), tostring(curY), myDef))

    -- Chọn bãi quái tốt nhất trong Tarkan theo phòng thủ thực tế
    local bestPt, _ = _G.GetBestMonsterFarmPoint and _G.GetBestMonsterFarmPoint(1009)
    local targetMap = 1009
    local targetX = (bestPt and bestPt.x) or 166
    local targetY = (bestPt and bestPt.y) or 196
    local spotName = (bestPt and bestPt.name) or "Tarkan Quái Lv195 (EXP x4.0)"
    local defReq = (bestPt and bestPt.defenseReq) or 390
    local monsterLvl = (bestPt and bestPt.monsterLvl) or 195

    AddLog(string.format("-> [ĐÁNH GIÁ THỦ %d]: Chọn bãi tối ưu nhất: %s (Quái Lv%d | Tọa độ: %d#%d | Thủ yêu cầu: %d)", 
        myDef, spotName, monsterLvl, targetX, targetY, defReq))

    -- 1. Tắt toàn bộ popup cản trở
    if _G.DismissBlockers then _G.DismissBlockers() end
    if _G.UIManager then
        if _G.UIID and _G.UIID.PromptTipUI then _G.UIManager.Hide(_G.UIID.PromptTipUI) end
        _G.UIManager.Hide("PromptTipUI")
        _G.UIManager.Hide("Activity_EventListUI")
        _G.UIManager.Hide("Task_MonsterListUI")
        _G.UIManager.Hide("Main_MapDetailUI")
    end

    -- 2. Dừng tìm đường cũ để không chạy ngược
    if pMe and pMe.StopMove then pMe:StopMove() end
    if _G.AutoTaskManage and _G.AutoTaskManage.SetAutoTask then _G.AutoTaskManage.SetAutoTask(false) end
    if _G.TaskManager and _G.TaskManager.TaskPathStop then _G.TaskManager.TaskPathStop() end

    -- 3. Bay thẳng tới bãi tối ưu qua ReqCallFlag + ReqTransmit
    local line = (_G.SceneData and _G.SceneData.line) or 1

    AddLog(string.format("-> Gửi lệnh ReqCallFlag(1, 0, %d, %d, %d, %d, 0)...", targetMap, line, targetX, targetY))
    if _G.networkRequest and _G.networkRequest.ReqCallFlag then
        _G.networkRequest.ReqCallFlag(1, 0, targetMap, line, targetX, targetY, 0)
    end
    if _G.NetManager and _G.MapMessage and _G.MapMessage.ReqCallFlag then
        _G.NetManager.Send(_G.MapMessage.ReqCallFlag, {
            type = 1,
            rid = 0,
            mapId = targetMap,
            line = line,
            x = targetX,
            y = targetY,
            hostId = 0
        })
    end
    if _G.networkRequest and _G.networkRequest.ReqTransmit then
        _G.networkRequest.ReqTransmit(targetMap, line, targetX, targetY)
    end

    -- 4. Bật AutoFight sau 1.5s và báo cáo kết quả
    if Timer and Timer.Start then
        Timer.Start(1.5, function()
            pcall(function()
                local pMe2 = _G.RoleManager and _G.RoleManager.me
                if pMe2 then
                    if pMe2.StopMove then pMe2:StopMove() end
                    if pMe2.SetAutoFight then pMe2:SetAutoFight(_G.AutoFightStrKey and _G.AutoFightStrKey.AutoFight or "AutoFight") end
                    if pMe2.SetAutoHookFight then pMe2:SetAutoHookFight(true) end
                    if pMe2.meAutoFight and pMe2.meAutoFight.SetAutoFightHookStart then pMe2.meAutoFight:SetAutoFightHookStart(true) end
                end
                if _G.QiJiHelperData and _G.QiJiHelperData.SetAutoFightData then
                    _G.QiJiHelperData.SetAutoFightData(true)
                end

                local newMap = (_G.SceneData and _G.SceneData.mapId) or 0
                local newPos = (pMe2 and pMe2.cellPos) or (pMe2 and pMe2.serverCoord)
                local nX = (newPos and newPos.x) or 0
                local nY = (newPos and newPos.y) or 0
                AddLog(string.format("-> Đã bật AutoFight! Vị trí hiện tại: Map %s (Tọa độ: %s#%s) tại bãi %s", tostring(newMap), tostring(nX), tostring(nY), spotName))
                AddLog("=========================================================================")
            end)
        end)
    end
end)
