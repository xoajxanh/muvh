local function DumpObject(obj, maxDepth, currentDepth, visited)
    if currentDepth == nil then currentDepth = 0 end
    if maxDepth == nil then maxDepth = 4 end
    if visited == nil then visited = {} end
    if type(obj) ~= "table" then return tostring(obj) end
    if visited[obj] then return "<CYCLE>" end
    if currentDepth > maxDepth then return "<TABLE>" end
    visited[obj] = true
    local str = "{\n"
    for k, v in pairs(obj) do
        local indent = string.rep("  ", currentDepth + 1)
        if type(v) == "table" then
            str = str .. indent .. tostring(k) .. " = " .. DumpObject(v, maxDepth, currentDepth + 1, visited) .. ",\n"
        elseif type(v) == "function" then
            str = str .. indent .. tostring(k) .. " = [function],\n"
        else
            str = str .. indent .. tostring(k) .. " = " .. tostring(v) .. ",\n"
        end
    end
    str = str .. string.rep("  ", currentDepth) .. "}"
    return str
end

local function DumpServerInfo()
    local msg = "\n=== DUMP SERVER INFO TRANG SỨC ===\n\n"
    local count = 0
    
    local items = _G.BagInfoData and _G.BagInfoData.TotalItems
    if items then
        for k, item in pairs(items) do
            local tblItem = item.tblItem or (item.data and item.data.tblItem) or {}
            local subType = tblItem.subType or 0
            
            -- Lọc Trang sức: 34-38
            if subType >= 34 and subType <= 38 then
                local name = tblItem.name or "Unknown"
                msg = msg .. "--- " .. name .. " (ID: " .. tostring(tblItem.id) .. ") ---\n"
                
                local sInfo = item.serverInfo or item.serverData
                if sInfo then
                    msg = msg .. "[serverInfo]:\n" .. DumpObject(sInfo, 2) .. "\n"
                else
                    msg = msg .. "Không có serverInfo!\n"
                end
                
                count = count + 1
                if count >= 3 then break end
            end
        end
    end
    
    pcall(function()
        local logPath = CS.UnityEngine.Application.persistentDataPath .. "/output.txt"
        local f = io.open(logPath, "w")
        if f then f:write(msg); f:close() end
        if _G.FloatingWordUtility then _G.FloatingWordUtility.QuickMsg("Đã dump serverInfo ra output.txt") end
    end)
end

DumpServerInfo()
