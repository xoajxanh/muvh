local function AppendLog(msg)
    pcall(function()
        local paths = {
            "/storage/emulated/0/Android/data/com.vnyh.gp/files/output.txt"
        }
        if CS and CS.UnityEngine and CS.UnityEngine.Application and CS.UnityEngine.Application.persistentDataPath then
            local p2 = CS.UnityEngine.Application.persistentDataPath .. "/output.txt"
            if p2 ~= paths[1] then table.insert(paths, p2) end
        end
        local line = string.format("[%s] %s\r\n", os.date("%H:%M:%S"), tostring(msg))
        for _, p in ipairs(paths) do
            local f = io.open(p, "a")
            if f then
                f:write(line)
                f:close()
            end
        end
    end)
end

local function NotifyMsg(msg)
    AppendLog(msg)
    pcall(function()
        if _G.FloatingWordUtility and _G.FloatingWordUtility.QuickMsg then
            _G.FloatingWordUtility.QuickMsg(tostring(msg))
        end
    end)
end

_G.Mod_AutoSpamChat_CurrentIndex = 0
local countSent = 0

local function GetSequentialSpamText()
    if #SPAM_LIST == 0 then return "" end
    _G.Mod_AutoSpamChat_CurrentIndex = (_G.Mod_AutoSpamChat_CurrentIndex or 0) + 1
    if _G.Mod_AutoSpamChat_CurrentIndex > #SPAM_LIST then
        _G.Mod_AutoSpamChat_CurrentIndex = 1
    end
    return SPAM_LIST[_G.Mod_AutoSpamChat_CurrentIndex]
end

-- 3. XÁC ĐỊNH DANH SÁCH KÊNH CẦN BẮN GÓI TIN THEO OPTION
-- =========================================================================
local function GetTargetChannels(option)
    local Enum = _G.ChatChannelEnum or {}
    local CROSS_REALM = Enum.CROSS_REALM or 2050002
    local LOCAL = Enum.LOCAL or 2050005
    local WORLD = Enum.WORLD or 2050001
    local GUILD = Enum.GUILD or 2050003
    local TEAM = Enum.TEAM or 2050004

    if option == 1 then
        return { { id = CROSS_REALM, name = "Liên Server" } }
    elseif option == 2 then
        return { { id = LOCAL, name = "Gần" } }
    elseif option == 3 then
        return { { id = WORLD, name = "Thế Giới" } }
    elseif option == 4 then
        return { { id = GUILD, name = "Guild" } }
    elseif option == 5 then
        return { { id = TEAM, name = "Đội" } }
    elseif option == 8 then
        return {
            { id = CROSS_REALM, name = "Liên Server" },
            { id = LOCAL, name = "Gần" }
        }
    elseif option == 9 then
        return {
            { id = CROSS_REALM, name = "Liên Server" },
            { id = LOCAL, name = "Gần" },
            { id = WORLD, name = "Thế Giới" }
        }
    else
        return { { id = LOCAL, name = "Gần" } }
    end
end

-- 4. HÀM GỬI CHAT THEO KÊNH
-- =========================================================================
local function SendChatToChannel(channelId, channelName, text)
    local myId = (_G.ViewData and _G.ViewData.meData and _G.ViewData.meData.id)
        or (_G.RoleManager and _G.RoleManager.me and _G.RoleManager.me.id) or 0

    local textData = {
        inputData = {},
        message = text
    }

    local isSent = false

    -- Cách 1: Bắn qua EventManager (Luồng chuẩn của game, tự format qua ChatController)
    if _G.EventManager and _G.Event and _G.Event.Chat_ReqChat then
        local msg = {
            chatType = channelId,
            textData = textData,
            toRoleId = myId
        }
        _G.EventManager.Dispatch(_G.Event.Chat_ReqChat, msg)
        isSent = true
    end

    -- Cách 2 (Dự phòng): Bắn trực tiếp qua NetManager gói ChatMessage.ReqChat
    if not isSent and _G.NetManager and _G.ChatMessage and _G.ChatMessage.ReqChat then
        local chatJson = ""
        if _G.json and _G.json.encode then
            chatJson = _G.json.encode(textData)
        else
            chatJson = string.format("{\"inputData\":{},\"message\":\"%s\"}", text)
        end

        _G.NetManager.Send(_G.ChatMessage.ReqChat, {
            chatType = channelId,
            chatMsg = chatJson,
            toRoleId = myId
        })
        isSent = true
    end

    return isSent
end

local function SendAutoChat()
    if not _G.Mod_AutoSpamChat_Running then return end

    local success, err = pcall(function()
        local currentText = GetSequentialSpamText()
        local channels = GetTargetChannels(CHAT_CHANNEL_OPTION)
        countSent = countSent + 1

        for _, ch in ipairs(channels) do
            local ok = SendChatToChannel(ch.id, ch.name, currentText)
            if ok then
                AppendLog(string.format("[AUTO CHAT #%d] [%s] [Câu %d/%d]: \"%s\"", countSent, ch.name, _G.Mod_AutoSpamChat_CurrentIndex, #SPAM_LIST, currentText))
            else
                AppendLog(string.format("[AUTO CHAT LOI] Không thể gửi tới kênh %s!", ch.name))
            end
        end
    end)

    if not success then
        AppendLog("[AUTO CHAT ERROR]: " .. tostring(err))
    end
end

_G.Mod_AutoSpamChat_Running = true

    _G.Mod_AutoSpamChat_Coroutine = _G.Coroutine.Start(function()
        while _G.Mod_AutoSpamChat_Running do
            _G.Coroutine.Wait(5.0)
            if _G.Mod_AutoSpamChat_Running then
                SendAutoChat()
            end
        end
    end)
