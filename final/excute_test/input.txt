pcall(function()
    if _G.WriteLog then _G.WriteLog("=== RUNNING ANSTATS TEST SCRIPT ===") end
    
    -- Giả lập tin nhắn hệ thống thông báo mở bản đồ ẩn
    local testMsgText = "[Hệ thống] Dino trong Bản đồ ẩn đã triệu hồi Thể Biến Dị Hoàn Toàn - Kỵ Sĩ"
    
    if _G.ChatData and _G.ChatData.AddMessage then
        _G.ChatData.AddMessage(8, { message = testMsgText })
        if _G.WriteLog then _G.WriteLog("[AnStats Test] Called ChatData.AddMessage successfully!") end
    else
        -- Fallback trực tiếp nếu ChatData không sẵn sàng
        _G.Mod_AnStats = _G.Mod_AnStats or {}
        _G.Mod_AnStats["Dino"] = (_G.Mod_AnStats["Dino"] or 0) + 1
        if _G.Mod_SaveAnStats then _G.Mod_SaveAnStats() end
        if _G.Mod_SendAnStatsTelegram then _G.Mod_SendAnStatsTelegram() end
        if _G.WriteLog then _G.WriteLog("[AnStats Test] Direct update Dino +1") end
    end
end)
