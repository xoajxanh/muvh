-- =========================================================================
-- [DEBUG SCRIPT]: QUÉT CHI TIẾT TÚI ĐỒ (BAG SCAN & EQUIP ATTRIBUTES DUMP)
-- Mục đích: Lấy chính xác cấu trúc dữ liệu của đồ đã cường hóa (+xxx), đồ Dung, đồ Bộ
-- Ghi kết quả ra: output.txt (tại persistentDataPath để push_test.ps1 kéo về)
-- =========================================================================

pcall(function()
    local logLines = {}
    local function AddLog(line)
        table.insert(logLines, line)
        if CS.UnityEngine.Debug and CS.UnityEngine.Debug.Log then
            CS.UnityEngine.Debug.Log("[EQUIP_DUMP] " .. tostring(line))
        end
    end

    AddLog("==================================================================")
    AddLog("                 BẮT ĐẦU QUÉT TRANG BỊ TRONG TÚI ĐỒ               ")
    AddLog("                 Thời gian: " .. os.date("%Y-%m-%d %H:%M:%S"))
    AddLog("==================================================================")

    local items = _G.BagInfoData and _G.BagInfoData.TotalItems
    if not items and _G.BagInfoData and _G.BagInfoData.GetTotalItems then
        pcall(function() items = _G.BagInfoData:GetTotalItems() end)
    end

    if not items then
        AddLog("[LỖI] Không tìm thấy BagInfoData.TotalItems!")
        if _G.FloatingWordUtility and _G.FloatingWordUtility.QuickMsg then
            _G.FloatingWordUtility.QuickMsg("LỖI: Không đọc được túi đồ!")
        end
        return
    end

    local totalCount = 0
    local equipCount = 0

    for k, item in pairs(items) do
        if item then
            totalCount = totalCount + 1
            local tblItem = item.tblItem or (item.data and item.data.tblItem) or {}
            local tblEquip = item.tblEquip or (item.data and item.data.tblEquip) or {}
            local itemType = tblItem.type or 0
            local itemId = tblItem.id or item.itemId or (item.data and item.data.itemId) or 0
            local name = tblItem.name or item.name or (item.data and item.data.name) or "Unknown"

            -- Lấy toàn bộ thông tin cường hóa và thuộc tính từ mọi tầng dữ liệu
            local sInfo = item.serverInfo or item.serverData or {}
            local d = item.data or {}
            local dSInfo = d.serverInfo or d.serverData or {}

            local intensify = item.intensify or d.intensify or sInfo.intensify or dSInfo.intensify
            local additional = item.additional or d.additional or sInfo.additional or dSInfo.additional
            local level = item.level or d.level or sInfo.level or dSInfo.level
            local breach = item.breach or d.breach or sInfo.breach or dSInfo.breach
            local canSmelt = sInfo.canSmelt or dSInfo.canSmelt
            local suitId = (tblEquip and tblEquip.suitId) or sInfo.suitId

            -- Thu thập các dòng Trác Việt nếu có
            local excDesList = {}
            if _G.RoleEquipUtility then
                local rawExc = item.excellence or sInfo.excellentList or sInfo.excellentInfo or sInfo.excellentAttrs
                if rawExc and _G.RoleEquipUtility.GetEquipExcellence then
                    pcall(function() excDesList = _G.RoleEquipUtility.GetEquipExcellence(rawExc, tblEquip) end)
                end
                if (#excDesList == 0) and _G.RoleEquipUtility.GetEquipExcellenceDesByServerInfo then
                    pcall(function() excDesList = _G.RoleEquipUtility.GetEquipExcellenceDesByServerInfo(sInfo) end)
                end
            end
            if (#excDesList == 0) and item.GetEquipExcellenceDesList then
                pcall(function() excDesList = item:GetEquipExcellenceDesList() end)
            end

            -- Nếu là trang bị hoặc có level cường hóa > 0
            local isEquip = (itemType == 2) or (tblEquip and tblEquip.id) or (intensify and tonumber(intensify) and tonumber(intensify) > 0)
            if isEquip then
                equipCount = equipCount + 1
                AddLog("------------------------------------------------------------------")
                AddLog(string.format("[%d] Trang Bị: %s (ItemID: %s, GUID: %s)", equipCount, tostring(name), tostring(itemId), tostring(item.id or d.id)))
                AddLog(string.format("    - Type: %s | SubType: %s | Quality: %s", tostring(itemType), tostring(tblItem.subType), tostring(tblItem.quality)))
                AddLog(string.format("    - Cường hóa (intensify) : %s", tostring(intensify)))
                AddLog(string.format("    - Gia cường (additional): %s", tostring(additional)))
                AddLog(string.format("    - Cấp bậc   (level)     : %s", tostring(level)))
                AddLog(string.format("    - Đột phá   (breach)    : %s", tostring(breach)))
                AddLog(string.format("    - Đồ Dung   (canSmelt)  : %s", tostring(canSmelt)))
                AddLog(string.format("    - Đồ Bộ     (suitId)    : %s", tostring(suitId)))

                if excDesList and #excDesList > 0 then
                    AddLog("    - Dòng Trác Việt (" .. #excDesList .. " dòng):")
                    for idx, des in ipairs(excDesList) do
                        AddLog(string.format("        * [%d] %s", idx, tostring(des)))
                    end
                else
                    AddLog("    - Dòng Trác Việt: (Không có)")
                end

                -- Dump tất cả key trong sInfo để tìm các trường ẩn
                if sInfo and type(sInfo) == "table" then
                    local sKeys = {}
                    for sk, sv in pairs(sInfo) do
                        table.insert(sKeys, string.format("%s=%s", tostring(sk), tostring(sv)))
                    end
                    if #sKeys > 0 then
                        AddLog("    - serverInfo keys: " .. table.concat(sKeys, ", "))
                    end
                end
            end
        end
    end

    AddLog("==================================================================")
    AddLog(string.format("TỔNG KẾT: Tìm thấy %d vật phẩm, trong đó có %d trang bị.", totalCount, equipCount))
    AddLog("==================================================================")

    -- Ghi file kết quả vào output.txt theo đúng quy chuẩn pull của push_test.ps1
    pcall(function()
        local fullContent = table.concat(logLines, "\n")
        local outPaths = {
            (CS.UnityEngine.Application and CS.UnityEngine.Application.persistentDataPath) and (CS.UnityEngine.Application.persistentDataPath .. "/output.txt"),
            "/storage/emulated/0/Android/data/com.vnyh.gp/files/output.txt"
        }
        for _, p in ipairs(outPaths) do
            if p and p ~= "" then
                pcall(function()
                    if CS.System.IO.File and CS.System.IO.File.WriteAllText then
                        CS.System.IO.File.WriteAllText(p, fullContent)
                    else
                        local f = io.open(p, "w")
                        if f then f:write(fullContent); f:close() end
                    end
                end)
            end
        end
    end)

    local msg = string.format("Đã quét %d trang bị! Xuất file output.txt thành công.", equipCount)
    if _G.FloatingWordUtility and _G.FloatingWordUtility.QuickMsg then
        _G.FloatingWordUtility.QuickMsg(msg)
    elseif _G.FloatingTipUtility and _G.FloatingTipUtility.QuickMsg then
        _G.FloatingTipUtility.QuickMsg(msg)
    end
end)
