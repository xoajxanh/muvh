-- SCRIPT THEO DÕI & CAN THIỆP TRỰC TIẾP: BẮT QUẢ TANG LỖI CON LẮC
-- Chạy script này một lần để bật LIVE MONITORING và THỬ CÁC BIỆN PHÁP CỨU HỘ

local report = {}
local function log(txt)
    local line = tostring(txt)
    table.insert(report, line)
    print(line)
end

log("==================================================")
log("=== LIVE MONITORING & THỬ NGHIỆM CAN THIỆP CON LẮC ===")
log("==================================================")

pcall(function()
    local me = _G.RoleManager and _G.RoleManager.me
    if not me then
        log("[ERROR] Không tìm thấy RoleManager.me!")
        return
    end

    -- ĐỌC TRẠNG THÁI HIỆN TẠI
    local curX, curY = me.cellPos.x, me.cellPos.y
    local sX, sY = me.serverCoord.x, me.serverCoord.y
    local nX, nY = (me.nextCell and me.nextCell.x or 0), (me.nextCell and me.nextCell.y or 0)
    local eX, eY = (me.moveEndCell and me.moveEndCell.x or 0), (me.moveEndCell and me.moveEndCell.y or 0)

    log(string.format("TỌA ĐỘ HIỆN TẠI: CellPos=(%d, %d) | ServerCoord=(%d, %d) | NextCell=(%d, %d) | EndCell=(%d, %d)", 
        curX, curY, sX, sY, nX, nY, eX, eY))

    -- 1. KIỂM TRA XEM Ô TIẾP THEO CÓ BỊ CHẶN (BLOCK / COLLISION) KHÔNG
    if me.nextCell and _G.Scene then
        local isBlock = _G.Scene.IsBlock and _G.Scene.IsBlock(me.nextCell)
        local canMove = _G.Scene.TileCanMoveToByRole and _G.Scene.TileCanMoveToByRole(me.nextCell)
        log(string.format("KIỂM TRA Ô NEXT (%d, %d): IsBlock=%s | CanMoveToByRole=%s", nX, nY, tostring(isBlock), tostring(canMove)))
    end

    -- 2. HOOK THEO DÕI MOVEFAILED & CHANGEPOS (BẮT SERVER ROLLBACK)
    if not _G.Mod_Hooked_ChangePos_Monitor and me.ChangePos then
        _G.Mod_Hooked_ChangePos_Monitor = true
        local old_ChangePos = me.ChangePos
        me.ChangePos = function(self, moveType, changeReason, reasonParam)
            local rName = tostring(changeReason)
            if _G.ERoleChangePosReason then
                for k, v in pairs(_G.ERoleChangePosReason) do
                    if v == changeReason then rName = k; break end
                end
            end
            print(string.format("[CHANGE_POS] Reason: %s | ServerCoord=(%s, %s)", rName, tostring(self.serverCoord.x), tostring(self.serverCoord.y)))

            -- NẾU LÀ MOVEFAILED (SERVER GIẬT LÙI LẠI) -> CHẶN ĐỨNG KHÔNG CHO GIẬT LÙI
            if _G.ERoleChangePosReason and changeReason == _G.ERoleChangePosReason.MoveFailed then
                print("[CAN THIỆP] -> PHÁT HIỆN SERVER GỬI MOVEFAILED! Chặn đứng giật lùi để phá vỡ con lắc!")
                if _G.FloatingWordUtility then
                    _G.FloatingWordUtility.QuickMsg("BẮT ĐƯỢC MOVEFAILED (SERVER ROLLBACK) -> ĐÃ CHẶN!")
                end
                -- Bỏ qua không gọi FlashTo giật lùi
                return
            end

            return old_ChangePos(self, moveType, changeReason, reasonParam)
        end
        log("-> ĐÃ HOOK BẢO VỆ CHỐNG MOVEFAILED (SERVER ROLLBACK)!")
    end

    -- 3. THỰC HIỆN FLASH VƯỢT QUA ĐIỂM KẸT (JUMP QUA 8 Ô VỀ PHÍA ĐÍCH)
    if eX ~= 0 and eY ~= 0 then
        -- Tính vector hướng tới đích
        local dirX = eX - curX
        local dirY = eY - curY
        local len = math.sqrt(dirX * dirX + dirY * dirY)
        if len > 0 then
            local jumpDist = math.min(10, math.floor(len))
            local jumpX = math.floor(curX + (dirX / len) * jumpDist)
            local jumpY = math.floor(curY + (dirY / len) * jumpDist)

            log(string.format("-> THỰC HIỆN JUMP VƯỢT KẸT: Flash từ (%d, %d) qua (%d, %d)...", curX, curY, jumpX, jumpY))

            if me.FlashTo then
                me:FlashTo(jumpX, jumpY)
            end
            if me.SetCell then
                me:SetCell(jumpX, jumpY)
            end

            -- Đồng bộ vị trí lên Server
            if _G.NetManager and _G.MapMessage and _G.MapMessage.ReqMove then
                _G.NetManager.Send(_G.MapMessage.ReqMove, {
                    x = jumpX,
                    y = jumpY,
                    action = 2,
                    time = _G.Time.GetServerTime()
                })
            end

            -- Tiếp tục chạy tới đích
            me.movePath = {}
            me.bestGuidePath = nil
            me.pathIndex = 1
            if me.MoveTo then
                me:MoveTo({ x = eX, y = eY }, 0)
                log(string.format("-> Đã MoveTo tiếp tục tới đích (%d, %d)!", eX, eY))
            end
        end
    end

    if _G.FloatingWordUtility then
        _G.FloatingWordUtility.QuickMsg("ĐÃ KÍCH HOẠT CHỐNG MOVEFAILED & JUMP VƯỢT KẸT!")
    end
end)

log("==================================================")
log("KẾT THÚC THỰC THI SCRIPT")

local finalStr = table.concat(report, "\n")
pcall(function()
    local path = "/storage/emulated/0/Android/data/com.vnyh.gp/files/output.txt"
    local f = io.open(path, "w")
    if f then
        f:write(finalStr)
        f:close()
    end
end)

return finalStr
