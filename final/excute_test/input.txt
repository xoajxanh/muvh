-- SCRIPT TEST TRICK MONITOR (FOV = 75)
-- Chạy script này để kiểm tra trực tiếp trạng thái Trick trên máy

local report = {}
local function log(txt)
    local line = tostring(txt)
    table.insert(report, line)
    print(line)
end

log("==================================================")
log("=== KIỂM TRA TRẠNG THÁI TRICK NHẶT ĐỒ (FOV = 75) ===")
log("==================================================")

pcall(function()
    -- 1. HỦY TIMER CŨ NẾU CÓ
    if _G.TrickMonitorTimer and _G.Timer and _G.Timer.Stop then
        _G.Timer.Stop(_G.TrickMonitorTimer)
        _G.TrickMonitorTimer = nil
        log("[INFO] Đã hủy Timer giám sát cũ.")
    end

    -- 2. HÀM CHECK ĐIỀU KIỆN TRICK
    local function CheckTrickStatus()
        local scopeVal = 0
        if _G.QiJiHelperData and _G.QiJiHelperData.SettingData then
            scopeVal = tonumber(_G.QiJiHelperData.SettingData.KillMonsterScope) or 0
        end

        local pickLimit = tonumber(_G.AutoPick_Limit) or 0

        local fovVal = 0
        if _G.SavedFOV then
            fovVal = tonumber(_G.SavedFOV) or 0
        elseif CS.UnityEngine.Camera.main then
            fovVal = tonumber(CS.UnityEngine.Camera.main.fieldOfView) or 0
        end
        local fovRounded = math.floor(fovVal + 0.5)

        local c1_ScopeEqualLimit = (scopeVal == pickLimit) and (scopeVal > 0)
        local c2_IsOddNumber     = (scopeVal % 2 == 1)
        local c3_IsFov75         = (fovRounded == 75)

        local isTrick = c1_ScopeEqualLimit and c2_IsOddNumber and c3_IsFov75

        return isTrick, scopeVal, pickLimit, fovRounded, c1_ScopeEqualLimit, c2_IsOddNumber, c3_IsFov75
    end

    -- 3. IN THÔNG TIN TỔNG QUAN HIỆN TẠI VÀO OUTPUT
    local isTrickNow, sVal, pLim, fVal, c1, c2, c3 = CheckTrickStatus()
    log(string.format("1. Phạm vi Treo máy (KillMonsterScope) : %s", tostring(sVal)))
    log(string.format("2. Số lượng nhặt Mod (AutoPick_Limit) : %s", tostring(pLim)))
    log(string.format("3. Camera FOV (SavedFOV / Cam.FOV)    : %s", tostring(fVal)))
    log("--------------------------------------------------")
    log(string.format("-> Điều kiện 1 (Phạm vi == Nhặt)      : %s", c1 and "ĐẠT [OK]" or "CHƯA ĐẠT [X]"))
    log(string.format("-> Điều kiện 2 (Giá trị là Số Lẻ)     : %s", c2 and "ĐẠT [OK]" or "CHƯA ĐẠT [X]"))
    log(string.format("-> Điều kiện 3 (FOV == 75)            : %s", c3 and "ĐẠT [OK]" or "CHƯA ĐẠT [X]"))
    log("--------------------------------------------------")
    if isTrickNow then
        log(">>> KẾT QUẢ: 🔥 TRICK ĐANG KÍCH HOẠT (ACTIVE)! <<<")
        if _G.FloatingWordUtility then
            _G.FloatingWordUtility.QuickMsg(string.format("🔥 TRICK ACTIVE! Scope=%d | Pick=%d | FOV=%d", sVal, pLim, fVal))
        end
    else
        log(">>> KẾT QUẢ: ❌ TRICK CHƯA KÍCH HOẠT! <<<")
        if _G.FloatingWordUtility then
            _G.FloatingWordUtility.QuickMsg(string.format("TRICK OFF: Scope=%d, Pick=%d, FOV=%d", sVal, pLim, fVal))
        end
    end

    -- 4. TẠO TIMER THEO DÕI LIVE (MỖI 0.5 GIÂY)
    _G.LastTrickAlertState = isTrickNow
    if _G.Timer and _G.Timer.StartLoop then
        _G.TrickMonitorTimer = _G.Timer.StartLoop(0.5, -1, function()
            local isActive, curScope, curPick, curFov = CheckTrickStatus()
            if isActive ~= _G.LastTrickAlertState then
                _G.LastTrickAlertState = isActive
                if isActive then
                    print(string.format("[TRICK MONITOR] 🔥 KÍCH HOẠT TRICK! Scope=%d, Pick=%d, FOV=%d", curScope, curPick, curFov))
                    if _G.FloatingWordUtility then
                        _G.FloatingWordUtility.QuickMsg(string.format("🔥 TRICK ĐÃ KÍCH HOẠT! (Scope: %d | Pick: %d | FOV: %d)", curScope, curPick, curFov))
                    end
                else
                    print(string.format("[TRICK MONITOR] ❌ TRICK ĐÃ TẮT! Scope=%d, Pick=%d, FOV=%d", curScope, curPick, curFov))
                    if _G.FloatingWordUtility then
                        _G.FloatingWordUtility.QuickMsg(string.format("❌ TRICK TẮT: Scope=%d | Pick=%d | FOV=%d", curScope, curPick, curFov))
                    end
                end
            end
        end)
        log("[INFO] Đã khởi chạy Live Monitor Timer (chu kỳ 0.5s). Khi bạn chỉnh thông số khớp trick, game sẽ tự động hiện thông báo nổi!")
    end
end)

log("==================================================")
log("KẾT THÚC THỰC THI SCRIPT")

local finalStr = table.concat(report, "\n")
pcall(function()
    local path = "/storage/emulated/0/Android/data/com.vnyh.gp/files/output.txt"
    local f = io.open(path, "w")
    if f then
        f:write(finalStr)
        f:close()
    end
end)

return finalStr
