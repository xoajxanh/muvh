-- SCRIPT TEST: CHẠY MƯỢT MÀ (120 FPS + PHYSICS + ANIMATION 1.0X + ĐỒNG BỘ CÀI ĐẶT GAME)

local function TestSmoothRun()
    local report = {}
    local function log(txt)
        table.insert(report, tostring(txt))
    end

    log("==================================================")
    log("=== TEST KÍCH HOẠT CHẠY MƯỢT MÀ (120 FPS) ===")
    log("==================================================")

    -- 1. ĐỌC THÔNG SỐ BAN ĐẦU
    pcall(function()
        local app = CS.UnityEngine.Application
        local qs = CS.UnityEngine.QualitySettings
        local t = CS.UnityEngine.Time
        local scr = CS.UnityEngine.Screen

        log("--- THÔNG SỐ HIỆN TẠI TRƯỚC KHI SET ---")
        log("Target FrameRate: " .. tostring(app.targetFrameRate))
        log("vSyncCount: " .. tostring(qs.vSyncCount))
        log("fixedDeltaTime: " .. tostring(t.fixedDeltaTime))
        log("maximumDeltaTime: " .. tostring(t.maximumDeltaTime))
        log("Screen Resolution: " .. tostring(scr.width) .. "x" .. tostring(scr.height))
        log("Refresh Rate: " .. tostring(scr.currentResolution.refreshRate or "N/A"))
        if _G.GameSettingsData then
            log("GameSettingsData.frameRate: " .. tostring(_G.GameSettingsData.frameRate))
        end
    end)

    -- 2. THỰC HIỆN SET THÔNG SỐ CHẠY MƯỢT (120 FPS)
    pcall(function()
        local app = CS.UnityEngine.Application
        local qs = CS.UnityEngine.QualitySettings
        local t = CS.UnityEngine.Time

        app.targetFrameRate = 120
        qs.vSyncCount = 0
        t.fixedDeltaTime = 1.0 / 120.0
        t.maximumDeltaTime = 0.1

        -- Đồng bộ với GameSettingsController
        if _G.GameSettingsController and _G.GameSettingsController.SetFrameRate then
            _G.GameSettingsController.SetFrameRate(120)
            log("-> Đã gọi GameSettingsController.SetFrameRate(120)")
        elseif _G.GameSettingsData then
            _G.GameSettingsData.frameRate = 120
            log("-> Đã gán GameSettingsData.frameRate = 120")
        end

        _G.Mod_SmoothRun_Enabled = true
        CS.UnityEngine.PlayerPrefs.SetInt("Mod_SmoothRun_Enabled", 1)
        CS.UnityEngine.PlayerPrefs.Save()
        log("-> Đã set PlayerPrefs Mod_SmoothRun_Enabled = 1")
    end)

    -- 3. KHÓA TỐC ĐỘ ANIMATION BODY VÀ CÁNH VỀ CHUẨN 1.0X KHI CHẠY NHANH
    pcall(function()
        if _G.AnimatorCtrl and not _G.Mod_Hooked_AnimatorCtrl then
            _G.Mod_Hooked_AnimatorCtrl = true
            local old_SetAnimatorSpeed = _G.AnimatorCtrl.SetAnimatorSpeed
            _G.AnimatorCtrl.SetAnimatorSpeed = function(self, speed)
                if (_G.Mod_IsActive and _G.Mod_IsActive()) and self.avatar and self.avatar.isMe then
                    speed = 1.0
                end
                return old_SetAnimatorSpeed(self, speed)
            end
            log("-> Đã hook AnimatorCtrl.SetAnimatorSpeed = 1.0x")
        end

        if _G.RoleEquip and not _G.Mod_Hooked_RoleEquipWing then
            _G.Mod_Hooked_RoleEquipWing = true
            local old_SetWingAni = _G.RoleEquip.SetWingAni
            _G.RoleEquip.SetWingAni = function(self, RoleMoveType, isCurIsSafeZone)
                old_SetWingAni(self, RoleMoveType, isCurIsSafeZone)
                if (_G.Mod_IsActive and _G.Mod_IsActive()) and self.avatar and self.avatar.isMe then
                    if self.wingAnimator and self.wingAnimator.animator and not IsNil(self.wingAnimator.animator) then
                        self.wingAnimator.animator.speed = 1.0
                    end
                end
            end
            log("-> Đã hook RoleEquip.SetWingAni = 1.0x")
        end
    end)

    -- 4. ĐỌC LẠI THÔNG SỐ SAU KHI ÁP DỤNG
    pcall(function()
        local app = CS.UnityEngine.Application
        local qs = CS.UnityEngine.QualitySettings
        local t = CS.UnityEngine.Time

        log("\n--- THÔNG SỐ SAU KHI ÁP DỤNG ---")
        log("Target FrameRate: " .. tostring(app.targetFrameRate))
        log("vSyncCount: " .. tostring(qs.vSyncCount))
        log("fixedDeltaTime: " .. tostring(t.fixedDeltaTime))
        log("maximumDeltaTime: " .. tostring(t.maximumDeltaTime))
        if _G.GameSettingsData then
            log("GameSettingsData.frameRate: " .. tostring(_G.GameSettingsData.frameRate))
        end
    end)

    -- 5. XUẤT RA FILE OUTPUT.TXT VÀ BẮN THÔNG BÁO QUICKMSG
    local finalStr = table.concat(report, "\n")
    pcall(function()
        local logPath = CS.UnityEngine.Application.persistentDataPath .. "/output.txt"
        local f = io.open(logPath, "w")
        if f then
            f:write(finalStr)
            f:close()
        end
        if _G.FloatingWordUtility then
            _G.FloatingWordUtility.QuickMsg("ĐÃ KÍCH HOẠT CHẠY MƯỢT (120 FPS / MƯỢT HÓA)!")
        end
    end)
end

TestSmoothRun()
