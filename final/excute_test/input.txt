local package_name = "com.vnyh.gp" 
local outputPath = "/storage/emulated/0/Android/data/" .. package_name .. "/files/output.txt"

local logs = {}
local function log(msg)
    table.insert(logs, tostring(msg))
    print("[EXEC_HTTP_TEST] " .. tostring(msg))
end

log("==========================================")
log("  TEST HTTP API FETCH & FULL DECRYPT      ")
log("==========================================")

local httpUrl = "http://g3events.asia/api/v1/config?sn=7cf7e3df6fd19f8dbd5c54bb65803680&uid=176827261712166000"
log("Testing URL: " .. httpUrl)

pcall(function()
    local req = CS.UnityEngine.Networking.UnityWebRequest.Get(httpUrl)
    req.timeout = 10
    local asyncOp = req:SendWebRequest()
    
    local count = 0
    if _G.Timer and _G.Timer.StartLoop then
        _G.Timer.StartLoop(0.2, 50, function()
            count = count + 1
            if asyncOp.isDone or count >= 50 then
                pcall(function()
                    log("isDone: " .. tostring(asyncOp.isDone) .. " (Loops: " .. tostring(count) .. ")")
                    log("Response Code: " .. tostring(req.responseCode))
                    log("isNetworkError: " .. tostring(req.isNetworkError))
                    log("isHttpError: " .. tostring(req.isHttpError))
                    log("Error Msg: " .. tostring(req.error))
                    if req.downloadHandler then
                        local jsonBody = req.downloadHandler.text
                        log("API Response Body: " .. tostring(jsonBody))
                    end
                end)
                
                local fullOutput = table.concat(logs, "\n")
                local f = io.open(outputPath, "w")
                if f then f:write(fullOutput); f:close() end
                if _G.FloatingWordUtility then _G.FloatingWordUtility.QuickMsg("TEST HTTP API HOÀN TẤT!") end
            end
        end)
    end
end)

return "Đang test API HTTP..."
