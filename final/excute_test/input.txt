local function LogAndShow(msg)
    print(msg)
    pcall(function()
        if _G.FloatingWordUtility and _G.FloatingWordUtility.QuickMsg then
            _G.FloatingWordUtility.QuickMsg(msg)
        end
    end)
end

if _G.TestTrainTimer then
    pcall(function() _G.Timer.Stop(_G.TestTrainTimer) end)
    _G.TestTrainTimer = nil
end

_G.Test_IsMovingToTrainPos = false

local targetCoord = "129#88"
local parts = {}
for p in string.gmatch(targetCoord, "[^#]+") do table.insert(parts, p) end
local tx, ty = tonumber(parts[1]), tonumber(parts[2])

LogAndShow(string.format("[TEST STRICT] Bắt đầu dẫn đường tới CHÍNH XÁC (%d,%d)...", tx, ty))

local loopCount = 0
if _G.Timer and _G.Timer.StartLoopForever then
    _G.TestTrainTimer = _G.Timer.StartLoopForever(0.3, function()
        loopCount = loopCount + 1
        local pMe = _G.RoleManager and _G.RoleManager.me
        local meX, meY = 0, 0
        if pMe then
            if pMe.cellPos then meX, meY = pMe.cellPos.x, pMe.cellPos.y
            elseif pMe.serverCoord then meX, meY = pMe.serverCoord.x, pMe.serverCoord.y end
        end
        
        local dx = meX - tx
        local dy = meY - ty
        local dist = math.sqrt(dx*dx + dy*dy)
        
        LogAndShow(string.format("[TEST #%d] Vị trí hiện tại: (%d,%d) -> Tọa độ bãi: (%d,%d) | Cự ly: %.1fm", loopCount, meX, meY, tx, ty, dist))
        
        -- CHƯA TỚI CHÍNH XÁC TỌA ĐỘ (Cự ly > 1.5m)
        if dist > 1.5 then
            -- TẮT HOÀN TOÀN AUTO FIGHT TRÊN ĐƯỜNG CHẠY (KHÔNG ĐÁNH QUÁI)
            if pMe then
                if pMe.isAutoTaskFight and pMe.isAutoTaskFight ~= "None" then
                    if pMe.SetAutoTaskFight then pMe:SetAutoTaskFight("None") end
                end
                if pMe.isAutoFight and pMe.isAutoFight ~= "None" then
                    if pMe.SetAutoFight then pMe:SetAutoFight("None") end
                end
            end
            if _G.QiJiHelperData and _G.QiJiHelperData.SetAutoFightData then
                _G.QiJiHelperData.SetAutoFightData(false)
            end
            
            -- Xử lý đá nhảy nếu quá xa (> 50m)
            if dist > 50 then
                _G.Test_IsMovingToTrainPos = false
                local stoneBagId = nil
                if _G.BagInfoData and _G.BagInfoData.TotalItems then
                    for _, itemData in pairs(_G.BagInfoData.TotalItems) do
                        if itemData then
                            local itemId = itemData.itemId or (itemData.data and itemData.data.itemId)
                            local instanceId = itemData.id or (itemData.data and itemData.data.id)
                            if itemId == 20000022 then
                                stoneBagId = instanceId
                                break
                            end
                        end
                    end
                end
                
                if stoneBagId then
                    LogAndShow(string.format("[TEST] Cự ly %.1fm > 50m -> Dùng Đá Nhảy!", dist))
                    if _G.networkRequest and _G.networkRequest.ReqUseItem then
                        _G.networkRequest.ReqUseItem(1, stoneBagId)
                    elseif _G.BagInfoController and _G.BagInfoController.UseItemReq then
                        _G.BagInfoController.UseItemReq(1, stoneBagId, nil, 20000022)
                    end
                else
                    LogAndShow(string.format("[TEST] Cự ly %.1fm > 50m (Không có đá) -> Dùng MoveToLinePos...", dist))
                    local tab = _G.ModAutoBossConfigTab or "C7"
                    local mapsConfig = (tab == "C8") and _G.Mod_MapsConfig_c8 or _G.Mod_MapsConfig_c7
                    local wildMapId = (mapsConfig and mapsConfig[1] and mapsConfig[1].mapId) or ((tab == "C8") and 1074 or 101096)
                    local wildTransferId = (mapsConfig and mapsConfig[1] and mapsConfig[1].bosses and mapsConfig[1].bosses[1] and mapsConfig[1].bosses[1].transferId) or ((tab == "C8") and 400229 or 400216)
                    if _G.PathFinderManager and _G.PathFinderManager.MoveToLinePos then
                        _G.PathFinderManager.MoveToLinePos(wildMapId, {x = tx, y = ty}, wildTransferId, 1, nil, nil, nil, nil, true)
                    end
                end
            else
                -- Cự ly <= 50m và > 1.5m: Kiểm tra xem nhân vật có đang di chuyển không
                local isMoving = pMe and pMe.IsMoving and pMe:IsMoving()
                
                if not _G.Test_IsMovingToTrainPos or not isMoving then
                    _G.Test_IsMovingToTrainPos = true
                    LogAndShow(string.format("[TEST] Chạy bộ tới tọa độ (%d,%d)... (cự ly %.1fm)", tx, ty, dist))
                    
                    if _G.PathFinderManager and _G.PathFinderManager.JumpMapToMoveToPos and _G.SceneData then
                        local targetPosData = (_G.PathFinderManager.GetCalcPosData and _G.PathFinderManager.GetCalcPosData(targetCoord)) or (_G.Vector2 and _G.Vector2(tx, ty)) or {x = tx, y = ty}
                        _G.PathFinderManager.JumpMapToMoveToPos(_G.SceneData.groupId, targetPosData, nil, nil, nil, Purpose.None or 0, nil, 1, true)
                    elseif pMe and pMe.MoveTo then
                        pMe:MoveTo({x = tx, y = ty}, 0)
                    end
                else
                    LogAndShow(string.format("[TEST] Đang duy trì chạy mượt tới (%d,%d)... (cự ly %.1fm)", tx, ty, dist))
                end
            end
        else
            -- ĐÃ ĐẾN CHÍNH XÁC VỊ TRÍ TỌA ĐỘ (Cự ly <= 1.5m)
            LogAndShow(string.format("[TEST] -> ĐÃ ĐẾN CHÍNH XÁC TỌA ĐỘ BÃI TRAIN (%d,%d) (Cự ly %.1fm)! Dừng di chuyển & Bật Auto Fight...", tx, ty, dist))
            if pMe then
                if pMe.StopMove then pMe:StopMove() end
                if pMe.SetAutoFight then pMe:SetAutoFight("ReleaseSkill") end
            end
            if _G.QiJiHelperData and _G.QiJiHelperData.SetAutoFightData then
                _G.QiJiHelperData.SetAutoFightData(true)
            end
            _G.Test_IsMovingToTrainPos = false
            if _G.TestTrainTimer then
                _G.Timer.Stop(_G.TestTrainTimer)
                _G.TestTrainTimer = nil
            end
        end
        
        if loopCount >= 40 then
            LogAndShow("[TEST] Hết 40s -> Tự động dừng Timer test.")
            if _G.TestTrainTimer then
                _G.Timer.Stop(_G.TestTrainTimer)
                _G.TestTrainTimer = nil
            end
        end
    end)
end
