local output = {}
local function Log(msg)
    table.insert(output, msg)
    if FloatingWordUtility and FloatingWordUtility.QuickMsg then
        pcall(FloatingWordUtility.QuickMsg, msg)
    end
end

local function SaveOutput()
    pcall(function()
        local path = CS.UnityEngine.Application.persistentDataPath .. "/output.txt"
        local content = table.concat(output, "\n")
        CS.System.IO.File.WriteAllText(path, content)
    end)
end

-- 2. HÀM TÌM KIẾM BOSS NỮ HOÀNG BĂNG GIÁ LV80
-- =========================================================================
local function FindBossNuHoangBangGia()
    -- Cách 1: Qua RoleManager.GetMonsterRoleByConfigId
    if RoleManager and RoleManager.GetMonsterRoleByConfigId then
        local list = RoleManager.GetMonsterRoleByConfigId({ 210100100, 210100101 })
        if list and #list > 0 then
            for _, m in ipairs(list) do
                if m and m.hp and m.hp > 0 then
                    return m
                end
            end
        end
    end

    -- Cách 2: Quét toàn bộ Monster trong Scene
    if RoleManager and RoleManager.GetRolesByType and ERoleType and ERoleType.Monster then
        local mTable = RoleManager.GetRolesByType(ERoleType.Monster)
        if mTable then
            for _, m in pairs(mTable) do
                if m and m.hp and m.hp > 0 then
                    local cfgId = (m.GetConfigId and m:GetConfigId()) or (m.data and (m.data.monsterId or m.data.configId))
                    local mName = tostring(m.name or (m.data and m.data.name) or "")
                    if cfgId == 210100100 or cfgId == 210100101 or string.find(mName, "Nữ Hoàng") or string.find(mName, "Băng Giá") or string.find(mName, "Nu Hoang") then
                        return m
                    end
                end
            end
        end
    end

    return nil
end

-- 3. HÀM VÀO ROOM BOSS PHÚC LỢI TẦNG 1 (CHUẨN FLOW UI USER TEST)
-- =========================================================================
local function DoEnterSecretRoom(callback)
    CleanAllTooltips()
    Log("[Room] Đang kích hoạt vào Room Boss Phúc Lợi Tầng 1...")

    pcall(function()
        if networkRequest then
            if networkRequest.ReqGetBossMapAndCount then networkRequest.ReqGetBossMapAndCount() end
            if networkRequest.ReqInstanceCoolDown then networkRequest.ReqInstanceCoolDown(1106) end
            if networkRequest.ReqBossStateByType and MonsterBossType then
                networkRequest.ReqBossStateByType(MonsterBossType.secretBoss)
            end
        end
    end)

    pcall(function()
        if UIManager and UIID and UIID.Instance_BossUI then
            UIManager.Show(UIID.Instance_BossUI, { openFirstTab = 7, Monsterid = 210100100 })
        end
    end)

    local timerObj = _G.Timer or Timer
    if timerObj and timerObj.Start then
        timerObj.Start(0.4, function()
            pcall(function()
                local bossUI = UIManager and UIManager.GetUI and UIID and UIManager.GetUI(UIID.Instance_BossUI)
                local secretTemp = bossUI and bossUI.secretBossTemplate

                if secretTemp and secretTemp.EnterSecret then
                    secretTemp:EnterSecret(210100001, { "31", "222" })
                else
                    if PathFinderManager and PathFinderManager.FlyTransferScene then
                        PathFinderManager.FlyTransferScene(210100001)
                    end
                end

                if UIManager and UIID and UIID.Instance_BossUI then
                    UIManager.Hide(UIID.Instance_BossUI)
                end
            end)

            if callback then
                timerObj.Start(2.2, callback)
            end
        end)
    else
        pcall(function()
            if PathFinderManager and PathFinderManager.FlyTransferScene then
                PathFinderManager.FlyTransferScene(210100001)
            end
        end)
        if callback then callback() end
    end
end

-- =========================================================================
-- 4. VÒNG LẶP STATE MACHINE FARM KIM CƯƠNG LIÊN TỤC
-- =========================================================================
_G.Test_FarmKCActive = true
_G.Test_FarmKCCount = _G.Test_FarmKCCount or 0

-- Dừng timer cũ nếu đang chạy
if _G.Test_FarmKCTimer then
    pcall(function() Timer.Stop(_G.Test_FarmKCTimer) end)
    _G.Test_FarmKCTimer = nil
end

local currentState = "INIT"  -- "INIT", "WAIT_MAP", "CHECK_BOSS", "FIGHTING", "LOOTING"
local stateTime = 0
local activeBoss = nil

local function StepLoop()
    if not _G.Test_FarmKCActive then
        Log("[-] ĐÃ DỪNG VÒNG LẶP FARM KIM CƯƠNG!")
        if _G.Test_FarmKCTimer then
            Timer.Stop(_G.Test_FarmKCTimer)
            _G.Test_FarmKCTimer = nil
        end
        SaveOutput()
        return
    end

    -- Luôn dọn sạch các tooltip hiện lên màn hình
    CleanAllTooltips()

    stateTime = stateTime + 0.5

    if currentState == "INIT" then
        currentState = "WAIT_MAP"
        stateTime = 0
        activeBoss = nil
        DoEnterSecretRoom(function()
            currentState = "CHECK_BOSS"
            stateTime = 0
        end)

    elseif currentState == "CHECK_BOSS" then
        local boss = FindBossNuHoangBangGia()
        if boss and boss.hp and boss.hp > 0 then
            activeBoss = boss
            _G.Test_FarmKCCount = _G.Test_FarmKCCount + 1
            currentState = "FIGHTING"
            stateTime = 0

            Log(string.format(">>> [FARM KC #%d]: PHÁT HIỆN NỮ HOÀNG BĂNG GIÁ! BẮT ĐẦU ĐÁNH! <<<", _G.Test_FarmKCCount))
            
            -- Bật Auto đánh boss
            pcall(function()
                if RoleManager and RoleManager.me and RoleManager.me.SetAutoTaskFight and AutoFightStrKey then
                    RoleManager.me:SetAutoTaskFight(AutoFightStrKey.AutoFight)
                end
            end)
            SaveOutput()
        else
            -- Không có boss trong Room -> Vào lại Room mới ngay lập tức!
            Log("[-] Không có Boss trong Room! Đang chuyển Room khác ngay...")
            currentState = "WAIT_MAP"
            stateTime = 0
            DoEnterSecretRoom(function()
                currentState = "CHECK_BOSS"
                stateTime = 0
            end)
        end

    elseif currentState == "FIGHTING" then
        -- Kiểm tra tình trạng Boss
        local isAlive = false
        if activeBoss then
            pcall(function()
                if activeBoss.hp and activeBoss.hp > 0 and not activeBoss.isDead then
                    isAlive = true
                end
            end)
        end

        if not isAlive or stateTime >= 45.0 then
            Log(string.format(">>> BOSS #%d ĐÃ BỊ TIÊU DIỆT! CHỜ 1.5S NHẶT KC VÀ VẬT PHẨM... <<<", _G.Test_FarmKCCount))
            currentState = "LOOTING"
            stateTime = 0
            SaveOutput()
        else
            -- Đảm bảo AutoFight vẫn chạy trong lúc đánh
            if math.floor(stateTime) % 3 == 0 then
                pcall(function()
                    if RoleManager and RoleManager.me and RoleManager.me.SetAutoTaskFight and AutoFightStrKey then
                        RoleManager.me:SetAutoTaskFight(AutoFightStrKey.AutoFight)
                    end
                end)
            end
        end

    elseif currentState == "LOOTING" then
        -- Chờ 1.5s nhặt đồ rơi
        if stateTime >= 1.5 then
            Log("[+] Nhặt xong! Kích hoạt vào Room mới...")
            currentState = "WAIT_MAP"
            stateTime = 0
            activeBoss = nil
            DoEnterSecretRoom(function()
                currentState = "CHECK_BOSS"
                stateTime = 0
            end)
        end

    elseif currentState == "WAIT_MAP" then
        -- Watchdog phòng trường hợp chuyển map quá 6s
        if stateTime >= 6.0 then
            Log("[Watchdog] Vào room quá 6s, thử kích hoạt lại...")
            currentState = "CHECK_BOSS"
            stateTime = 0
        end
    end
end
