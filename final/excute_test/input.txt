-- =========================================================================
-- [EXECUTE TEST SCRIPT]: BỘ KIỂM THỬ TOÀN DIỆN CHO FARM BOT (TẠO NV, ĐẶT TÊN, LOGIN, AUTO FARM)
-- =========================================================================

pcall(function()
    local logLines = {}
    local function AddLog(msg)
        table.insert(logLines, msg)
        if CS and CS.UnityEngine and CS.UnityEngine.Debug and CS.UnityEngine.Debug.Log then
            CS.UnityEngine.Debug.Log("[TEST_FARM] " .. tostring(msg))
        end
        if _G.FloatingWordUtility and _G.FloatingWordUtility.QuickMsg then
            _G.FloatingWordUtility.QuickMsg(tostring(msg))
        end
    end

    local function SaveOutput()
        local fullContent = table.concat(logLines, "\n")
        local p = "/storage/emulated/0/Android/data/com.vnyh.gp/files/output.txt"
        pcall(function()
            if CS.System.IO.File and CS.System.IO.File.WriteAllText then
                CS.System.IO.File.WriteAllText(p, fullContent)
            else
                local f = io.open(p, "w")
                if f then f:write(fullContent); f:close() end
            end
        end)
    end

    AddLog("=========================================================================")
    AddLog(">>> [EXECUTE TEST]: BẮT ĐẦU KIỂM TRA & KÍCH HOẠT QUY TRÌNH FARM <<<")
    AddLog("=========================================================================")

    -- Helper sinh tên độc nhất
    local function GenName()
        local NamePrefixes = { "Rex", "Leo", "Kai", "Ken", "Zin", "Max", "Vut", "Roy", "Ben", "Sam", "Tom", "Zen", "Lux", "Fox", "Dan", "Jin", "Jay", "Ron", "Ace", "Neo", "Sky", "Ray", "Taj", "Kev", "Ian", "Eli", "Guy", "Lee", "Ted", "Val", "Zak" }
        local p = NamePrefixes[math.random(1, #NamePrefixes)]
        local t = (os.time() + math.random(1, 999)) % 46656
        local b36 = "0123456789abcdefghijklmnopqrstuvwxyz"
        local s = ""
        for i = 1, 3 do
            local rem = t % 36
            s = string.sub(b36, rem + 1, rem + 1) .. s
            t = math.floor(t / 36)
        end
        return p .. s
    end

    -- 1. KIỂM TRA ĐANG Ở TRONG GAME WORLD
    local isInGame = false
    local pMe = _G.RoleManager and _G.RoleManager.me
    if (_G.LoginData and _G.LoginData.InGame) or (pMe and (pMe.level or (pMe.data and pMe.data.level))) then
        isInGame = true
    end

    if isInGame then
        local lvl = (pMe and pMe.level) or (pMe and pMe.data and pMe.data.level) or 1
        local rName = (pMe and pMe.name) or (pMe and pMe.data and pMe.data.name) or "Me"
        AddLog(string.format("-> [TRONG GAME]: Nhân vật %s (Cấp %d) đang trong thế giới game!", tostring(rName), lvl))

        -- Kiểm tra nếu đang kẹt ở giao diện Thu Hồi (Task 3010)
        local curTask = _G.TaskData and _G.TaskData.curMainTask
        local taskId = (curTask and curTask.id) or 0
        local isSellUIOpen = false
        pcall(function()
            if _G.UIManager and _G.UIManager.IsVisible then
                isSellUIOpen = _G.UIManager.IsVisible("Bag_SellInfoUI") or (_G.UIID and _G.UIID.Bag_SellInfoUI and _G.UIManager.IsVisible(_G.UIID.Bag_SellInfoUI))
            end
        end)

        if taskId == 3010 or isSellUIOpen then
            AddLog("-> Phát hiện nhiệm vụ Thu Hồi (Task 3010)! Đang tự động giải phóng...")
            local candidate = nil
            if _G.BagInfoData and _G.BagInfoData.TotalItems then
                for _, item in pairs(_G.BagInfoData.TotalItems) do
                    if item and item.id and item.count and item.count > 0 and item.tblItem then
                        if item.tblItem.type == 1 then candidate = item break end
                    end
                end
                if not candidate then
                    for _, item in pairs(_G.BagInfoData.TotalItems) do
                        if item and item.id and item.count and item.count > 0 then candidate = item break end
                    end
                end
            end

            local wayType = (_G.RecycleWayType and _G.RecycleWayType.BlackSmith) or 2
            if candidate then
                local sendMap = {}
                sendMap[candidate.id] = 1
                AddLog(string.format("-> Gửi lệnh thu hồi 1 item (%s) để xong nhiệm vụ...", tostring(candidate.tblItem and candidate.tblItem.name or candidate.id)))
                if _G.networkRequest and _G.networkRequest.ReqItemRecycle then
                    _G.networkRequest.ReqItemRecycle(sendMap, wayType)
                elseif _G.NetManager and _G.ItemRecycleMessage and _G.ItemRecycleMessage.ReqItemRecycle then
                    _G.NetManager.Send(_G.ItemRecycleMessage.ReqItemRecycle, { recycleItems = sendMap, recycleType = wayType })
                end
            end

            Timer.Start(1.0, function()
                pcall(function()
                    if _G.networkRequest then
                        if _G.networkRequest.ReqCompleteTask then _G.networkRequest.ReqCompleteTask(3010) end
                        if _G.networkRequest.ReqSubmitTask then _G.networkRequest.ReqSubmitTask(3010) end
                    end
                    if _G.UIManager then
                        if _G.UIID and _G.UIID.Bag_SellInfoUI then _G.UIManager.Hide(_G.UIID.Bag_SellInfoUI) end
                        if _G.UIID and _G.UIID.BagUI then _G.UIManager.Hide(_G.UIID.BagUI) end
                        _G.UIManager.Hide("Bag_SellInfoUI")
                        _G.UIManager.Hide("Bag_3DBagInfoUI")
                    end
                end)
            end)
        end


        AddLog("-> Kích hoạt chu trình Bot Auto Farm & Quest Pipeline...")
        if _G.StartAutoFarmBot then
            _G.StartAutoFarmBot()
        end
        SaveOutput()
        return
    end

    -- 2. KIỂM TRA ĐANG Ở MÀN HÌNH TẠO NHÂN VẬT (LoginCreateRoleUI)
    local isAtCreateUI = false
    local createUI = nil
    pcall(function()
        if _G.UIManager and _G.UIManager.GetUiByName then
            createUI = _G.UIManager.GetUiByName("Login_LoginCreateRoleUI")
        end
        if _G.UIManager and _G.UIManager.IsVisible then
            if _G.UIManager.IsVisible("Login_LoginCreateRoleUI") or (_G.UIID and _G.UIID.LoginCreateRoleUI and _G.UIManager.IsVisible(_G.UIID.LoginCreateRoleUI)) then
                isAtCreateUI = true
            end
        end
        if createUI and not IsNil(createUI) then
            isAtCreateUI = true
        end
    end)

    if isAtCreateUI then
        local newName = GenName()
        local targetCareer = 13 -- Cung Thủ (Archer)
        local targetSex = 1 -- BẮT BUỘC 1

        AddLog(string.format("-> [MÀN HÌNH TẠO NHÂN VẬT]: Phát hiện LoginCreateRoleUI! Đang tạo Cung Thủ tên: %s", newName))
        if _G.LoginData then _G.LoginData.playerCreateName = newName end

        if createUI and not IsNil(createUI) then
            pcall(function()
                if createUI.SelectCareer then createUI:SelectCareer(targetCareer) end
                if createUI.tog_archer and createUI.tog_archer.SetIsOn then createUI.tog_archer:SetIsOn(true) end
                if createUI.InputField_Name and createUI.InputField_Name.SetInputText then
                    createUI.InputField_Name:SetInputText(newName)
                end
                if createUI.Button_OkOnClick then
                    AddLog("-> Kích hoạt Button_OkOnClick trực tiếp trên UI...")
                    createUI:Button_OkOnClick()
                end
            end)
        else
            AddLog("-> Gửi gói tin ReqCreateRole trực tiếp tới server...")
            if _G.networkRequest and _G.networkRequest.ReqCreateRole then
                _G.networkRequest.ReqCreateRole(newName, targetSex, targetCareer)
            elseif _G.NetManager and _G.NetManager.Send and _G.UserMessage and _G.UserMessage.ReqCreateRole then
                _G.NetManager.Send(_G.UserMessage.ReqCreateRole, { roleName = newName, sex = targetSex, career = targetCareer })
            end
        end
        SaveOutput()
        return
    end

    -- 3. KIỂM TRA ĐANG Ở MÀN HÌNH CHỌN NHÂN VẬT (LoginRoleUI)
    local isAtSelectUI = false
    local roleCount = 0
    pcall(function()
        if _G.LoginData and _G.LoginData.roleList then
            roleCount = #_G.LoginData.roleList
        end
        if _G.UIManager and _G.UIManager.IsVisible then
            if _G.UIManager.IsVisible("Login_LoginRoleUI") or (_G.UIID and _G.UIID.LoginRoleUI and _G.UIManager.IsVisible(_G.UIID.LoginRoleUI)) then
                isAtSelectUI = true
            end
        end
    end)

    if isAtSelectUI or roleCount > 0 then
        if roleCount > 0 then
            AddLog(string.format("-> [CHỌN NHÂN VẬT]: Tìm thấy %d nhân vật -> Đang chọn Slot 1 để vào game...", roleCount))
            if _G.SelectRoleByIndex then
                _G.SelectRoleByIndex(1)
            end
        else
            AddLog("-> [BỆ ĐÁ TRỐNG]: Chưa có nhân vật nào -> Tự động gọi CreateRoleAndEnterGame(1)...")
            if _G.CreateRoleAndEnterGame then
                _G.CreateRoleAndEnterGame(1)
            end
        end
        SaveOutput()
        return
    end

    -- 4. ĐANG Ở MÀN HÌNH SPLASH / LOGIN
    AddLog("-> [MÀN HÌNH SPLASH / LOGIN]: Bắt đầu tiến trình Đăng nhập tự động & Kết nối Server...")
    if _G.StartAutoFarmBot then
        _G.StartAutoFarmBot()
    end
    SaveOutput()
end)
