-- =========================================================================
-- [TEST SCRIPT]: TEST NHẬN NHIỆM VỤ "CẤP ĐẠT 80" (ID: 3104) GÂY LAG (1/1)
-- Mô tả: Mô phỏng chính xác hành vi khi click vào khung nhiệm vụ trên màn hình
-- (Chỉ gửi ReqCompleteTask / Task_BtnRewardClick hoặc gọi TaskItemClick,
--  TUYỆT ĐỐI KHÔNG gửi ReqSubmitTask để nhiệm vụ bị kẹt ở Đang tiến hành 1/1)
-- =========================================================================

local logs = {}
local function log(msg)
    table.insert(logs, tostring(msg))
end

local function saveOutput()
    local path = CS.UnityEngine.Application.persistentDataPath .. "/output.txt"
    local f = io.open(path, "w")
    if f then
        f:write(table.concat(logs, "\n"))
        f:close()
    end
end

local targetId = 3104 -- Nhiệm vụ Chính: Cấp đạt 80
local taskObj = _G.TaskData and _G.TaskData.GetTaskById and _G.TaskData.GetTaskById(targetId)

-- Dự phòng tìm theo tên nếu ID khác 3104
if not taskObj and _G.TaskData and _G.TaskData.GetAllTasks then
    local ok, all = pcall(_G.TaskData.GetAllTasks)
    if ok and type(all) == "table" then
        for _, v in pairs(all) do
            if type(v) == "table" and v.taskTbl then
                local name = tostring(v.taskTbl.name or "")
                if string.find(name, "80") then
                    taskObj = v
                    targetId = v.taskId
                    break
                end
            end
        end
    end
end

log("=== TEST NHAN NHIEM VU CAP DAT 80 (ID: " .. tostring(targetId) .. ") ===")
if taskObj then
    local name = taskObj.taskTbl and taskObj.taskTbl.name or "?"
    local stateBefore = taskObj.GetState and taskObj:GetState() or taskObj.state
    local curGoal = (taskObj.goals and taskObj.goals[1] and taskObj.goals[1].curCount) or "?"
    local maxGoal = (taskObj.goals and taskObj.goals[1] and taskObj.goals[1].maxCount) or "?"
    log(string.format("[BEFORE] ID=%s | Name=%s | State=%s | Goal=%s/%s",
        tostring(targetId), tostring(name), tostring(stateBefore), tostring(curGoal), tostring(maxGoal)))
else
    log("[WARN] Khong tim thay task 3104 trong TaskData. Vẫn thử gửi lệnh với ID = 3104.")
end

-- Cách 1: Gọi trực tiếp hàm click item trên bảng Task_TaskUI (giống hệt tay bấm vào khung nhiệm vụ)
local clickedUI = false
if _G.UIManager and _G.UIManager.GetUI and _G.UIID and _G.UIID.Task_TaskUI then
    local okUI, taskUI = pcall(_G.UIManager.GetUI, _G.UIID.Task_TaskUI)
    if okUI and taskUI and taskUI.TaskItemClick then
        pcall(function() taskUI:TaskItemClick(targetId) end)
        clickedUI = true
        log("[ACTION 1] Da goi Task_TaskUI:TaskItemClick(" .. tostring(targetId) .. ")")
    end
end

-- Cách 2: Gửi đúng gói tin mà hook btn_rewardOnClick gửi khi bảng TaskInfoUI bật lên
-- (CHỈ gửi Event.Task_BtnRewardClick / ReqCompleteTask 4007, KHÔNG gửi ReqSubmitTask 4005)
pcall(function()
    if _G.EventManager and _G.Event and _G.Event.Task_BtnRewardClick then
        _G.EventManager.Dispatch(_G.Event.Task_BtnRewardClick, targetId)
        log("[ACTION 2] Da Dispatch Event.Task_BtnRewardClick(" .. tostring(targetId) .. ")")
    elseif _G.NetManager and _G.TaskMessage and _G.TaskMessage.ReqCompleteTask then
        _G.NetManager.Send(_G.TaskMessage.ReqCompleteTask, { taskId = targetId })
        log("[ACTION 2] Da Send TaskMessage.ReqCompleteTask(" .. tostring(targetId) .. ")")
    end
end)

if _G. Utility and _G.Utility.ShowTip then
    pcall(_G.Utility.ShowTip, "Test nhận NV Cấp 80 (ID " .. tostring(targetId) .. ") - kiểu gây lag!")
end

-- Kiểm tra lại trạng thái sau 1.5 giây
if _G. LuaTimer and _G.LuaTimer.Add then
    _G.LuaTimer.Add(1500, function()
        local tAfter = _G.TaskData and _G.TaskData.GetTaskById and _G.TaskData.GetTaskById(targetId)
        if tAfter then
            local sAfter = tAfter.GetState and tAfter:GetState() or tAfter.state
            local cG = (tAfter.goals and tAfter.goals[1] and tAfter.goals[1].curCount) or "?"
            local mG = (tAfter.goals and tAfter.goals[1] and tAfter.goals[1].maxCount) or "?"
            log(string.format("[AFTER 1.5s] ID=%s | State=%s | Goal=%s/%s",
                tostring(targetId), tostring(sAfter), tostring(cG), tostring(mG)))
        else
            log("[AFTER 1.5s] Task " .. tostring(targetId) .. " da bi xoa khoi TaskData (Submitted).")
        end
        saveOutput()
    end)
else
    saveOutput()
end
