-- =========================================================================
-- [TEST SCRIPT]: THỬ NGHIỆM CÁC CƠ CHẾ PHỤC HỒI ĐỒNG BỘ VỊ TRÍ SERVER
-- =========================================================================

local function log_msg(msg)
    local text = tostring(msg)
    pcall(function()
        if _G.FloatingWordUtility and _G.FloatingWordUtility.QuickMsg then
            _G.FloatingWordUtility.QuickMsg(text)
        end
    end)
    pcall(function()
        if _G.WriteLog then _G.WriteLog("[SYNC_TEST] " .. text) end
    end)
end

local ok, err = pcall(function()
    local me = _G.RoleManager and _G.RoleManager.me
    if not me then
        log_msg("LỖI: Không tìm thấy RoleManager.me!")
        return
    end

    local cX = me.cellPos and me.cellPos.x or 0
    local cY = me.cellPos and me.cellPos.y or 0

    local sX = me.serverCoord and me.serverCoord.x or (me.data and me.data.serverCoord and me.data.serverCoord.x) or 0
    local sY = me.serverCoord and me.serverCoord.y or (me.data and me.data.serverCoord and me.data.serverCoord.y) or 0

    local dist = math.floor(math.sqrt((cX - sX)^2 + (cY - sY)^2) + 0.5)
    log_msg(string.format("TRƯỚC PHỤC HỒI: Client:(%d,%d) | Server:(%d,%d) => LỆCH: %d Ô", cX, cY, sX, sY, dist))

    if dist <= 2 then
        log_msg("VỊ TRÍ ĐÃ ĐỒNG BỘ TỐT (Không bị lệch)!")
        return
    end

    -- =========================================================================
    -- CƠ CHẾ PHỤC HỒI 1: TÌM ĐƯỜNG PATH TỪ SERVER XY -> CLIENT XY VÀ GỬI TỪNG BƯỚC
    -- Dẫn Server đi từ điểm kẹt đến đích theo đúng bản đồ
    -- =========================================================================
    local mapId = (_G.SceneData and _G.SceneData.mapId) or 0
    local line = (_G.SceneData and _G.SceneData.line) or 0

    if _G.Scene and _G.Scene.SearchTilePath then
        local startCell = CS.UnityEngine.Vector2Int(sX, sY)
        local endCell = CS.UnityEngine.Vector2Int(cX, cY)
        local canArrive, path = _G.Scene.SearchTilePath(startCell, endCell, 0, true)

        if canArrive and path and #path > 1 then
            log_msg(string.format("-> Tìm thấy đường hợp lệ từ Server tới Client (#path = %d)", #path))
            local sTime = (_G.Time and _G.Time.GetServerTime and _G.Time.GetServerTime()) or os.time()
            for i = 2, #path do
                local pt = path[i]
                if pt then
                    _G.NetManager.Send(_G.MapMessage.ReqMove, {
                        x = pt.x,
                        y = pt.y,
                        action = 2,
                        time = sTime
                    })
                end
            end
        else
            -- Nếu không tìm được path, gửi gói truyền tin Transmit
            if _G.NetManager and _G.MapMessage and _G.MapMessage.ReqTransmit and mapId > 0 then
                _G.NetManager.Send(_G.MapMessage.ReqTransmit, {
                    mapId = mapId,
                    line = line,
                    x = cX,
                    y = cY
                })
            end
        end
    end

    -- =========================================================================
    -- CƠ CHẾ PHỤC HỒI 2: KÉO AVATAR VỀ LẠI VỊ TRÍ SERVER NẾU SERVER BỊ KHÓA
    -- (Bỏ comment dòng dưới nếu muốn Avatar tức thì snap về Server để không mất máu)
    -- me:FlashTo(sX, sY)
    -- =========================================================================

    -- Kiểm tra lại sau 0.5s
    if _G.Timer and _G.Timer.Start then
        _G.Timer.Start(0.5, function()
            local curSX = me.serverCoord and me.serverCoord.x or 0
            local curSY = me.serverCoord and me.serverCoord.y or 0
            local newDist = math.floor(math.sqrt((cX - curSX)^2 + (cY - curSY)^2) + 0.5)
            log_msg(string.format("KẾT QUẢ: Server hiện tại: (%d,%d) | Lệch: %d Ô", curSX, curSY, newDist))
        end)
    end
end)

if not ok then
    log_msg("LỖI TEST: " .. tostring(err))
end
