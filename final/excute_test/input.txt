-- ============================================================================
-- SCRIPT TEST DỊCH CHUYỂN BẰNG CỔNG MAP CHÍNH (GetTransIdByGroupId)
-- ============================================================================
pcall(function()
    local targetBossId = 10179408 -- Phẫn Nộ
    local targetMapId = 101094    -- Hoang Dã C3
    local targetLine = 1
    local bossName = "Phẫn Nộ C3"

    local currentMapId = _G.SceneData and (_G.SceneData.mapId or _G.SceneData.groupId) or 0
    local currentLine = _G.SceneData and (_G.SceneData.line or _G.SceneData.cline) or 1

    local logs = {}
    local function logMsg(str)
        print("[TEST MAP TRANS] " .. str)
        table.insert(logs, str)
        if _G.FloatingWordUtility then _G.FloatingWordUtility.QuickMsg(str) end
    end

    logMsg(string.format("Test dịch chuyển cổng Map chính sang Map %d...", targetMapId))

    -- Lấy Transfer ID cổng chính của Map
    local mainMapTransId = _G.PathFinderManager and _G.PathFinderManager.GetTransIdByGroupId and _G.PathFinderManager.GetTransIdByGroupId(targetMapId)
    logMsg(string.format("Transfer ID Cổng chính Map %d là: %s", targetMapId, tostring(mainMapTransId)))

    if currentMapId ~= targetMapId or currentLine ~= targetLine then
        logMsg("Đang ở ngoài Map. Phát lệnh dịch chuyển cổng chính Map...")
        
        if mainMapTransId and _G.SceneController and _G.SceneController.OnReqTransferTransmitMap then
            _G.SceneController.OnReqTransferTransmitMap(nil, { mapId = mainMapTransId, line = targetLine, changeLine = true })
            logMsg("Đã gửi tin dịch chuyển cổng Map thành công!")
        end
    else
        logMsg("Đã ở trong Map! Đang tìm con Boss sống gần nhất...")
        local alivePos, aliveCount = nil, 0
        if _G.GetAliveBossPosition then
            alivePos, aliveCount = _G.GetAliveBossPosition(targetBossId, targetMapId)
        end
        local targetPos = alivePos or (_G.GetBossPosition and _G.GetBossPosition(targetBossId, targetMapId)) or {x = 100, y = 100}
        logMsg(string.format("Chạy bộ tới bãi Boss (%d, %d)...", targetPos.x, targetPos.y))
        
        if _G.RoleManager and _G.RoleManager.me and _G.RoleManager.me.MoveTo then
            _G.RoleManager.me:MoveTo({x = targetPos.x, y = targetPos.y}, 0)
        end
    end

    local outputPath = "/storage/emulated/0/Android/data/com.vnyh.gp/files/output.txt"
    if CS and CS.UnityEngine and CS.UnityEngine.Application and CS.UnityEngine.Application.persistentDataPath then
        outputPath = CS.UnityEngine.Application.persistentDataPath .. "/output.txt"
    end
    local f = io.open(outputPath, "w")
    if f then
        f:write(table.concat(logs, "\n"))
        f:close()
    end
end)
