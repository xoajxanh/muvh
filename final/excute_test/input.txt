local items = _G.BagInfoData and _G.BagInfoData.TotalItems
if not items then
    _G.WriteLog('[SmeltTest] BagInfoData.TotalItems is nil')
    return
end

local recycleItems = {}
local recycleNames = {}

for k, item in pairs(items) do
    if item and item.tblItem then
        local subType = item.tblItem.subType or 0
        local quality = item.tblItem.quality or 0
        
        -- Nhẫn (subType = 18) và Chuyển 6 (quality = 6)
        if subType == 18 and quality == 6 then
            table.insert(recycleItems, item.id)
            local name = item.tblItem.name or 'Unknown'
            table.insert(recycleNames, name)
        end
    end
end

if #recycleItems > 0 then
    _G.WriteLog('[SmeltTest] Đã tìm thấy ' .. tostring(#recycleItems) .. ' món Nhẫn Chuyển 6!')
    for i, name in ipairs(recycleNames) do
        _G.WriteLog(' - Chuẩn bị tách: ' .. tostring(name))
    end
    
    -- Gửi lệnh phân giải (Tách trang bị) lên server
    if _G.networkRequest and _G.networkRequest.ReqEquipDecompose then
        _G.networkRequest.ReqEquipDecompose(recycleItems)
        _G.WriteLog('[SmeltTest] Đã gửi lệnh ReqEquipDecompose tách thành công!')
    else
        _G.WriteLog('[SmeltTest] LỖI: Không tìm thấy hàm ReqEquipDecompose!')
    end
else
    _G.WriteLog('[SmeltTest] Trong túi không có món Nhẫn Chuyển 6 nào (subType=18, quality=6) để tách.')
end
