-- SCRIPT CHẨN ĐOÁN VỊ TRÍ & ĐỒNG BỘ SERVER / CLIENT KHI TỐC ĐỘ CAO
local report = {}
local function log(txt)
    local line = tostring(txt)
    table.insert(report, line)
    print(line)
end

log("==================================================")
log("=== KIỂM TRA ĐỒNG BỘ VỊ TRÍ CLIENT vs SERVER ===")
log("==================================================")

pcall(function()
    local me = _G.RoleManager and _G.RoleManager.me
    local meData = _G.ViewData and _G.ViewData.meData

    local clientCellX, clientCellY = -1, -1
    local clientPosX, clientPosY, clientPosZ = 0, 0, 0
    if me then
        if me.cellPos then
            clientCellX = me.cellPos.x
            clientCellY = me.cellPos.y
        end
        if me.pos then
            clientPosX = me.pos.x
            clientPosY = me.pos.y
            clientPosZ = me.pos.z
        end
    end

    local serverCellX, serverCellY = -1, -1
    if meData and meData.serverCoord then
        serverCellX = meData.serverCoord.x
        serverCellY = meData.serverCoord.y
    elseif me and me.serverCoord then
        serverCellX = me.serverCoord.x
        serverCellY = me.serverCoord.y
    end

    local dx = clientCellX - serverCellX
    local dy = clientCellY - serverCellY
    local dist = math.sqrt(dx * dx + dy * dy)

    log(string.format("1. Vị trí Client (cellPos)        : (%d, %d)", clientCellX, clientCellY))
    log(string.format("2. Vị trí Server ghi nhận (Coord) : (%d, %d)", serverCellX, serverCellY))
    log(string.format("3. Độ lệch khoảng cách (Desync)   : %.2f ô (tiles)", dist))
    log(string.format("4. Vị trí thực tế 3D (World Pos)  : (%.2f, %.2f, %.2f)", clientPosX, clientPosY, clientPosZ))
    log("--------------------------------------------------")

    if dist > 5 then
        log(string.format(">>> CẢNH BÁO: BỊ LỆCH TỌA ĐỘ NGHIÊM TRỌNG (%.1f ô)! <<<", dist))
        log("-> Giải thích: Client đã chạy tới đích nhưng Server vẫn giữ bạn ở vị trí cũ.")
        log("-> Do đó Server KHÔNG gửi dữ liệu Boss/Người chơi ở điểm đến!")
    else
        log(">>> TRẠNG THÁI: Tọa độ Client và Server ĐỒNG BỘ TỐT (Khớp nhau) <<<")
    end
    log("--------------------------------------------------")

    -- Thống kê số lượng Quái / Boss / Người chơi hiện có trong bộ nhớ RoleManager
    local monsterCount = 0
    local playerCount = 0
    local bossList = {}

    if _G.RoleManager and _G.RoleManager.GetRoleListByType then
        local monsters = _G.RoleManager.GetRoleListByType(_G.ERoleType.Monster)
        if monsters then
            for id, m in pairs(monsters) do
                monsterCount = monsterCount + 1
                local name = (m.data and m.data.name) or "Quái"
                local mx = (m.cellPos and m.cellPos.x) or 0
                local my = (m.cellPos and m.cellPos.y) or 0
                if m.isBoss or (m.data and m.data.isBoss) or string.find(name, "Boss") or string.find(name, "Kundun") then
                    table.insert(bossList, string.format("%s [ID:%s] tại (%d, %d)", name, tostring(id), mx, my))
                end
            end
        end

        local players = _G.RoleManager.GetRoleListByType(_G.ERoleType.Player)
        if players then
            for id, p in pairs(players) do
                if not p.isMe then
                    playerCount = playerCount + 1
                end
            end
        end
    end

    log(string.format("5. Tổng số Quái/Boss Client đang có : %d", monsterCount))
    log(string.format("6. Tổng số Người chơi khác trong bộ nhớ: %d", playerCount))
    if #bossList > 0 then
        log("Danh sách Boss phát hiện:")
        for _, b in ipairs(bossList) do
            log("   + " .. b)
        end
    else
        log("Danh sách Boss: Không có con nào quanh vùng Server ghi nhận.")
    end

    -- Thử nghiệm lệnh sửa lệch tọa độ tức thì
    log("--------------------------------------------------")
    if dist > 5 then
        log("[ĐỀ XUẤT TEST SỬA]:")
        log("Đang thử gửi gói tin sửa tọa độ hoặc phục hồi...")
    end
end)

log("==================================================")
log("KẾT THÚC THỰC THI SCRIPT")

local finalStr = table.concat(report, "\n")
pcall(function()
    local path = "/storage/emulated/0/Android/data/com.vnyh.gp/files/output.txt"
    local f = io.open(path, "w")
    if f then
        f:write(finalStr)
        f:close()
    end
end)

if _G.FloatingWordUtility then
    _G.FloatingWordUtility.QuickMsg("Đã xuất kết quả kiểm tra tọa độ!")
end

return finalStr
