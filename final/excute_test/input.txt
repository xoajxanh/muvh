local outputPath = nil
local function GetOutputPath()
    if outputPath then return outputPath end
    if CS.UnityEngine.Application and CS.UnityEngine.Application.persistentDataPath then
        outputPath = CS.UnityEngine.Application.persistentDataPath .. "/output.txt"
        return outputPath
    end
    outputPath = "/storage/emulated/0/Android/data/com.vnyh.gp/files/output.txt"
    return outputPath
end

local function SafeLog(msg)
    pcall(function()
        local path = GetOutputPath()
        local f = io.open(path, "a")
        if f then
            f:write(tostring(msg) .. "\n")
            f:close()
        end
    end)
end

pcall(function()

    local f = io.open(path, "w")
    if f then
        f:write("=========================================================================\n")
        f:write(">>> THỬ NGHIỆM CƠ CHẾ NẠP PHÙ VĂN HANDSHAKE (CHỐNG RƠI RỚT 100%) <<<\n")
        f:write(string.format("Thời gian: %s\n", os.date and os.date("%Y/%m/%d %H:%M:%S") or "now"))
        f:write("=========================================================================\n")
        f:close()
    end
end)

local function ReqImplant(h, uid)
    pcall(function()
        if _G.networkRequest and _G.networkRequest.ReqImplantRune then
            _G.networkRequest.ReqImplantRune(h, uid)
        elseif networkRequest and networkRequest.ReqImplantRune then

        elseif _G.NetManager and _G.EquipMessage and _G.EquipMessage.ReqImplantRune then
            _G.NetManager.Send(_G.EquipMessage.ReqImplantRune, { index = h, itemId = uid })
        end
    end)
end

local runeNames = {
    [190100010] = "Phù Văn Đỏ Lv10",
    [190100009] = "Phù Văn Đỏ Lv9",
    [190200010] = "Phù Văn Lam Lv10",
    [190300010] = "Phù Văn Lục Lv10",
    [190300009] = "Phù Văn Lục Lv9",
}

local timerObj = _G.Timer or Timer

-- Helper: Lấy danh sách ngọc trong túi theo thời gian thực
local function GetAvailableBagRunes()
    local list = {}
    local bagRunes = (runeMgr.GetBagAllRunes and runeMgr:GetBagAllRunes()) or {}
    for _, rData in ipairs(bagRunes) do
        local uid = rData.id or (rData.serverData and rData.serverData.id) or 0
        local itemId = rData.itemId or (rData.serverData and rData.serverData.itemId) or (rData.tblItem and rData.tblItem.id) or 0
        if uid ~= 0 and itemId ~= 0 then
            table.insert(list, { uid = uid, itemId = itemId })
        end
    end
    if #list == 0 and _G.BagInfoData and _G.BagInfoData.TotalItems then
        for _, v in pairs(_G.BagInfoData.TotalItems) do
            if v and v.itemId and (v.itemId >= 190100000 and v.itemId <= 190400000) then
                table.insert(list, { uid = v.id, itemId = v.itemId })
            end
        end
    end
    return list
end

-- =========================================================================
-- [CƠ CHẾ NẠP PRESET BẬC THẦY: STATE-DRIVEN HANDSHAKE]
-- 1. Phân tích ô cần tháo & ô cần khảm.
-- 2. Tháo tuần tự từng ô sai, chờ Server xác nhận ô đã trống.
-- 3. Khảm từng ô: Quét túi đồ thời gian thực -> Gửi lệnh khảm ->
--    Chờ Server xác nhận ô đó ĐÃ CÓ ĐÚNG NGỌC -> Mới khảm ô tiếp theo!
-- =========================================================================
local function HandshakeApplyRunePreset(targetConfig, onFinished)
    local allData = runeMgr:GetAllServerRuneData()
    local needUninlay = {}
    local needInlay = {}

    Log("\n--- [BƯỚC 1: PHÂN TÍCH SO SÁNH 7 Ô VỚI PRESET MỤC TIÊU] ---")
    for h = 1, 7 do
        local slot = allData[h]
        local rItem = slot and slot.runeItem
        local curItemId = (rItem and rItem.itemId and rItem.itemId ~= 0) and rItem.itemId or 0
        local expItemId = targetConfig[h] or 0

        if curItemId == expItemId and expItemId ~= 0 then
            Log(string.format("  Ô [%d]: Đã đúng ngọc (%s) -> GIỮ NGUYÊN", h, runeNames[curItemId] or tostring(curItemId)))
        else
            if curItemId ~= 0 then
                table.insert(needUninlay, h)
                Log(string.format("  Ô [%d]: Đang có %s (Cần %s) -> Cần tháo", h, runeNames[curItemId] or tostring(curItemId), runeNames[expItemId] or tostring(expItemId)))
            else
                Log(string.format("  Ô [%d]: Đang TRỐNG (Cần %s) -> Cần khảm", h, runeNames[expItemId] or tostring(expItemId)))
            end
            if expItemId ~= 0 then
                table.insert(needInlay, { hole = h, itemId = expItemId })
            end
        end
    end

    if #needUninlay == 0 and #needInlay == 0 then
        Log("\n-> Tất cả 7 ô Phù Văn đã hoàn toàn chính xác! Không cần thao tác gì thêm.")
        RefreshRuneUI()
        if onFinished then onFinished() end
        return
    end

    -- Hàm tháo tuần tự có Handshake
    local function ProcessUninlay(index, doneCallback)
        if index > #needUninlay then
            -- Chờ thêm 0.3s để túi đồ cập nhật đầy đủ các viên vừa tháo
            timerObj.Start(0.3, function()
                if doneCallback then doneCallback() end
            end)
            return
        end

        local h = needUninlay[index]
        Log(string.format("  -> [Tháo %d/%d]: Gửi lệnh tháo Ô [%d]...", index, #needUninlay, h))
        ReqImplant(h, 0)

        -- Polling chờ server xác nhận ô h đã trống (tối đa 1s)
        local elapsed = 0
        local function CheckEmpty()
            timerObj.Start(0.1, function()
                elapsed = elapsed + 0.1
                local curSlot = runeMgr:GetAllServerRuneData()[h]
                local curR = curSlot and curSlot.runeItem
                local isClean = (curR == nil or curR.itemId == nil or curR.itemId == 0)

                if isClean or elapsed >= 0.8 then
                    ProcessUninlay(index + 1, doneCallback)
                else
                    CheckEmpty()
                end
            end)
        end
        CheckEmpty()
    end

    -- Hàm khảm tuần tự có Handshake (Từng ô một, kiểm tra Server confirm mới đi tiếp)
    local function ProcessInlay(index, doneCallback)
        if index > #needInlay then
            if doneCallback then doneCallback() end
            return
        end

        local req = needInlay[index]
        local currentBag = GetAvailableBagRunes()
        local chosenRune = nil

        -- Tìm viên ngọc có itemId khớp trong túi đồ
        for _, b in ipairs(currentBag) do
            if b.itemId == req.itemId then
                chosenRune = b
                break
            end
        end

        if not chosenRune then
            Log(string.format("  [!] Ô [%d]: Cần %s nhưng trong túi đồ hiện tại KHÔNG CÓ! (Bỏ qua ô này)", req.hole, runeNames[req.itemId] or tostring(req.itemId)))
            ProcessInlay(index + 1, doneCallback)
            return
        end

        Log(string.format("  -> [Khảm %d/%d]: Ô [%d] <- %s (UID: %s)", index, #needInlay, req.hole, runeNames[req.itemId] or "Rune", tostring(chosenRune.uid)))
        ReqImplant(req.hole, chosenRune.uid)

        -- Polling kiểm tra Server xác nhận ô req.hole đã nhận ngọc (tối đa 1.2s)
        local elapsed = 0
        local function CheckInlaid()
            timerObj.Start(0.12, function()
                elapsed = elapsed + 0.12
                local curSlot = runeMgr:GetAllServerRuneData()[req.hole]
                local curR = curSlot and curSlot.runeItem
                local curId = (curR and curR.itemId and curR.itemId ~= 0) and curR.itemId or 0

                if curId == req.itemId then
                    -- Server đã xác nhận khảm thành công! Dừng lại 0.1s rồi sang ô kế
                    timerObj.Start(0.1, function()
                        ProcessInlay(index + 1, doneCallback)
                    end)
                elseif elapsed >= 1.2 then
                    Log(string.format("  [!] Hết thời gian chờ phản hồi Ô [%d], chuyển tiếp...", req.hole))
                    ProcessInlay(index + 1, doneCallback)
                else
                    CheckInlaid()
                end
            end)
        end
        CheckInlaid()
    end

    -- Bắt đầu quy trình
    if #needUninlay > 0 then
        Log(string.format("\n--- [BƯỚC 2: TIẾN HÀNH THÁO %d Ô SAI (HANDSHAKE)] ---", #needUninlay))
        ProcessUninlay(1, function()
            Log(string.format("\n--- [BƯỚC 3: TIẾN HÀNH KHẢM %d Ô (HANDSHAKE)] ---", #needInlay))
            ProcessInlay(1, function()
                RefreshRuneUI()
                if onFinished then onFinished() end
            end)
        end)
    else
        Log("\n--- [BƯỚC 2: KHÔNG CÓ Ô NÀO CẦN THÁO] ---")
        Log(string.format("\n--- [BƯỚC 3: TIẾN HÀNH KHẢM %d Ô (HANDSHAKE)] ---", #needInlay))
        ProcessInlay(1, function()
            RefreshRuneUI()
            if onFinished then onFinished() end
        end)
    end
end

local targetConfig = {
    [1] = 190100010, -- Đỏ Lv10
    [2] = 190300010, -- Lục Lv10
    [3] = 190300009, -- Lục Lv9
    [4] = 190300010, -- Lục Lv10
    [5] = 190200010, -- Lam Lv10
    [6] = 190200010, -- Lam Lv10
    [7] = 190100009, -- Đỏ Lv9
}

QuickMsg("BẮT ĐẦU: Chạy cơ chế nạp Handshake (xác nhận từng ô)...")
Log("\n>>> BẮT ĐẦU QUY TRÌNH NẠP PRESET HANDSHAKE <<<")

HandshakeApplyRunePreset(targetConfig, function()
    timerObj.Start(0.3, function()
        RefreshRuneUI()
        local finalData = runeMgr:GetAllServerRuneData()
        local matchCount = 0

        Log("\n=========================================================================")
        Log(">>> KẾT QUẢ KIỂM TRA SAU KHI NẠP BẰNG CƠ CHẾ HANDSHAKE <<<")
        Log("=========================================================================")

        for h = 1, 7 do
            local s = finalData[h]
            local r = s and s.runeItem
            local curId = (r and r.itemId and r.itemId ~= 0) and r.itemId or 0
            local expId = targetConfig[h] or 0
            local isMatch = (curId == expId and expId ~= 0)
            if isMatch then matchCount = matchCount + 1 end
            Log(string.format("  Ô [%d]: %s (ItemId: %d) | Mục tiêu: %s -> %s",
                h, runeNames[curId] or "TRỐNG", curId, runeNames[expId] or "TRỐNG",
                isMatch and "CHÍNH XÁC 100%" or "CHƯA KHỚP"))
        end

        Log("-------------------------------------------------------------------------")
        Log(string.format("TỔNG KẾT: Khớp chuẩn xác %d / 7 ô Phù Văn!", matchCount))
        if matchCount == 7 then
            QuickMsg("THÀNH CÔNG RỰC RỠ: KHỚP ĐỦ 7/7 Ô PHÙ VĂN!")
            Log(">>> ĐÁNH GIÁ: CƠ CHẾ HANDSHAKE ĐẠT ĐỘ CHÍNH XÁC TUYỆT ĐỐI 7/7 Ô (100%)! <<<")
        else
            QuickMsg(string.format("Kết quả: Khớp %d/7 ô Phù Văn!", matchCount))
            Log(string.format(">>> CẢNH BÁO: Còn %d ô chưa khớp. <<<", 7 - matchCount))
        end
        Log("=========================================================================\n")
    end)
end)
