-- =========================================================================
-- [TEST SCRIPT]: TEST ĐI XUYÊN QUÁI & NGƯỜI CHƠI (ROLE COLLISION BYPASS)
-- Mô tả: Hook Scene.TileCanMoveToByRole, Scene.SetRoleTilesInfor, 
--        SceneData.orThrough và SceneData.overlap để đi xuyên quái / người chơi.
-- Cách dùng: Bấm EXECUTE SCRIPT một lần để BẬT, bấm lần nữa để TẮT (Toggle).
-- =========================================================================

local outputLines = {}

local function log_msg(msg)
    local text = tostring(msg)
    table.insert(outputLines, text)
    print(text)
    pcall(function()
        if _G.FloatingWordUtility and _G.FloatingWordUtility.QuickMsg then
            _G.FloatingWordUtility.QuickMsg(text)
        end
    end)
    pcall(function()
        if _G.WriteLog then
            _G.WriteLog("[BYPASS_COLLISION] " .. text)
        end
    end)
end

local ok, err = pcall(function()
    log_msg("=========================================================================")
    log_msg("=== TEST ĐI XUYÊN QUÁI & NGƯỜI CHƠI (ROLE COLLISION BYPASS) ===")
    log_msg("=========================================================================")

    local me = _G.RoleManager and _G.RoleManager.me
    if not me then
        log_msg("CẢNH BÁO: Chưa tìm thấy RoleManager.me (chưa vào map hoặc chưa login)!")
    else
        local meX = me.serverCoord and me.serverCoord.x or (me.cellPos and me.cellPos.x) or (me.data and me.data.x) or 0
        local meY = me.serverCoord and me.serverCoord.y or (me.cellPos and me.cellPos.y) or (me.data and me.data.y) or 0
        log_msg(string.format("Tọa độ Me hiện tại: (%d, %d) | MapId: %s", meX, meY, tostring(_G.SceneData and _G.SceneData.mapId)))
    end

    -- Khởi tạo biến lưu trạng thái gốc nếu chưa có
    if _G.Mod_Orig_TileCanMoveToByRole == nil and _G.Scene and _G.Scene.TileCanMoveToByRole then
        _G.Mod_Orig_TileCanMoveToByRole = _G.Scene.TileCanMoveToByRole
    end
    if _G.Mod_Orig_SetRoleTilesInfor == nil and _G.Scene and _G.Scene.SetRoleTilesInfor then
        _G.Mod_Orig_SetRoleTilesInfor = _G.Scene.SetRoleTilesInfor
    end

    -- Đảo trạng thái (Toggle)
    _G.Mod_CollisionBypass_Enabled = not _G.Mod_CollisionBypass_Enabled

    if _G.Mod_CollisionBypass_Enabled then
        ---------------------------------------------------------------------
        -- 1. KÍCH HOẠT BẬT BYPASS
        ---------------------------------------------------------------------
        log_msg(">>> TRẠNG THÁI: [ BẬT ] ĐI XUYÊN QUÁI & NGƯỜI CHƠI")

        -- Gán cờ SceneData cho map hiện tại
        if _G.SceneData then
            log_msg(string.format("  -> SceneData.orThrough cũ: %s | SceneData.overlap cũ: %s", tostring(_G.SceneData.orThrough), tostring(_G.SceneData.overlap)))
            _G.SceneData.orThrough = 1
            _G.SceneData.overlap = 1
            log_msg("  -> Đã set SceneData.orThrough = 1, SceneData.overlap = 1")
        end

        -- Hook Scene.TileCanMoveToByRole luôn trả về true
        if _G.Scene then
            _G.Scene.TileCanMoveToByRole = function(cell)
                return true
            end
            log_msg("  -> Đã Hook Scene.TileCanMoveToByRole = function() return true end")

            -- Hook Scene.SetRoleTilesInfor để vô hiệu hóa việc ghi đè tile cản vào thuật toán tìm đường (A*)
            _G.Scene.SetRoleTilesInfor = function(val)
                -- Không ghi đè quái/người thành vật cản
                return
            end
            log_msg("  -> Đã Hook Scene.SetRoleTilesInfor (vô hiệu cản đường A* PathFinder)")
        end

        -- Duy trì cấu hình tự động mỗi 1s hoặc khi chuyển map
        if _G.Timer and _G.Timer.StartLoop then
            if _G.Mod_CollisionBypass_TimerId then
                pcall(function() _G.Timer.Cancel(_G.Mod_CollisionBypass_TimerId) end)
            end
            _G.Mod_CollisionBypass_TimerId = _G.Timer.StartLoop(1, function()
                if not _G.Mod_CollisionBypass_Enabled then return end
                if _G.SceneData then
                    if _G.SceneData.orThrough ~= 1 then _G.SceneData.orThrough = 1 end
                    if _G.SceneData.overlap ~= 1 then _G.SceneData.overlap = 1 end
                end
                if _G.Scene and _G.Scene.TileCanMoveToByRole ~= nil then
                    -- Đảm bảo hook không bị load map đè lại
                    if _G.Scene.TileCanMoveToByRole ~= _G.Mod_Bypass_DummyFunc then
                        _G.Scene.TileCanMoveToByRole = _G.Mod_Bypass_DummyFunc
                    end
                end
            end)
        end
        _G.Mod_Bypass_DummyFunc = function(cell) return true end

        log_msg("KẾT QUẢ: ĐÃ KÍCH HOẠT THÀNH CÔNG! Hãy thử chạy qua quái hoặc người chơi.")
    else
        ---------------------------------------------------------------------
        -- 2. TẮT BYPASS & KHÔI PHỤC GỐC
        ---------------------------------------------------------------------
        log_msg(">>> TRẠNG THÁI: [ TẮT ] KHÔI PHỤC VẬT CẢN GỐC CỦA GAME")

        -- Hủy Timer duy trì
        if _G.Mod_CollisionBypass_TimerId and _G.Timer and _G.Timer.Cancel then
            pcall(function() _G.Timer.Cancel(_G.Mod_CollisionBypass_TimerId) end)
            _G.Mod_CollisionBypass_TimerId = nil
        end

        -- Khôi phục hook Scene
        if _G.Scene and _G.Mod_Orig_TileCanMoveToByRole then
            _G.Scene.TileCanMoveToByRole = _G.Mod_Orig_TileCanMoveToByRole
            log_msg("  -> Đã khôi phục Scene.TileCanMoveToByRole gốc")
        end
        if _G.Scene and _G.Mod_Orig_SetRoleTilesInfor then
            _G.Scene.SetRoleTilesInfor = _G.Mod_Orig_SetRoleTilesInfor
            log_msg("  -> Đã khôi phục Scene.SetRoleTilesInfor gốc")
        end

        -- Khôi phục SceneData theo config map
        if _G.SceneData and _G.SceneData.mapId and _G.Table_Map_map then
            local mapCfg = _G.Table_Map_map[_G.SceneData.mapId]
            if mapCfg then
                _G.SceneData.orThrough = tonumber(mapCfg.orThrough)
                _G.SceneData.overlap = tonumber(mapCfg.overlap)
                log_msg(string.format("  -> Đã khôi phục SceneData theo mapId %d: orThrough=%s, overlap=%s", 
                    _G.SceneData.mapId, tostring(_G.SceneData.orThrough), tostring(_G.SceneData.overlap)))
            end
        end

        log_msg("KẾT QUẢ: ĐÃ TẮT BẢN TEST XUYÊN QUÁI! Đã về trạng thái bình thường.")
    end

    log_msg("=========================================================================")
end)

if not ok then
    log_msg("LỖI RUNTIME TEST: " .. tostring(err))
end

-- Ghi đè vào output.txt trên Android
pcall(function()
    local fullContent = table.concat(outputLines, "\n")
    local outPath = "/storage/emulated/0/Android/data/com.vnyh.gp/files/output.txt"
    local f = io.open(outPath, "w")
    if f then
        f:write(fullContent)
        f:close()
    end
end)

return table.concat(outputLines, "\n")
