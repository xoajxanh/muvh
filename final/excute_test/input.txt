local function LogOut(text)
    if _G.WriteLog then 
        _G.WriteLog(text)
    else
        print(text)
    end
end

LogOut("\n========== BẮT ĐẦU TEST ADMIN BURST PICKUP ==========")
LogOut("Vui lòng tự mở rương vàng. Script đã override AddDropSceneCellPos...")

-- Khôi phục hook cũ nếu có để tránh đè nhiều lớp khi chạy excute nhiều lần
if _G.Original_AddDropSceneCellPos_Test then
    _G.PickupManager.AddDropSceneCellPos = _G.Original_AddDropSceneCellPos_Test
end

if _G.PickupManager and _G.PickupManager.AddDropSceneCellPos then
    _G.Original_AddDropSceneCellPos_Test = _G.PickupManager.AddDropSceneCellPos

    _G.PickupManager.AddDropSceneCellPos = function(item)
        -- Gọi lại hàm gốc để game vẫn hiển thị item bình thường
        if _G.Original_AddDropSceneCellPos_Test then
            _G.Original_AddDropSceneCellPos_Test(item)
        end

        if not (item and item.data) then return end
        local dropItemData = item.data

        local itemId = dropItemData.id or dropItemData.objId or (dropItemData.item and dropItemData.item.id)
        local itemX = dropItemData.x or (dropItemData.pos and dropItemData.pos.x)
        local itemY = dropItemData.y or (dropItemData.pos and dropItemData.pos.y)

        if not itemId or not itemX or not itemY then
            return
        end

        LogOut(string.format("[TEST] Phát hiện item rớt: ID=%s, X=%s, Y=%s", tostring(itemId), tostring(itemX), tostring(itemY)))

        -- =========================================================================
        -- [MOD FEATURE]: BÃO NHẶT SIÊU TỐC ADMIN (ADMIN SUPER BURST LOOT)
        -- =========================================================================
        if _G.PickupManager and _G.PickupManager.ReqPickUpMapItem then
            _G.PickupManager.ReqPickUpMapItem(itemId)
            LogOut("[TEST] Gửi ReqPickUpMapItem lần 1")
        end

        -- 1. Gửi lệnh MoveTo 1 lần duy nhất để bắt đầu di chuyển tới vật phẩm
        if _G.RoleManager and _G.RoleManager.me and itemX and itemY then
            pcall(function()
                _G.RoleManager.me:MoveTo({ x = itemX, y = itemY }, 0)
                LogOut(string.format("[TEST] Đã gọi MoveTo({ x = %s, y = %s }, 0)", tostring(itemX), tostring(itemY)))
            end)
        end

        -- 2. Trong khi di chuyển, liên tục spam gói nhặt (40 lần x 0.05s = 2s)
        if _G.Timer and _G.Timer.StartLoop then
            _G.Timer.StartLoop(0.1, 10, function()
                if _G.PickupManager and _G.PickupManager.ReqPickUpMapItem then
                    _G.PickupManager.ReqPickUpMapItem(itemId)
                end
            end)
            LogOut("[TEST] Bắt đầu spam gói nhặt 40 lần")
        end
    end

    LogOut("Đã hook thành công vào _G.PickupManager.AddDropSceneCellPos!")
else
    LogOut("LỖI: Không tìm thấy _G.PickupManager.AddDropSceneCellPos để hook!")
end
