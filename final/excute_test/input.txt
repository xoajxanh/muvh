-- ====================================================
-- SCRIPT TEST DEV: RESET ALL SAVED CONFIGS & RE-FETCH FROM SERVER
-- ====================================================

-- 1. Reset toàn bộ PlayerPrefs và các mảng bộ nhớ (Mod_AutoFarmBoss_Config, Mod_SmeltConfig...)
if _G.Mod_ResetSavedModConfigs then
    _G.Mod_ResetSavedModConfigs()
else
    pcall(function()
        CS.UnityEngine.PlayerPrefs.DeleteAll()
        CS.UnityEngine.PlayerPrefs.Save()
    end)
    _G.Mod_AutoFarmBoss_Config = {}
    _G.Mod_SmeltConfig = {}
    _G.Mod_AutoFarmBoss_Target = nil
end

-- 2. Đánh dấu reset trạng thái kéo config
_G.Mod_HasFetchedConfig = false
_G.Mod_ActiveConfig = nil

-- 3. Gọi lại API Server để kéo cấu hình mới tinh
if _G.Mod_FetchRemoteConfig then
    _G.Mod_FetchRemoteConfig(function(success, config)
        if success then
            print("[DEV TEST] Đã kéo và áp dụng Cấu hình mới từ Server thành công!")
            
            -- Làm mới lại tất cả giao diện
            pcall(function()
                if _G.UpdateBossWatchUIText then _G.UpdateBossWatchUIText() end
                if _G.ModUpdateKundunUI then _G.ModUpdateKundunUI() end
                if _G.UpdateTierTabs then _G.UpdateTierTabs() end
            end)
        else
            print("[DEV TEST] Lỗi khi kéo Cấu hình từ Server!")
        end
    end)
else
    print("[DEV TEST] Không tìm thấy hàm Mod_FetchRemoteConfig!")
end
