pcall(function()
    print("==================== ADVANCED TEST HOOK [NOTICE ANNOUNCE & CHAT] ====================")
    
    local function ProcessSystemAnnouncement(rawText, sourceTag)
        pcall(function()
            if not rawText or rawText == "" then return end
            
            -- 1. Thay thế đặc thù của game: <color=#ffffff00>a</color> -> space
            local textWithoutColorA = string.gsub(rawText, "<color=#ffffff00>a</color>", " ")
            
            -- 2. Lọc sạch toàn bộ thẻ HTML còn lại
            local cleanText = string.gsub(textWithoutColorA, "<[^>]+>", "")
            
            -- 3. Chuẩn hóa khoảng trắng kép/nhiều khoảng trắng thành 1 khoảng trắng
            cleanText = string.gsub(cleanText, "%s+", " ")
            cleanText = string.trim and string.trim(cleanText) or cleanText
            
            print(string.format("\n[%s RECEIVE]", tostring(sourceTag)))
            print(string.format("  -> RAW:   %s", rawText))
            print(string.format("  -> CLEAN: %s", cleanText))
            
            -- Hiển thị QuickMsg tin nhắn vừa nhận
            if _G.FloatingWordUtility then
                _G.FloatingWordUtility.QuickMsg(string.format("[%s] %s", sourceTag, cleanText))
            end
            
            -- Lọc kiểm tra Boss Biến Dị
            if string.find(cleanText, "Biến Dị") or string.find(cleanText, "BOSS") or string.find(cleanText, "triệu hồi") then
                print("=== [PHÁT HIỆN BOSS BIẾN DỊ IN ANNOUNCEMENT] ===")
                
                -- Bóc tách Tên Nhân Vật
                local pName = string.match(cleanText, "%]%s*([^%s]+)%s+trong")
                if not pName or pName == "" then
                    pName = string.match(cleanText, "([^%s]+)%s+trong%s+Bản")
                end
                
                -- Bóc tách Thông Tin Boss
                local bossInfo = string.match(cleanText, "triệu%s+hồi%s+thành%s+công%s+(.+)") or cleanText
                
                -- Kiểm tra Kỵ Sĩ (và không phải Đất Sét)
                local isKySi = (string.find(cleanText, "Kỵ Sĩ") ~= nil) and (string.find(cleanText, "Đất Sét") == nil)
                
                local resultStr = string.format("PARSED -> Player: '%s' | Boss: '%s' | IsKySi: %s",
                    tostring(pName or "Unknown"), tostring(bossInfo), tostring(isKySi))
                print("  " .. resultStr)
                
                if _G.FloatingWordUtility then
                    _G.FloatingWordUtility.QuickMsg(string.format("TỔNG HỢP ẨN: Tên=%s | Kỵ Sĩ=%s", tostring(pName), tostring(isKySi)))
                end
            end
        end)
    end
    
    -- HOOK 1: NoticeController.ResAnnounce (Engine thông báo hệ thống chính)
    if _G.NoticeController and not _G.Mod_TestNoticeHooked then
        _G.Mod_TestNoticeHooked = true
        local orig_ResAnnounce = _G.NoticeController.ResAnnounce
        _G.NoticeController.ResAnnounce = function(eventId, data)
            pcall(function()
                if data and data.parameter then
                    local paramStr = table.concat(data.parameter, " | ")
                    ProcessSystemAnnouncement("NoticeID:" .. tostring(data.id) .. " Param:" .. paramStr, "NOTICE_RAW")
                end
            end)
            if orig_ResAnnounce then orig_ResAnnounce(eventId, data) end
        end
        print("-> Đã Hook thành công NoticeController.ResAnnounce!")
    end
    
    -- HOOK 2: ChatData.AddMessage (Hộp thoại Chat)
    if _G.ChatData and not _G.Mod_TestChatHooked then
        _G.Mod_TestChatHooked = true
        local orig_AddMessage = _G.ChatData.AddMessage
        _G.ChatData.AddMessage = function(channel, message)
            pcall(function()
                local rawText = ""
                if type(message) == "table" then
                    if type(message.chatMsg) == "table" and message.chatMsg.message then
                        rawText = tostring(message.chatMsg.message)
                    elseif type(message.chatMsg) == "string" then
                        rawText = tostring(message.chatMsg)
                    elseif message.message then
                        rawText = tostring(message.message)
                    elseif message.content then
                        rawText = tostring(message.content)
                    end
                else
                    rawText = tostring(message or "")
                end
                
                ProcessSystemAnnouncement(rawText, "CHAT_MSG")
            end)
            if orig_AddMessage then orig_AddMessage(channel, message) end
        end
        print("-> Đã Hook thành công ChatData.AddMessage!")
    end
    
    print("==================== HOÀN TẤT SETUP HOOK ANNOUNCEMENT ====================")
end)
