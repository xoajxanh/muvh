-- ========================================================
-- SCRIPT TEST HỒI SINH MIỄN PHÍ SIÊU TỐC (0S DELAY + BỎ POPUP 3S)
-- Hồi sinh về điểm xuất phát / thành lập tức khi chết (RoleReliveType.BornPoint / KSBattle)
-- Tự động dập tắt Popup 3s khi nhân vật đã sống lại
-- Nạp qua input.txt -> Chạy qua nút EXEC
-- ========================================================

local function Log(msg)
    print("[TEST FREE REVIVE] " .. tostring(msg))
    pcall(function()
        if _G.Mod_DebugMsg then
            _G.Mod_DebugMsg(msg)
        elseif _G.FloatingWordUtility and _G.FloatingWordUtility.QuickMsg then
            _G.FloatingWordUtility.QuickMsg(msg)
        end
    end)
end

Log("ĐÃ NẠP SCRIPT TEST HỒI SINH MIỄN PHÍ SIÊU TỐC!")

-- HÀM THỰC THI HỒI SINH MIỄN PHÍ
local function DoInstantFreeRevive()
    pcall(function()
        local me = _G.RoleManager and _G.RoleManager.me
        if not me then return end
        
        local mapId = 0
        if _G.SceneData and _G.SceneData.mapId then
            mapId = _G.SceneData.mapId
        elseif me.mapId then
            mapId = me.mapId
        end

        local isKS = (mapId == 1077) or
            (_G.TranScriptData and _G.TranScriptData.IsInRefineKSBattle and _G.TranScriptData.IsInRefineKSBattle())

        if _G.MeController and _G.RoleReliveType then
            if isKS then
                Log("Gửi lệnh Hồi Sinh KS Battle lập tức...")
                _G.MeController.ReqReqRelive(_G.RoleReliveType.KSBattle)
            else
                Log("Gửi lệnh Hồi Sinh Miễn Phí (Điểm xuất phát) lập tức...")
                _G.MeController.ReqReqRelive(_G.RoleReliveType.BornPoint)
            end
        end

        -- Tắt Popup Hồi sinh lập tức nếu đang mở
        if _G.UIManager and _G.UIManager.Hide and _G.UIID and _G.UIID.Role_ResurgenceUI then
            _G.UIManager.Hide(_G.UIID.Role_ResurgenceUI)
        end
    end)
end

-- VÒNG LẶP THEO DÕI HỒI SINH MIỄN PHÍ & DẬP POPUP 3S
if _G.Test_FreeReviveLoopTimer then
    pcall(function() _G.Timer.Stop(_G.Test_FreeReviveLoopTimer) end)
    _G.Test_FreeReviveLoopTimer = nil
end

if _G.Timer and _G.Timer.StartLoop then
    _G.Test_FreeReviveLoopTimer = _G.Timer.StartLoop(0.05, -1, function()
        pcall(function()
            local me = _G.RoleManager and _G.RoleManager.me
            if not me then return end
            
            -- 1. Nếu nhân vật đang CHẾT -> Thực thi Hồi sinh miễn phí siêu tốc
            if me.isDead or (me.hp and me.hp <= 0) then
                if not _G.Test_LastFreeReviveTime or (CS.UnityEngine.Time.realtimeSinceStartup - _G.Test_LastFreeReviveTime) >= 1.0 then
                    _G.Test_LastFreeReviveTime = CS.UnityEngine.Time.realtimeSinceStartup
                    DoInstantFreeRevive()
                end
            end
            
            -- 2. BẢO VỆ: Nếu nhân vật ĐÃ SỐNG (HP > 0) mà Popup Hồi Sinh 3s lại trồi lên -> TẮT NGAY
            if not me.isDead and me.hp and me.hp > 0 then
                if _G.UIManager and _G.UIManager.IsVisible and _G.UIID and _G.UIID.Role_ResurgenceUI then
                    if _G.UIManager.IsVisible(_G.UIID.Role_ResurgenceUI) then
                        Log("Phát hiện Popup 3s đếm lùi hiện đè khi đã sống -> Đã tự động dập tắt!")
                        _G.UIManager.Hide(_G.UIID.Role_ResurgenceUI)
                    end
                end
            end
        end)
    end)
end
