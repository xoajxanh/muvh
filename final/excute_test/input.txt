-- =========================================================================
-- [DEBUG SCRIPT]: KIỂM TRA ĐIỀU KIỆN (DETECT SKILL CỰC HẠN, 2 BUFF ELF & MAP)
-- Mục đích: Kiểm tra chính xác trạng thái Skill Cực Hạn, 2 Buff Elf, Map hiện tại
-- Ghi kết quả ra: output.txt (tại persistentDataPath để push_test.ps1 kéo về)
-- =========================================================================

pcall(function()
    local logLines = {}
    local function AddLog(line)
        table.insert(logLines, line)
        if CS.UnityEngine.Debug and CS.UnityEngine.Debug.Log then
            CS.UnityEngine.Debug.Log("[DETECT_DEBUG] " .. tostring(line))
        end
    end

    AddLog("==================================================================")
    AddLog("         BẮT ĐẦU DETECT: SKILL CỰC HẠN, BUFF ELF & BẢN ĐỒ         ")
    AddLog("                 Thời gian: " .. os.date("%Y-%m-%d %H:%M:%S"))
    AddLog("==================================================================")

    local me = _G.RoleManager and _G.RoleManager.me
    if not me then
        AddLog("[LỖI] Không tìm thấy RoleManager.me (Chưa vào game hoặc nhân vật chưa sẵn sàng)!")
        if _G.FloatingWordUtility and _G.FloatingWordUtility.QuickMsg then
            _G.FloatingWordUtility.QuickMsg("LỖI: Chưa tải xong nhân vật!")
        end
        return
    end

    local meId = me.id or (me.data and me.data.id) or 0
    local serverTime = (_G.Time and _G.Time.GetServerTime and _G.Time.GetServerTime()) or 0

    -- =========================================================================
    -- 1. KIỂM TRA BẢN ĐỒ (MAP DETECTION)
    -- =========================================================================
    local mapId = (_G.SceneData and _G.SceneData.mapId) or 0
    local groupId = (_G.SceneData and _G.SceneData.groupId) or 0
    local inCopy = (_G.TranScriptData and _G.TranScriptData.InTranscript) or false
    local isLorencia = (mapId == 1001 or groupId == 1001) and not inCopy

    AddLog(string.format("1. BẢN ĐỒ: mapId=%s | groupId=%s | InTranscript=%s", tostring(mapId), tostring(groupId), tostring(inCopy)))
    local mapStatusStr = ""
    if isLorencia then
        mapStatusStr = "Lorencia (1001) [HỢP LỆ]"
        AddLog("   -> KẾT QUẢ MAP: Đang ở Lorencia [OK]")
    else
        mapStatusStr = string.format("MapID %s%s [KHÔNG PHẢI LOREN]", tostring(mapId), inCopy and " (Trong PB)" or "")
        AddLog("   -> KẾT QUẢ MAP: " .. mapStatusStr)
    end

    -- =========================================================================
    -- 2. KIỂM TRA SKILL CỰC HẠN (SKILL GROUP 410700)
    -- =========================================================================
    local limitSkillGroup = 410700
    local cdTable = me.cd or {}
    local cdData = cdTable[limitSkillGroup]
    local isLimitReady = true
    local limitCdRemain = 0

    if cdData and cdData.endTime and cdData.endTime > serverTime then
        isLimitReady = false
        limitCdRemain = math.max(0, math.floor((cdData.endTime - serverTime) / 1000))
    end

    AddLog(string.format("2. SKILL CỰC HẠN (Group %d):", limitSkillGroup))
    if cdData then
        AddLog(string.format("   - Có thông tin CD: endTime=%s | serverTime=%s | Còn lại=%ss", tostring(cdData.endTime), tostring(serverTime), tostring(limitCdRemain)))
    else
        AddLog("   - Không có thông tin CD (Chưa dùng hoặc đã hồi xong hoàn toàn)")
    end

    local limitStatusStr = ""
    if isLimitReady then
        limitStatusStr = "SẴN SÀNG [OK]"
        AddLog("   -> KẾT QUẢ CỰC HẠN: Sẵn sàng sử dụng [OK]")
    else
        limitStatusStr = string.format("Đang hồi (%ds) [CHƯA XONG]", limitCdRemain)
        AddLog("   -> KẾT QUẢ CỰC HẠN: " .. limitStatusStr)
    end

    -- =========================================================================
    -- 3. KIỂM TRA 2 BUFF CỦA ELF (CHUẨN XÁC: ÁNH SÁNG THỦ HỘ & SỨC MẠNH CHIẾN THẦN)
    -- =========================================================================
    local defBuffGroup = 31000070 -- Ánh Sáng Thủ Hộ (Elf)
    local atkBuffGroup = 31000080 -- Sức Mạnh Chiến Thần (Elf)

    local hasDefBuff = false
    local hasAtkBuff = false
    local defBuffRemain = 0
    local atkBuffRemain = 0
    local defBuffName = ""
    local atkBuffName = ""

    AddLog("3. QUÉT TOÀN BỘ BUFF ĐANG CÓ TRÊN NHÂN VẬT:")
    local allBuffs = {}
    if _G.BuffData and _G.BuffData.GetBuffs then
        allBuffs = _G.BuffData.GetBuffs(meId) or {}
    elseif _G.BuffData and _G.BuffData.BuffDic then
        allBuffs = _G.BuffData.BuffDic[meId] or {}
    end

    local buffCount = 0
    for idx, b in pairs(allBuffs) do
        if b then
            buffCount = buffCount + 1
            local cfg = b.buffConfig or {}
            local bCId = b.buffCId or cfg.id or 0
            local gId = cfg.buffGroup or 0
            local icon = cfg.icon or ""
            local name = cfg.name or "Unknown"
            local timeRemain = b.time or 0

            AddLog(string.format("   [%d] Buff: %s | buffCId=%s | buffGroup=%s | icon=%s | time=%.1fs",
                buffCount, tostring(name), tostring(bCId), tostring(gId), tostring(icon), tonumber(timeRemain) or 0))

            -- KIỂM TRA CHÍNH XÁC Ánh Sáng Thủ Hộ:
            -- Phải là Group 31000070 hoặc icon buff_Defend hoặc tên chứa chính xác "Ánh Sáng Thủ Hộ"
            -- (Tuyệt đối KHÔNG kiểm tra chữ "Thủ Hộ" chung vì sẽ bị dính Ma Hồn Thủ Hộ của MG/Soul)
            local isElfDefBuff = (gId == defBuffGroup) or (icon == "buff_Defend") or string.find(tostring(name), "Ánh Sáng Thủ Hộ")
            if isElfDefBuff and (tonumber(timeRemain) or 0) > 0 then
                hasDefBuff = true
                defBuffRemain = math.floor(tonumber(timeRemain) or 0)
                defBuffName = tostring(name)
            end

            -- KIỂM TRA CHÍNH XÁC Sức Mạnh Chiến Thần:
            -- Phải là Group 31000080 hoặc icon buff_Attack hoặc tên chứa chính xác "Sức Mạnh Chiến Thần"
            local isElfAtkBuff = (gId == atkBuffGroup) or (icon == "buff_Attack") or string.find(tostring(name), "Sức Mạnh Chiến Thần")
            if isElfAtkBuff and (tonumber(timeRemain) or 0) > 0 then
                hasAtkBuff = true
                atkBuffRemain = math.floor(tonumber(timeRemain) or 0)
                atkBuffName = tostring(name)
            end
        end
    end

    if buffCount == 0 then
        AddLog("   -> Hiện tại nhân vật KHÔNG CÓ buff nào!")
    end

    -- Thử kiểm tra bổ sung trực tiếp qua Group ID chuẩn của game
    if not hasDefBuff and _G.BuffData and _G.BuffData.IsHasBuffStateByGroupId then
        hasDefBuff = (_G.BuffData.IsHasBuffStateByGroupId(meId, defBuffGroup) == true)
    end
    if not hasAtkBuff and _G.BuffData and _G.BuffData.IsHasBuffStateByGroupId then
        hasAtkBuff = (_G.BuffData.IsHasBuffStateByGroupId(meId, atkBuffGroup) == true)
    end

    local defStatusStr = hasDefBuff and string.format("CÓ (%ds - %s) [OK]", defBuffRemain, defBuffName) or "THIẾU [X]"
    local atkStatusStr = hasAtkBuff and string.format("CÓ (%ds - %s) [OK]", atkBuffRemain, atkBuffName) or "THIẾU [X]"

    AddLog("------------------------------------------------------------------")
    AddLog("KẾT QUẢ BUFF ELF:")
    AddLog("   * Ánh Sáng Thủ Hộ   : " .. defStatusStr)
    AddLog("   * Sức Mạnh Chiến Thần: " .. atkStatusStr)

    -- =========================================================================
    -- 4. TỔNG HỢP ĐIỀU KIỆN
    -- =========================================================================
    local allConditionsMet = isLorencia and isLimitReady and hasDefBuff and hasAtkBuff

    AddLog("==================================================================")
    if allConditionsMet then
        AddLog(">>> KẾT LUẬN: ĐỦ TẤT CẢ ĐIỀU KIỆN ĐỂ TỰ VÀO THÁP! <<<")
    else
        AddLog(">>> KẾT LUẬN: CHƯA ĐỦ ĐIỀU KIỆN! Chi tiết thiếu:")
        if not isLorencia then AddLog("   - Chưa ở map Lorencia") end
        if not isLimitReady then AddLog("   - Skill Cực Hạn đang hồi chiêu") end
        if not hasDefBuff then AddLog("   - Thiếu buff Ánh Sáng Thủ Hộ (Elf)") end
        if not hasAtkBuff then AddLog("   - Thiếu buff Sức Mạnh Chiến Thần (Elf)") end
    end
    AddLog("==================================================================")

    -- Ghi kết quả vào output.txt theo đúng quy chuẩn pull của push_test.ps1
    pcall(function()
        local fullContent = table.concat(logLines, "\n")
        local outPaths = {
            (CS.UnityEngine.Application and CS.UnityEngine.Application.persistentDataPath) and (CS.UnityEngine.Application.persistentDataPath .. "/output.txt"),
            "/storage/emulated/0/Android/data/com.vnyh.gp/files/output.txt"
        }
        for _, p in ipairs(outPaths) do
            if p and p ~= "" then
                pcall(function()
                    if CS.System.IO.File and CS.System.IO.File.WriteAllText then
                        CS.System.IO.File.WriteAllText(p, fullContent)
                    else
                        local f = io.open(p, "w")
                        if f then f:write(fullContent); f:close() end
                    end
                end)
            end
        end
    end)

    -- Hiển thị thông báo nổi lên màn hình game (FloatingWord)
    local msgLoren = isLorencia and "Map: Loren [OK]" or string.format("Map: %s [SAI]", tostring(mapId))
    local msgLimit = isLimitReady and "Cực Hạn: OK" or string.format("Cực Hạn: CD %ds", limitCdRemain)
    local msgBuff = (hasDefBuff and hasAtkBuff) and "2 Buff: ĐỦ" or string.format("Buff: %s|%s", hasDefBuff and "ThủHộ" or "MấtThủHộ", hasAtkBuff and "CT" or "MấtCT")
    local msgFinal = allConditionsMet and "=> ĐỦ ĐIỀU KIỆN!" or "=> CHƯA ĐỦ!"

    local screenMsg = string.format("[%s] | [%s] | [%s] %s", msgLoren, msgLimit, msgBuff, msgFinal)

    if _G.FloatingWordUtility and _G.FloatingWordUtility.QuickMsg then
        _G.FloatingWordUtility.QuickMsg(screenMsg)
    elseif _G.FloatingTipUtility and _G.FloatingTipUtility.QuickMsg then
        _G.FloatingTipUtility.QuickMsg(screenMsg)
    end
end)
