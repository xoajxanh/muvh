-- ============================================================================
-- SCRIPT TEST GIẢI MÃ TOKEN CLIENT BẰNG INPUT.TXT (BẮT CHI TIẾT LỖI C# & PURE LUA)
-- ============================================================================
local logs = {}
local function logMsg(str)
    print("[TEST DECODE] " .. str)
    table.insert(logs, str)
    if _G.FloatingWordUtility and _G.FloatingWordUtility.QuickMsg then
        _G.FloatingWordUtility.QuickMsg(str)
    end
end

pcall(function()
    logMsg("=== ĐANG CHẠY SCRIPT TEST GIẢI MÃ TOKEN ===")

    local tokenStr = "NjU3MmEzMjI3ZTIxZjBiMDQyNGE1ZDE2OWFkYTVmNTl8MTc2ODI3MjYxNzEyMTY2MDAwfDd8NnwzfDE3ODcwNzY4MzN8M2Y1MmYwNmU3MTg4YzIyODgxZTE0ZmJhYzU0MzRjMjk="
    tokenStr = string.gsub(tokenStr, "^%s*(.-)%s*$", "%1")

    logMsg("Token Input: " .. tokenStr)

    -- 1. Test C# Base64Decode
    local statusC, resC = pcall(function()
        local bytes = CS.System.Convert.FromBase64String(tokenStr)
        return CS.System.Text.Encoding.UTF8:GetString(bytes)
    end)

    if statusC and resC and resC ~= "" then
        logMsg("Base64 C# Decode THANH CONG: " .. resC)
    else
        logMsg("Base64 C# Decode THAT BAI: " .. tostring(resC))
    end

    -- 2. Test Pure Lua Base64Decode (Không phụ thuộc C#)
    local b64chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/'
    local b64lut = {}
    for i = 1, #b64chars do
        b64lut[string.byte(b64chars, i)] = i - 1
    end

    local function PureLuaBase64Decode(data)
        data = string.gsub(data, '[^' .. b64chars .. '=]', '')
        local bitTable = {}
        for i = 1, #data do
            local char = string.sub(data, i, i)
            if char == '=' then break end
            local val = b64lut[string.byte(char)]
            if val then
                for j = 5, 0, -1 do
                    table.insert(bitTable, (math.floor(val / (2 ^ j)) % 2 == 1 and '1' or '0'))
                end
            end
        end
        local result = {}
        for i = 1, #bitTable - 7, 8 do
            local chunk = table.concat(bitTable, '', i, i + 7)
            local byteVal = 0
            for j = 1, 8 do
                if string.sub(chunk, j, j) == '1' then
                    byteVal = byteVal + 2 ^ (8 - j)
                end
            end
            table.insert(result, string.char(byteVal))
        end
        return table.concat(result)
    end

    local statusL, resL = pcall(function()
        return PureLuaBase64Decode(tokenStr)
    end)

    local decodedStr = ""
    if statusL and resL and resL ~= "" then
        logMsg("Base64 Pure Lua Decode THANH CONG: " .. resL)
        decodedStr = resL
    else
        logMsg("Base64 Pure Lua Decode THAT BAI: " .. tostring(resL))
    end

    if decodedStr == "" and (statusC and resC) then
        decodedStr = resC
    end

    if decodedStr ~= "" then
        -- 3. Split chuỗi
        local parts = {}
        for part in string.gmatch(decodedStr .. "|", "(.-)|") do
            table.insert(parts, part)
        end
        if #parts > 0 and parts[#parts] == "" then
            table.remove(parts, #parts)
        end

        logMsg("Tong so phan (Parts): " .. tostring(#parts))
        for i, p in ipairs(parts) do
            logMsg(string.format("Part[%d] = %s", i, p))
        end

        if #parts == 7 then
            local pCode, pUID, pMainTier, pSubTier, pDuration, pTime, pSig =
                parts[1], parts[2], tonumber(parts[3]), tonumber(parts[4]), tonumber(parts[5]), tonumber(parts[6]), parts[7]

            local dataToHash = pCode .. "|" .. pUID .. "|" .. tostring(pMainTier) .. "|" .. tostring(pSubTier) .. "|" .. tostring(pDuration) .. "|" .. tostring(pTime) .. "MUVH_SECRET_SALT_XOAI"
            logMsg("Data to Hash: " .. dataToHash)

            local expectedSig = string.lower(tostring(CS.PCUtility.Md5(dataToHash)))
            logMsg("Expected Sig: " .. expectedSig)
            logMsg("Token Sig   : " .. tostring(pSig))

            if string.lower(expectedSig) == string.lower(pSig) then
                logMsg("=> CHU KY MD5 TRUNG KHOP 100%! XAC THUC THANH CONG!")
            else
                logMsg("=> LOI: CHU KY MD5 KHONG TRUNG KHOP!")
            end
        end
    end

    -- Ghi log ra file output.txt để xem chi tiết
    pcall(function()
        local path = CS.UnityEngine.Application.persistentDataPath .. "/output.txt"
        local file = io.open(path, "w")
        if file then
            file:write(table.concat(logs, "\n"))
            file:close()
        end
    end)
end)
