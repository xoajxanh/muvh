-- 3. HÀM TÌM CẤU HÌNH TẦNG PHÓ BẢN THEO LEVEL
-- =========================================================================
local function GetDungeonConfigForLevel(isBloodCastle, lvl)
    lvl = tonumber(lvl or GetCurLvl())
    local result = {
        name = isBloodCastle and "Huyết Lâu" or "Quảng Trường Quỷ",
        tier = 1,
        mapId = isBloodCastle and 1012001 or 1010001,
        transferId = isBloodCastle and 101200101 or 101000101,
        minLvl = 60,
        maxLvl = 99,
        cost = "1000080#1",
        ticketId = 1000080,
        ticketCount = 1
    }

    pcall(function()
        if _G.TranScriptData and _G.TranScriptData.GetEnterConditionData then
            local subType = isBloodCastle and _G.TranScriptData.TranScriptSubType.BloodCastle or _G.TranScriptData.TranScriptSubType.DemonPlaza
            local condId = isBloodCastle and 100414 or 100408
            local maxNumId = isBloodCastle and 100404 or 100702
            local enterData, levelInfo = _G.TranScriptData.GetEnterConditionData(subType, condId, maxNumId, lvl)
            
            if enterData and enterData.id then
                result.transferId = tonumber(enterData.id)
                result.mapId = tonumber(enterData.groupId or (_G.TranScriptData.ContentData and _G.TranScriptData.ContentData.mapId) or result.mapId)
                result.cost = enterData.cost or result.cost
                if enterData.name then result.name = tostring(enterData.name) end
            end
            if levelInfo and #levelInfo >= 3 then
                result.tier = tonumber(levelInfo[1]) or result.tier
                result.minLvl = tonumber(levelInfo[2]) or result.minLvl
                result.maxLvl = tonumber(levelInfo[3]) or result.maxLvl
            end
        end
    end)

    if not result.transferId or result.transferId < 100000000 then
        local tiers = isBloodCastle and HL_TIERS or QTQ_TIERS
        for _, t in ipairs(tiers) do
            if lvl >= t.minLvl and lvl <= t.maxLvl then
                result.tier = t.tier
                result.mapId = t.mapId
                result.transferId = t.transferId
                result.minLvl = t.minLvl
                result.maxLvl = t.maxLvl
                result.name = string.format("%s %d", isBloodCastle and "Huyết Lâu" or "Quảng Trường Quỷ", t.tier)
                break
            end
        end
        if lvl > tiers[#tiers].maxLvl then
            local last = tiers[#tiers]
            result.tier = last.tier
            result.mapId = last.mapId
            result.transferId = last.transferId
            result.minLvl = last.minLvl
            result.maxLvl = last.maxLvl
            result.name = string.format("%s %d", isBloodCastle and "Huyết Lâu" or "Quảng Trường Quỷ", last.tier)
        end
    end

    if result.cost then
        local parts = string.split(result.cost, "#")
        result.ticketId = tonumber(parts[1]) or result.ticketId
        result.ticketCount = tonumber(parts[2]) or 1
    end

    return result
end

-- 4. HÀM MỞ TOÀN BỘ RƯƠNG QUÀ / GÓI QUÀ / VẬT PHẨM TRONG TÚI (AUTO OPEN CHESTS)
-- =========================================================================
local function AutoOpenAllChestsAndBags()
    Log("-> Đang kiểm tra và mở các rương/gói quà trong túi đồ...", true)
    pcall(function()
        local openedCount = 0
        if _G.BagInfoData and _G.BagInfoData.TotalItems then
            for _, item in pairs(_G.BagInfoData.TotalItems) do
                if item and item.id and item.tblItem then
                    local isChest = (item.tblItem.type == 2) or (item.tblItem.subType == 2) or (item.tblItem.subType == 21) or (item.tblItem.quickUse and item.tblItem.quickUse > 0)
                    local name = tostring(item.tblItem.name or "")
                    local isGiftBox = string.find(name, "Quà") or string.find(name, "Rương") or string.find(name, "Gói") or string.find(name, "Hộp") or string.find(name, "Túi") or string.find(name, "Tân Thủ")
                    
                    if (isChest or isGiftBox) and item.tblItem.type ~= 1 then
                        local useNum = tonumber(item.count or 1)
                        if useNum > 0 then
                            openedCount = openedCount + 1
                            Log(string.format("   [MỞ QUÀ] Đang mở %dx '%s' (ID: %d)...", useNum, name, item.id))
                            if _G.networkRequest and _G.networkRequest.ReqUseItem then
                                _G.networkRequest.ReqUseItem(useNum, item.id)
                            elseif _G.NetManager and _G.BagMessage and _G.BagMessage.ReqUseItem then
                                _G.NetManager.Send(_G.BagMessage.ReqUseItem, { itemId = item.id, count = useNum, clientParams = {} })
                            end
                        end
                    end
                end
            end
        end
        if openedCount > 0 then
            Log(string.format("-> Đã gửi lệnh mở %d loại hộp/rương quà!", openedCount), true)
        else
            Log("-> Không tìm thấy hộp quà nào cần mở thêm trong túi.")
        end
    end)
end
_G.AutoOpenAllChestsAndBags = AutoOpenAllChestsAndBags

-- =========================================================================
local function RunAutoGiftcode(onFinished)
    local total = #GIFT_CODES
    local idx = 1
    Log("=========================================================================", true)
    Log(string.format(">>> BẮT ĐẦU NHẬP %d MÃ GIFTCODE VÀ NHẬN QUÀ HÒM THƯ <<<", total), true)
    Log("=========================================================================", true)

    local codeLoop = nil
    codeLoop = Timer.StartLoop(0.2, -1, function()
        if idx <= total then
            local code = GIFT_CODES[idx]
            Log(string.format("[GIFTCODE] Đang gửi mã [%02d/%d]: %s", idx, total, code))
            pcall(function()
                if _G.NetManager and _G.BagMessage and _G.BagMessage.ReqUseCDKey then
                    _G.NetManager.Send(_G.BagMessage.ReqUseCDKey, { cdKey = code })
                elseif _G.networkRequest and _G.networkRequest.ReqUseCDKey then
                    _G.networkRequest.ReqUseCDKey(code)
                end
            end)
            idx = idx + 1
        else
            if codeLoop then Timer.Stop(codeLoop) end
            Log("[GIFTCODE] Đã gửi xong 34 mã! Chờ 2.5s Server tạo thư quà...", true)

            Timer.Start(2.5, function()
                Log("[GIFTCODE] Đang đồng bộ danh sách hòm thư (ReqGetMailList)...", true)
                pcall(function()
                    if _G.NetManager and _G.MailMessage and _G.MailMessage.ReqGetMailList then
                        _G.NetManager.Send(_G.MailMessage.ReqGetMailList, {})
                    end
                end)

                Timer.Start(1.5, function()
                    Log("[GIFTCODE] Đang nhận tất cả quà đính kèm vào túi (ReqGetMailItems)...", true)
                    pcall(function()
                        if _G.NetManager and _G.MailMessage and _G.MailMessage.ReqGetMailItems then
                            _G.NetManager.Send(_G.MailMessage.ReqGetMailItems)
                        end
                    end)

                    Timer.Start(1.5, function()
                        -- Tự động mở các hộp quà / rương quà vừa nhận
                        AutoOpenAllChestsAndBags()

                        Timer.Start(1.0, function()
                            Log("=========================================================================", true)
                            Log(">>> HOÀN TẤT NHẬP 34 GIFTCODE & NHẬN TOÀN BỘ QUÀ THÀNH CÔNG! <<<", true)
                            Log("=========================================================================", true)
                            
                            if onFinished then
                                pcall(function() onFinished() end)
                            end
                        end)
                    end)
                end)
            end)
        end
    end)
end
_G.RunAutoGiftcode = RunAutoGiftcode

-- 6. HÀM IN SNAPSHOT LƯỢT & VÉ & THỜI GIAN ĐẾM NGƯỢC
-- =========================================================================
local function PrintDungeonSnapshot()
    local curLvl = GetCurLvl()
    local curMap = GetCurMap()
    
    local hlLeftCount = 0
    local dsLeftCount = 0
    pcall(function()
        local gTblHL = _G.ClientTable.cfg_Global_globalManager:TryGetValue(60000001)
        if gTblHL and gTblHL.effect and _G.RefreshData and _G.RefreshData.GetInstanceCount then
            hlLeftCount = _G.RefreshData.GetInstanceCount(tonumber(gTblHL.effect)) or 0
        end
        local gTblDS = _G.ClientTable.cfg_Global_globalManager:TryGetValue(60000002)
        if gTblDS and gTblDS.effect and _G.RefreshData and _G.RefreshData.GetInstanceCount then
            dsLeftCount = _G.RefreshData.GetInstanceCount(tonumber(gTblDS.effect)) or 0
        end
    end)

    local hlCfg = GetDungeonConfigForLevel(true, curLvl)
    local dsCfg = GetDungeonConfigForLevel(false, curLvl)

    local hlTickets = 0
    local dsTickets = 0
    pcall(function()
        if _G.BagInfoData and _G.BagInfoData.GetItemTotalCountByItemId then
            hlTickets = _G.BagInfoData.GetItemTotalCountByItemId(hlCfg.ticketId) or 0
            dsTickets = _G.BagInfoData.GetItemTotalCountByItemId(dsCfg.ticketId) or 0
            if hlTickets == 0 then hlTickets = _G.BagInfoData.GetItemTotalCountByItemId(100414) or 0 end
            if dsTickets == 0 then dsTickets = _G.BagInfoData.GetItemTotalCountByItemId(100413) or 0 end
        end
    end)

    local serverSec = 0
    pcall(function()
        if _G.Time and _G.Time.GetServerSecondTime then
            serverSec = _G.Time.GetServerSecondTime()
        end
    end)
    if serverSec == 0 then serverSec = os.time() end

    local isFirstEnterHL = false
    local isFirstEnterDS = false
    local hlRemainSec = 0
    local dsRemainSec = 0

    pcall(function()
        if _G.RefreshData and _G.RefreshData.CheckIsFirstEnter then
            isFirstEnterHL = _G.RefreshData:CheckIsFirstEnter(3020402)
            isFirstEnterDS = _G.RefreshData:CheckIsFirstEnter(4020202)
        end
    end)

    pcall(function()
        if _G.TranScriptData and _G.TranScriptData.RefreshOpenTime then
            _G.TranScriptData.RefreshOpenTime(hlCfg.mapId)
            local st = _G.TranScriptData.startTime or 0
            hlRemainSec = st - serverSec
            if hlRemainSec < 0 then hlRemainSec = 0 end

            _G.TranScriptData.RefreshOpenTime(dsCfg.mapId)
            local stDS = _G.TranScriptData.startTime or 0
            dsRemainSec = stDS - serverSec
            if dsRemainSec < 0 then dsRemainSec = 0 end
        end
    end)

    local hlStatusStr = isFirstEnterHL and "ĐẶC QUYỀN TÂN THỦ (VÀO NGAY 24/24)" or (hlRemainSec <= 0 and "ĐANG MỞ CỬA!" or string.format("Chờ mở cửa sau %d giây (%02d:%02d)", hlRemainSec, math.floor(hlRemainSec/60), hlRemainSec%60))
    local dsStatusStr = isFirstEnterDS and "ĐẶC QUYỀN TÂN THỦ (VÀO NGAY 24/24)" or (dsRemainSec <= 0 and "ĐANG MỞ CỬA!" or string.format("Chờ mở cửa sau %d giây (%02d:%02d)", dsRemainSec, math.floor(dsRemainSec/60), dsRemainSec%60))

    Log("=========================================================================", true)
    Log(string.format(">>> [SNAPSHOT PHÓ BẢN] LEVEL: %d | MAP HIỆN TẠI: %d <<<", curLvl, curMap), true)
    Log(string.format("-> HUYẾT LÂU: [%s] (Map: %d | TransferId: %s)", hlCfg.name, hlCfg.mapId, tostring(hlCfg.transferId)), true)
    Log(string.format("   Trạng thái: %s", hlStatusStr), true)
    Log(string.format("   Cấp yêu cầu: Lv.%d ~ Lv.%d | Lượt đi: %d | Vé ID %d: %d", 
        hlCfg.minLvl, hlCfg.maxLvl, hlLeftCount, hlCfg.ticketId, hlTickets), true)
    Log(string.format("-> QUẢNG TRƯỜNG QUỶ: [%s] (Map: %d | TransferId: %s)", dsCfg.name, dsCfg.mapId, tostring(dsCfg.transferId)), true)
    Log(string.format("   Trạng thái: %s", dsStatusStr), true)
    Log(string.format("   Cấp yêu cầu: Lv.%d ~ Lv.%d | Lượt đi: %d | Vé ID %d: %d", 
        dsCfg.minLvl, dsCfg.maxLvl, dsLeftCount, dsCfg.ticketId, dsTickets), true)
    Log(string.format("-> Server Time: %s (Unix: %d)", os.date("%H:%M:%S", serverSec), serverSec), true)
    Log("=========================================================================", true)

    return {
        curLvl = curLvl,
        curMap = curMap,
        hlLeft = hlLeftCount,
        dsLeft = dsLeftCount,
        hlTickets = hlTickets,
        dsTickets = dsTickets,
        hlCfg = hlCfg,
        dsCfg = dsCfg,
        isFirstEnterHL = isFirstEnterHL,
        isFirstEnterDS = isFirstEnterDS,
        hlRemainSec = hlRemainSec,
        dsRemainSec = dsRemainSec
    }
end
_G.PrintDungeonSnapshot = PrintDungeonSnapshot

    local curLvl = GetCurLvl()

    if tier and HL_TIERS[tier] then

        cfg.tier = t.tier

        cfg.minLvl = t.minLvl
        cfg.maxLvl = t.maxLvl
        cfg.name = string.format("Huyết Lâu %d", t.tier)
    end

        cfg.name, tostring(cfg.transferId), cfg.mapId), true)

            if _G.EventManager and _G.Event and _G.Event.Map_ChangeMap then
                _G.EventManager.Dispatch(_G.Event.Map_ChangeMap, { mapId = cfg.transferId })
            end
        end)
    elseif method == 2 then
        Log(string.format("-> NetManager.Send ReqTransferTransmit (transferId = %s)...", tostring(cfg.transferId)), true)
        pcall(function()
            if _G.NetManager and _G.MapMessage and _G.MapMessage.ReqTransferTransmit then
                _G.NetManager.Send(_G.MapMessage.ReqTransferTransmit, {
                    transferId = cfg.transferId,
                    line = 0
                })
            end

    end

    pcall(function()
        if _G.UIManager then
            if _G.UIID and _G.UIID.Instance_BloodCastleUI then _G.UIManager.Hide(_G.UIID.Instance_BloodCastleUI) end
            if _G.UIID and _G.UIID.PromptTipUI then _G.UIManager.Hide(_G.UIID.PromptTipUI) end
        end
    end)

    Timer.Start(2.0, function()
        local afterMap = GetCurMap()
        local inBC = (afterMap >= 1012000 and afterMap <= 1012099)
        if inBC then
            Log(string.format(">>> [THÀNH CÔNG] Nhân vật đã vào trong Huyết Lâu (Map %d)! <<<", afterMap), true)
            TriggerSkipWait()
            Timer.Start(1.0, function()
                pcall(function()
                    local me = _G.RoleManager and _G.RoleManager.me
                    if me then
                        if me.MoveTo then me:MoveTo({ x = 14, y = 33 }, 0) end
                        if me.SetAutoFight then me:SetAutoFight(_G.AutoFightStrKey and _G.AutoFightStrKey.AutoFight or "AutoFight") end
                    end
                end)
            end)
        else
            Log(string.format("-> [KẾT QUẢ]: Chưa vào được Huyết Lâu (Map hiện tại: %d).", afterMap), true)
        end
    end)
end
_G.TestEnterBloodCastle = TestEnterBloodCastle
_G.EnterHL = TestEnterBloodCastle

-- =========================================================================
-- 9. TRUYỀN TỐNG VÀO QUẢNG TRƯỜNG QUỶ (TEST ENTER DEMON PLAZA)
-- =========================================================================
local function TestEnterDemonPlaza(tier, method)
    local curLvl = GetCurLvl()
    local cfg = GetDungeonConfigForLevel(false, curLvl)
    if tier and QTQ_TIERS[tier] then
        local t = QTQ_TIERS[tier]
        cfg.tier = t.tier
        cfg.mapId = t.mapId
        cfg.transferId = t.transferId
        cfg.minLvl = t.minLvl
        cfg.maxLvl = t.maxLvl
        cfg.name = string.format("Quảng Trường Quỷ %d", t.tier)
    end
    method = method or 1

    Log(string.format("=== [VÀO QUẢNG TRƯỜNG QUỶ] TẦNG: %s (TransferId: %s | MapId: %d) ===", 
        cfg.name, tostring(cfg.transferId), cfg.mapId), true)

    Log(string.format("-> Gửi Event.Map_ChangeMap (mapId = %s)...", tostring(cfg.transferId)), true)
    pcall(function()
        if _G.EventManager and _G.Event and _G.Event.Map_ChangeMap then
            _G.EventManager.Dispatch(_G.Event.Map_ChangeMap, { mapId = cfg.transferId })
        end
    end)

    pcall(function()
        if _G.UIManager then
            if _G.UIID and _G.UIID.Instance_DemonPlazaUI then _G.UIManager.Hide(_G.UIID.Instance_DemonPlazaUI) end
            if _G.UIID and _G.UIID.PromptTipUI then _G.UIManager.Hide(_G.UIID.PromptTipUI) end
        end
    end)

    Timer.Start(2.0, function()
        local afterMap = GetCurMap()
        local inDS = (afterMap >= 1010000 and afterMap <= 1010099)
        if inDS then
            Log(string.format(">>> [THÀNH CÔNG] Nhân vật đã vào trong Quảng Trường Quỷ (Map %d)! <<<", afterMap), true)
            TriggerSkipWait()
            Timer.Start(1.0, function()
                pcall(function()
                    local me = _G.RoleManager and _G.RoleManager.me
                    if me then
                        local cx, cy = 137, 167
                        if curLvl >= 150 then cx, cy = 135, 97 end
                        if me.MoveTo then me:MoveTo({ x = cx, y = cy }, 0) end
                        if me.SetAutoFight then me:SetAutoFight(_G.AutoFightStrKey and _G.AutoFightStrKey.AutoFight or "AutoFight") end
                    end
                end)
            end)
        else
            Log(string.format("-> [KẾT QUẢ]: Chưa vào được Quảng Trường Quỷ (Map hiện tại: %d).", afterMap), true)
        end
    end)
end
_G.TestEnterDemonPlaza = TestEnterDemonPlaza
_G.EnterDS = TestEnterDemonPlaza

-- =========================================================================
-- 10. THỰC THI CHÍNH KHI BẤM NÚT [EXEC]
-- =========================================================================
local curMap = GetCurMap()
local inBC = (curMap >= 1012000 and curMap <= 1012099)
local inDS = (curMap >= 1010000 and curMap <= 1010099)

if inBC or inDS then
    Log(string.format(">>> ĐANG TRONG PHÓ BẢN (MAP %d) -> KÍCH HOẠT [MỞ NGAY] & AUTO CHIẾN ĐẤU! <<<", curMap), true)
    TriggerSkipWait()
    pcall(function()
        local me = _G.RoleManager and _G.RoleManager.me
        if me and me.SetAutoFight then
            me:SetAutoFight(_G.AutoFightStrKey and _G.AutoFightStrKey.AutoFight or "AutoFight")
        end
    end)
else
    -- Bước 1: Chạy tự động nhập 34 Code & Nhận quà hòm thư & Mở rương quà
    RunAutoGiftcode(function()
        -- Bước 2: In snapshot kiểm tra vé và thời gian
        local info = PrintDungeonSnapshot()

        -- Bước 3: Tự động vào Huyết Lâu (hoặc QTQ nếu đang mở)
        if info.isFirstEnterHL or info.hlRemainSec <= 0 then
            Log(string.format(">>> ĐÃ CÓ VÉ & ĐẾN GIỜ MỞ! TỰ ĐỘNG VÀO [%s] (Map %d | TransferId: %s)... <<<", 
                info.hlCfg.name, info.hlCfg.mapId, tostring(info.hlCfg.transferId)), true)
            TestEnterBloodCastle(nil, 1)
        elseif info.isFirstEnterDS or info.dsRemainSec <= 0 then
            Log(string.format(">>> HUYẾT LÂU CHƯA MỞ (CÒN %ds) -> CHUYỂN SANG VÀO QUẢNG TRƯỜNG QUỶ [%s] (Map %d)... <<<", 
                info.hlRemainSec, info.dsCfg.name, info.dsCfg.mapId), true)
            TestEnterDemonPlaza(nil, 1)
        else
            Log(string.format(">>> CẢ 2 PHÓ BẢN ĐỀU CHƯA MỞ (HL còn %ds, QTQ còn %ds) -> Gửi lệnh vào Huyết Lâu [%s]... <<<", 
                info.hlRemainSec, info.dsRemainSec, info.hlCfg.name), true)
            TestEnterBloodCastle(nil, 1)
        end
    end)
end
