local function SafeDumpTable(t, maxDepth, currentDepth)
    maxDepth = maxDepth or 2
    currentDepth = currentDepth or 0
    if currentDepth > maxDepth then return "{...}" end
    if type(t) ~= "table" then return tostring(t) end
    
    local str = "{\n"
    for k, v in pairs(t) do
        local keyStr = tostring(k)
        local valStr = ""
        if type(v) == "table" then
            valStr = SafeDumpTable(v, maxDepth, currentDepth + 1)
        elseif type(v) == "string" then
            valStr = "\"" .. v .. "\""
        else
            valStr = tostring(v)
        end
        str = str .. string.rep("  ", currentDepth + 1) .. keyStr .. " = " .. valStr .. ",\n"
    end
    str = str .. string.rep("  ", currentDepth) .. "}"
    return str
end

local function CheckDungItems()
    local outPath = ""
    if CS and CS.UnityEngine and CS.UnityEngine.Application then
        outPath = CS.UnityEngine.Application.persistentDataPath .. "/output.txt"
    else
        outPath = "/storage/emulated/0/Android/data/com.vnyh.gp/files/output.txt"
    end

    local file = io.open(outPath, "w")
    
    local function LogMsg(msg)
        if file then
            file:write(msg .. "\n")
        else
            print(msg)
        end
    end

    local items = _G.BagInfoData and _G.BagInfoData.TotalItems
    if not items then 
        LogMsg("[DUMP] BagInfoData.TotalItems is nil")
        if file then file:close() end
        return 
    end

    LogMsg("================ BẮT ĐẦU QUÉT TÚI TÌM ĐỒ DUNG (C8+) ================")
    local found = 0
    
    for k, item in pairs(items) do
        if item then
            local tblItem = item.tblItem or (item.data and item.data.tblItem) or {}
            local needLevel = tblItem.needLevel or 0
            local tier = math.floor(needLevel / 400)
            
            -- Chỉ quét trang bị (type == 2) từ Chuyển 8 trở lên
            if tblItem.type == 2 and tier >= 8 then
                local sInfo = item.serverInfo or item.serverData or {}
                
                local dumpStr = SafeDumpTable(sInfo, 3)
                
                local msg = string.format("[DUMP_ITEM] Tên: %s | Cấp: %d (C%d) | BagIndex: %s\n-> serverInfo: %s\n-----------------------------------------", 
                    tostring(tblItem.name), needLevel, tier, tostring(k), dumpStr)
                LogMsg(msg)
                found = found + 1
            end
        end
    end
    
    LogMsg("================ HOÀN TẤT. QUÉT ĐƯỢC " .. found .. " MÓN ================")
    
    if file then 
        file:close() 
    end
    if _G.WriteLog then _G.WriteLog("Đã dump xong đồ Dung ra file output.txt, hãy chạy lệnh adb pull hoặc ấn Enter trên tool.") end
end

pcall(CheckDungItems)
