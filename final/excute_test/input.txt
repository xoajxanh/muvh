local function Mod_IsAtTrainSpot_New()
    if not _G.Mod_TrainCoord or _G.Mod_TrainCoord == "" then return false end
    if _G.Mod_TrainArrivedAtPos then return true end

    local coordStr = _G.Mod_TrainCoord
    local sx, sy = string.match(coordStr, "^(%d+)#(%d+)$")
    local tMap = nil
    if not sx then
        tMap, sx, sy = string.match(coordStr, "^(%d+);(%d+)#(%d+)$")
    end
    if sx and sy then
        local tx, ty = tonumber(sx), tonumber(sy)
        local curMap = _G.SceneData and (_G.SceneData.mapId or _G.SceneData.groupId) or 0
        local expectedMap = tonumber(tMap)
        if not expectedMap then
            local primaryTier = _G.Mod_Config_Reincarnation_Primary or (CS.UnityEngine.PlayerPrefs and CS.UnityEngine.PlayerPrefs.GetInt("Mod_PrimaryTier", 8)) or 8
            local primaryTag = "C" .. tostring(primaryTier)
            local tab = _G.ModAutoBossConfigTab or primaryTag
            local mapsConfig = _G.GetMapsConfigByTier and _G.GetMapsConfigByTier(tab)
            if not mapsConfig or #mapsConfig == 0 then mapsConfig = _G.GetMapsConfigByTier and _G.GetMapsConfigByTier(primaryTag) end
            expectedMap = (mapsConfig and mapsConfig[1] and mapsConfig[1].mapId)
        end
        if expectedMap and curMap == expectedMap then
            local me = _G.RoleManager and _G.RoleManager.me
            local meX = (me and me.serverCoord and me.serverCoord.x) or (me and me.cellPos and me.cellPos.x) or 0
            local meY = (me and me.serverCoord and me.serverCoord.y) or (me and me.cellPos and me.cellPos.y) or 0
            local dist = math.sqrt((meX - tx)^2 + (meY - ty)^2)
            if dist <= 15 then
                _G.Mod_TrainArrivedAtPos = true
                return true
            end
        end
    end
    return false
end
_G.Mod_IsAtTrainSpot = Mod_IsAtTrainSpot_New

local function AutoDungeonMonitorStep_Fixed()
    if not (_G.Mod_IsActive and _G.Mod_IsActive()) then return end

    if not _G.Mod_AutoBloodCastle_Enabled and not _G.Mod_AutoDemonPlaza_Enabled then
        _G.Mod_IsRunningDungeon = false
        _G.Mod_IsPriorityDungeon = false
        _G.Mod_DungeonJustFinished = false
        return
    end

    local curMap = (_G.SceneData and (_G.SceneData.mapId or _G.SceneData.groupId)) or 0
    local isBC = (curMap >= 1012000 and curMap <= 1012099)
    local isDS = (curMap >= 1010000 and curMap <= 1010099)
    local inDungeon = (isBC and _G.Mod_AutoBloodCastle_Enabled) or (isDS and _G.Mod_AutoDemonPlaza_Enabled)

    if inDungeon then
        _G.Mod_IsRunningDungeon = true
        _G.Mod_IsPriorityDungeon = true
        _G.Mod_DungeonJustFinished = false

        if _G.TriggerDungeonSkipWait then _G.TriggerDungeonSkipWait() end

        -- Dùng thẻ EXP
        if _G.Mod_AutoUseExpCard_Enabled then
            if (_G.Mod_DungeonExpCardsUsed or 0) < 2 then
                local mId = _G.RoleManager and _G.RoleManager.me and _G.RoleManager.me.id
                local bList = (_G.BuffData and _G.BuffData.GetBuffs and _G.BuffData.GetBuffs(mId)) or {}
                local expBuffRemainTime = 0
                for _, b in pairs(bList) do
                    if b and b.buffConfig and b.buffConfig.name then
                        local bName = b.buffConfig.name
                        if bName:find("EXP") or bName:find("Kinh Nghiệm") then
                            local rTime = tonumber(b.time) or 0
                            if rTime > expBuffRemainTime then expBuffRemainTime = rTime end
                        end
                    end
                end

                if expBuffRemainTime < 600 then
                    local cardItem = nil
                    if _G.BagInfoData and _G.BagInfoData.GetItemByConfigID then
                        pcall(function() cardItem = _G.BagInfoData.GetItemByConfigID(3000400) end)
                    end
                    if not cardItem and _G.BagInfoData and _G.BagInfoData.TotalItems then
                        for _, it in pairs(_G.BagInfoData.TotalItems) do
                            if it and (it.itemId == 3000400 or (it.tblItem and it.tblItem.name and it.tblItem.name:find("Buff EXP"))) then
                                cardItem = it
                                break
                            end
                        end
                    end

                    if cardItem then
                        local now = CS.UnityEngine.Time.realtimeSinceStartup
                        if (now - (_G.Mod_LastExpCardUseTime or 0)) >= 2.0 then
                            _G.Mod_LastExpCardUseTime = now
                            _G.Mod_DungeonExpCardsUsed = (_G.Mod_DungeonExpCardsUsed or 0) + 1
                            pcall(function()
                                if _G.ItemUtility and _G.ItemUtility.UseItem then
                                    _G.ItemUtility.UseItem({
                                        useCount = 1,
                                        useItemId = cardItem.id,
                                        configId = cardItem.itemId,
                                        useParam = cardItem.tblItem and cardItem.tblItem.useParam,
                                        params = nil,
                                        itemInfo = cardItem
                                    })
                                elseif _G.BagInfoController and _G.BagInfoController.UseItemReq then
                                    _G.BagInfoController.UseItemReq(1, cardItem.id, nil, cardItem.itemId)
                                end
                            end)
                            if _G.FloatingWordUtility then
                                _G.FloatingWordUtility.QuickMsg(string.format("Đã dùng Thẻ Buff EXP (%d/2)", _G.Mod_DungeonExpCardsUsed))
                            end
                        end
                    end
                end
            end
        end

        -- Di chuyển cắm đầu chạy khi dist > 1.5m
        local targetCoordStr = nil
        if isBC then
            targetCoordStr = _G.Mod_Coord_BloodCastle
            if not targetCoordStr or targetCoordStr == "" then targetCoordStr = "14#60" end
        elseif isDS then
            targetCoordStr = _G.Mod_Coord_DemonPlaza
            if not targetCoordStr or targetCoordStr == "" then
                local curL = tonumber((pMe and pMe.level) or 1)
                targetCoordStr = (curL >= 150) and "135#97" or "137#167"
            end
        end

        if targetCoordStr and targetCoordStr ~= "" then
            local sx, sy = string.match(targetCoordStr, "^(%d+)#(%d+)$")
            if not sx then _, sx, sy = string.match(targetCoordStr, "^(%d+);(%d+)#(%d+)$") end
            if sx and sy then
                local tx, ty = tonumber(sx), tonumber(sy)
                local me = _G.RoleManager and _G.RoleManager.me
                if me and not me.isDead then
                    local curX = (me.serverCoord and me.serverCoord.x) or (me.cellPos and me.cellPos.x) or (me.Position and me.Position.x) or 0
                    local curY = (me.serverCoord and me.serverCoord.y) or (me.cellPos and me.cellPos.y) or (me.Position and me.Position.y) or 0
                    local dist = math.sqrt((curX - tx)^2 + (curY - ty)^2)

                    if dist > 1.5 then
                        -- CẮM ĐẦU CHẠY KHÔNG ĐƯỢC ĐÁNH
                        if me.isAutoFight then
                            if me.StopAutoFight then me:StopAutoFight() end
                            if _G.EventManager and _G.Event and _G.Event.Role_StopAutoFight then
                                _G.EventManager.Dispatch(_G.Event.Role_StopAutoFight)
                            end
                        end
                        me.TargetAvatar = nil
                        if me.MoveTo then me:MoveTo({ x = tx, y = ty }, 0) end
                        return
                    else
                        -- ĐÃ TỚI GẦN VỊ TRÍ <= 1.5m -> BẬT AUTO FIGHT
                        if me.SetAutoFight and not me.isAutoFight then
                            me:SetAutoFight(_G.AutoFightStrKey and _G.AutoFightStrKey.AutoFight or "AutoFight")
                        end
                        return
                    end
                end
            end
        end

        local me = _G.RoleManager and _G.RoleManager.me
        if me and not me.isDead and me.SetAutoFight and not me.isAutoFight then
            me:SetAutoFight(_G.AutoFightStrKey and _G.AutoFightStrKey.AutoFight or "AutoFight")
        end
        return
    end

    -- Ngoài phó bản
    if _G.Mod_IsRunningDungeon then
        _G.Mod_IsRunningDungeon = false
        _G.Mod_DungeonExpCardsUsed = 0
        _G.Mod_LastDungeonExitTime = CS.UnityEngine.Time.realtimeSinceStartup
        _G.Mod_DungeonJustFinished = true
    end

    local curH = tonumber(os.date("%H")) or 0
    local curM = tonumber(os.date("%M")) or 0
    if curM >= 15 then
        _G.Mod_IsPriorityDungeon = false
        if _G.Mod_DungeonJustFinished then
            _G.Mod_DungeonJustFinished = false
            if _G.Mod_TrainCoord and _G.Mod_TrainCoord ~= "" then
                if _G.Mod_ReturnToTrainAfterDungeon then _G.Mod_ReturnToTrainAfterDungeon() end
            end
        end
        return
    end
end
