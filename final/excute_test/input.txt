-- Script Test Vô hạn phó bản Huyết Lâu / Quảng Trường Quỷ
-- Chạy script này SAU KHI ĐÃ MỞ GIAO DIỆN Huyết Lâu hoặc QTQ trên màn hình để lấy đúng mapId hiện tại của nhân vật.

local isSuccess = false

-- 1. Xử lý Huyết Lâu (Blood Castle)
if _G.Instance_BloodCastleUI and _G.Instance_BloodCastleUI.ContentData then
    local mapId = _G.Instance_BloodCastleUI.ContentData.mapId
    if _G.NetManager and _G.InstanceMatchMessage then
        _G.NetManager.Send(_G.InstanceMatchMessage.ReqInstanceMatchCreateTeam, {
            instanceId = mapId
        })
        
        -- THỬ GỬI THÊM LỆNH ENTER VÀO MAP TRỰC TIẾP (BỎ QUA MATCHING TEAM)
        if _G.InstanceMessage and _G.InstanceMessage.ReqEnterInstance then
            _G.NetManager.Send(_G.InstanceMessage.ReqEnterInstance, {
                instanceId = mapId
            })
        end
        
        if _G.UIManager then _G.UIManager.Hide(_G.UIID.Instance_BloodCastleUI) end
        
        local msg = "[TEST] Đã gửi ReqInstanceMatchCreateTeam & ReqEnterInstance Huyết Lâu với mapId: " .. tostring(mapId)
        print(msg)
        if _G.LogMsg then _G.LogMsg(msg) end
        isSuccess = true
    end
end

-- 2. Xử lý Quảng Trường Quỷ (Demon Plaza / Devil Square)
if not isSuccess and _G.Instance_DemonPlazaUI and _G.Instance_DemonPlazaUI.ContentData then
    local mapId = _G.Instance_DemonPlazaUI.ContentData.mapId
    if _G.NetManager and _G.InstanceMatchMessage then
        _G.NetManager.Send(_G.InstanceMatchMessage.ReqInstanceMatchCreateTeam, {
            instanceId = mapId
        })
        
        -- THỬ GỬI THÊM LỆNH ENTER VÀO MAP TRỰC TIẾP (BỎ QUA MATCHING TEAM)
        if _G.InstanceMessage and _G.InstanceMessage.ReqEnterInstance then
            _G.NetManager.Send(_G.InstanceMessage.ReqEnterInstance, {
                instanceId = mapId
            })
        end
        
        if _G.UIManager then _G.UIManager.Hide(_G.UIID.Instance_DemonPlazaUI) end
        
        local msg = "[TEST] Đã gửi ReqInstanceMatchCreateTeam & ReqEnterInstance QTQ với mapId: " .. tostring(mapId)
        print(msg)
        if _G.LogMsg then _G.LogMsg(msg) end
        isSuccess = true
    end
end

-- 3. Thông báo lỗi nếu chưa mở UI
if not isSuccess then
    local err = "[TEST] LỖI: Vui lòng mở giao diện Huyết Lâu hoặc Quảng Trường Quỷ trước khi chạy script!"
    print(err)
    if _G.LogMsg then _G.LogMsg(err) end
end
