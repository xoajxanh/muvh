-- =========================================================================
-- [TEST SCRIPT]: FIX LỖI RENDER 10FPS / BÓNG DÍNH VỊ TRÍ CŨ (SKINNING & SHADER FIX)
-- Mô tả:
-- 1. Ép SkinnedMeshRenderer.updateWhenOffscreen = true (Khắc phục trễ khung xương)
-- 2. Ép Animator.cullingMode = AlwaysAnimate (Không bỏ nhịp frame xương)
-- 3. Đặt Shader.globalMaximumLOD = 100 (Chế độ shader nhẹ không delay buffer)
-- 4. Khóa 60 FPS mượt mà
-- =========================================================================

local function log_msg(msg)
    local text = tostring(msg)
    pcall(function()
        if _G.FloatingWordUtility and _G.FloatingWordUtility.QuickMsg then
            _G.FloatingWordUtility.QuickMsg(text)
        end
    end)
    pcall(function()
        if _G.WriteLog then _G.WriteLog("[SM_FIX] " .. text) end
    end)
end

local ok, err = pcall(function()
    _G.Test_SkinFix_State = not _G.Test_SkinFix_State

    if _G.Test_SkinFix_State then
        -- 1. TỐC CHẠY 6.0X
        _G.RunSpeedMultiplier = 6.0
        if _G.RoleManager and _G.RoleManager.me and _G.RoleManager.me.SetMoveSpeed then
            local baseSpeed = _G.RoleManager.me._modBaseMoveSpeed or (_G.RoleManager.me.data and _G.RoleManager.me.data.moveSpeed) or 350
            _G.RoleManager.me:SetMoveSpeed(baseSpeed)
        end

        -- 2. KHÓA 60 FPS
        if CS.UnityEngine.Application then
            CS.UnityEngine.Application.targetFrameRate = 60
        end

        -- 3. ÉP SHADER NHẸ (LOD 100 - CHÍNH LÀ CHẾ ĐỘ CỦA GIẢM CẤU HÌNH)
        if CS.UnityEngine.Shader then
            CS.UnityEngine.Shader.globalMaximumLOD = 100
        end

        -- 4. ÉP UPDATE TOÀN BỘ SKINNED MESH & ANIMATOR KHÔNG CULLING
        local me = _G.RoleManager and _G.RoleManager.me
        if me then
            local targets = {}
            if me.gameObject then table.insert(targets, me.gameObject) end
            if me.model and me.model.modelObject then table.insert(targets, me.model.modelObject) end
            if me.model and me.model.transform then table.insert(targets, me.model.transform) end
            if me.AvatarEquip then
                if me.AvatarEquip.footPrintObj then table.insert(targets, me.AvatarEquip.footPrintObj) end
                if me.AvatarEquip.wingObj then table.insert(targets, me.AvatarEquip.wingObj) end
            end

            for _, targetGo in ipairs(targets) do
                if targetGo and not IsNil(targetGo) then
                    -- SkinnedMeshRenderer fix
                    local smrs = targetGo:GetComponentsInChildren(typeof(CS.UnityEngine.SkinnedMeshRenderer))
                    if smrs then
                        for i = 0, smrs.Length - 1 do
                            local s = smrs[i]
                            if s and not IsNil(s) then
                                s.updateWhenOffscreen = true
                            end
                        end
                    end

                    -- Animator Culling fix
                    local anims = targetGo:GetComponentsInChildren(typeof(CS.UnityEngine.Animator))
                    if anims then
                        for i = 0, anims.Length - 1 do
                            local a = anims[i]
                            if a and not IsNil(a) then
                                a.cullingMode = CS.UnityEngine.AnimatorCullingMode.AlwaysAnimate
                            end
                        end
                    end
                end
            end
        end

        log_msg("ĐÃ BẬT TỐC 6.0X + UPDATEWHENOFFSCREEN + SHADER LOD 100 + 60FPS!")
    else
        _G.RunSpeedMultiplier = 1.0
        if _G.RoleManager and _G.RoleManager.me and _G.RoleManager.me.SetMoveSpeed then
            local baseSpeed = _G.RoleManager.me._modBaseMoveSpeed or (_G.RoleManager.me.data and _G.RoleManager.me.data.moveSpeed) or 350
            _G.RoleManager.me:SetMoveSpeed(baseSpeed)
        end

        if CS.UnityEngine.Shader then
            CS.UnityEngine.Shader.globalMaximumLOD = 900
        end

        log_msg("ĐÃ TẮT TEST, KHÔI PHỤC NGUYÊN BẢN!")
    end
end)

if not ok then
    log_msg("LỖI TEST: " .. tostring(err))
end
