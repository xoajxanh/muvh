-- =========================================================================
-- [SCRIPT TEST]: KIỂM TRA TOÀN DIỆN COOLDOWN & BUFF HỒI SINH (ĐẠI THIÊN SỨ)
-- =========================================================================
local lines = {}
local function log(s)
    table.insert(lines, tostring(s))
end

log("=========================================================================")
log(">>> KIỂM TRA TRẠNG THÁI SKILL & COOLDOWN TRÊN NHÂN VẬT <<<")
log("Thời gian thực thi: " .. os.date("%Y/%m/%d %H:%M:%S"))
log("=========================================================================")

local me = _G.RoleManager and _G.RoleManager.me
if not me then
    log("[LỖI] Không tìm thấy RoleManager.me!")
else
    local meId = me.id or (me.data and me.data.id) or 0
    log(string.format("Nhân vật: ID=%s, Tên=%s, Career=%s, Dead=%s, HP=%s/%s",
        tostring(meId), tostring(me.name), tostring(me.career), tostring(me.isDead),
        tostring(me.hp or (me.data and me.data.hp)),
        tostring(me.maxHp or (me.data and me.data.maxHp))))

    -- 1. KIỂM TRA CÁC BIẾN MOD TOWER CHECK
    log("\n--- 1. CÁC BIẾN MOD CHECK SKILL THÁP ---")
    log("Mod_AutoChallengeTower_Enabled : " .. tostring(_G.Mod_AutoChallengeTower_Enabled))
    log("Mod_AutoTower_CheckSkill_Enabled: " .. tostring(_G.Mod_AutoTower_CheckSkill_Enabled))
    log("Mod_TowerCheck_HoiSinh_Enabled  : " .. tostring(_G.Mod_TowerCheck_HoiSinh_Enabled))
    log("Mod_TowerCheck_Limit_Enabled    : " .. tostring(_G.Mod_TowerCheck_Limit_Enabled))
    log("Mod_TowerCheck_AngelSpike_Enabled: " .. tostring(_G.Mod_TowerCheck_AngelSpike_Enabled))
    log("Mod_TowerCheck_NhiepHon_Enabled : " .. tostring(_G.Mod_TowerCheck_NhiepHon_Enabled))

    -- 2. DUMP TOÀN BỘ COOLDOWN TRONG me.cd
    log("\n--- 2. TOÀN BỘ DANH SÁCH COOLDOWN (me.cd) ---")
    local curMs = (_G.Time and _G.Time.GetServerTime and _G.Time.GetServerTime()) or 0
    local curSec = (_G.Time and _G.Time.GetServerSecondTime and _G.Time.GetServerSecondTime()) or os.time()
    log(string.format("ServerTime (ms): %s | ServerSecondTime (s): %s", tostring(curMs), tostring(curSec)))

    local cdCount = 0
    if me.cd then
        pcall(function()
            for k, v in pairs(me.cd) do
                cdCount = cdCount + 1
                local endTime = 0
                local cdType = type(v)
                if cdType == "table" or cdType == "userdata" then
                    endTime = v.endTime or 0
                elseif cdType == "number" then
                    endTime = v
                end
                
                local remSec = 0
                if endTime > 1000000000000 then
                    remSec = math.ceil((endTime - curMs) / 1000)
                elseif endTime > 0 then
                    remSec = math.ceil(endTime - curSec)
                end

                local skillName = "Unknown"
                pcall(function()
                    if _G.ClientTable and _G.ClientTable.cfg_Skill_skillManager then
                        local cfg = _G.ClientTable.cfg_Skill_skillManager:TryGetValue(k)
                        if cfg and cfg.name then skillName = cfg.name end
                    end
                end)

                log(string.format("  CD Key: %-10s | RemSec: %-4d | EndTime: %-14s | Name: %s",
                    tostring(k), remSec, tostring(endTime), tostring(skillName)))
            end
        end)
    end
    log("Tổng số CD đang có: " .. tostring(cdCount))

    -- Kiểm tra riêng các key liên quan đến 10200100
    log("\n--- 2.1. KIỂM TRA ĐẶC BIỆT CÁC KEY LIÊN QUAN HỒI SINH ---")
    local testKeys = { 10200100, 10200101, 10200102, 10200103, 10200104, 10200105, 10200106, 10200107, 10200110, 53010101 }
    for _, tk in ipairs(testKeys) do
        local cdVal = me.cd and me.cd[tk]
        if cdVal then
            local et = (type(cdVal) == "table" and cdVal.endTime) or (type(cdVal) == "number" and cdVal) or 0
            log(string.format("  [FOUND IN me.cd] Key %s => EndTime=%s", tostring(tk), tostring(et)))
        else
            log(string.format("  [NIL in me.cd] Key %s", tostring(tk)))
        end
    end

    -- 3. DUMP TOÀN BỘ BUFF ĐANG CÓ TRÊN NHÂN VẬT
    log("\n--- 3. DANH SÁCH BUFF HIỆN CÓ TRÊN BẢN THÂN (BuffDic) ---")
    local buffCount = 0
    if _G.BuffData and _G.BuffData.BuffDic and _G.BuffData.BuffDic[meId] then
        for idx, b in ipairs(_G.BuffData.BuffDic[meId]) do
            buffCount = buffCount + 1
            local bCId = b.buffCId or (b.buffConfig and b.buffConfig.id) or 0
            local bGrp = (b.buffConfig and b.buffConfig.buffGroup) or 0
            local bName = (b.buffConfig and b.buffConfig.name) or "Unknown"
            local bEnd = b.endTime or (b.buffConfig and b.buffConfig.totalTime) or 0
            log(string.format("  Buff [%d]: ID=%s | Group=%s | Name=%s | EndTime=%s",
                idx, tostring(bCId), tostring(bGrp), tostring(bName), tostring(bEnd)))
        end
    end
    log("Tổng số Buff đang có: " .. tostring(buffCount))

    -- 4. KIỂM TRA ViewData.meData
    log("\n--- 4. KIỂM TRA ViewData.meData ---")
    if _G.ViewData and _G.ViewData.meData then
        local md = _G.ViewData.meData
        if md.skills then
            local suitSkillCount = 0
            for skId, skVal in pairs(md.skills) do
                if tostring(skId):find("102001") then
                    suitSkillCount = suitSkillCount + 1
                    log(string.format("  ViewData skill: Key=%s, sid=%s", tostring(skId), tostring(skVal and skVal.sid)))
                end
            end
            log("Số skill 102001 trong ViewData.meData.skills: " .. tostring(suitSkillCount))
        end
        if md.cdMap then
            log("ViewData.meData.cdMap:")
            for ck, cv in pairs(md.cdMap) do
                log(string.format("  cdMap[%s] = %s", tostring(ck), tostring(cv)))
            end
        end
    end

    -- 5. KIỂM TRA SuitData & BỘ ĐỒ ĐANG MẶC
    log("\n--- 5. KIỂM TRA THÔNG TIN BỘ ĐỒ (SUIT DATA) ---")
    pcall(function()
        if _G.ViewData and _G.ViewData.meData and _G.ViewData.meData.suits then
            log("ViewData.meData.suits:")
            for sk, sv in pairs(_G.ViewData.meData.suits) do
                log(string.format("  Suit[%s] = %s", tostring(sk), tostring(sv)))
            end
        end
    end)

    -- 6. KIỂM TRA NÚT UI HỒI SINH TRÊN MÀN HÌNH
    log("\n--- 6. KIỂM TRA NÚT UI HỒI SINH TRÊN MÀN HÌNH ---")
    pcall(function()
        local tGo = CS.UnityEngine.GameObject.Find("Mod_TowerCheck_HoiSinh_Enabled_Toggle")
        if tGo then
            log("Tìm thấy GameObject: Mod_TowerCheck_HoiSinh_Enabled_Toggle (Active=" .. tostring(tGo.activeInHierarchy) .. ")")
            local txtComp = tGo:GetComponentInChildren(typeof(CS.UnityEngine.UI.Text))
            if txtComp then
                log("Text hiện tại trên nút: '" .. tostring(txtComp.text) .. "'")
            else
                log("Không tìm thấy component Text trong nút!")
            end
        else
            log("Không tìm thấy GameObject: Mod_TowerCheck_HoiSinh_Enabled_Toggle (Menu đang đóng hoặc chưa tạo)")
        end
    end)

    -- 7. THỬ GỌI TRỰC TIẾP Mod_UpdateTowerSkillButtons
    log("\n--- 7. THỬ CHẠY Mod_UpdateTowerSkillButtons() ---")
    pcall(function()
        if _G.Mod_UpdateTowerSkillButtons then
            _G.Mod_UpdateTowerSkillButtons()
            log("Đã gọi _G.Mod_UpdateTowerSkillButtons() thành công!")
            local tGo = CS.UnityEngine.GameObject.Find("Mod_TowerCheck_HoiSinh_Enabled_Toggle")
            if tGo then
                local txtComp = tGo:GetComponentInChildren(typeof(CS.UnityEngine.UI.Text))
                if txtComp then
                    log("Text sau khi gọi Update: '" .. tostring(txtComp.text) .. "'")
                end
            end
        else
            log("Chưa có hàm _G.Mod_UpdateTowerSkillButtons!")
        end
    end)
end

log("\n=========================================================================")
log("Hoàn tất kiểm tra!")

local outputContent = table.concat(lines, "\n")

-- Ghi ra output.txt
local outPath = ""
pcall(function()
    outPath = CS.UnityEngine.Application.persistentDataPath .. "/output.txt"
end)
if outPath == "" then
    outPath = "/storage/emulated/0/Android/data/com.vnyh.gp/files/output.txt"
end

local written = false
pcall(function()
    if CS.System and CS.System.IO and CS.System.IO.File and CS.System.Text and CS.System.Text.Encoding then
        CS.System.IO.File.WriteAllText(outPath, outputContent, CS.System.Text.Encoding.UTF8)
        written = true
    end
end)
if not written then
    pcall(function()
        local f = io.open(outPath, "w")
        if f then
            f:write(outputContent)
            f:close()
            written = true
        end
    end)
end

if _G.FloatingWordUtility and _G.FloatingWordUtility.QuickMsg then
    _G.FloatingWordUtility.QuickMsg(written and "[DUMP OK] Đã ghi kết quả kiểm tra vào output.txt!" or "[LỖI] Không thể ghi output.txt")
end
