-- =========================================================================
-- [SCRIPT TRIỆU HỒI KÉO SU VÀO QUẢNG TRƯỜNG QUỶ 15 (REQCALLFLAG CHUẨN XÁC)]
-- =========================================================================

local logLines = {}
local function Log(str)
    table.insert(logLines, tostring(str))
    print(tostring(str))
end

Log("=========================================================================")
Log("=== KÍCH HOẠT LỆNH BÀI TRIỆU HỒI VÀO QUẢNG TRƯỜNG QUỶ 15 ===")
Log("Thời gian: " .. os.date("%Y-%m-%d %H:%M:%S"))
Log("=========================================================================")

local meName = (ViewData and ViewData.meData and ViewData.meData.name) or "Unknown"
local myRid = (ViewData and ViewData.meData and ViewData.meData.id) or 0
local curMap = (SceneData and SceneData.mapId) or 0
local curLine = (SceneData and SceneData.line) or 1

-- THÔNG SỐ CHÍNH XÁC TỪ ĐỘI TRƯỞNG LIẾMHẾT TRONG QTQ 15:
local targetMapId = 1010015
local targetLine = 1
local targetX = 70
local targetY = 89
local targetLeaderRid = 5944970060856670296
local teamId = 5944970074295620621

Log("Nhân vật thực thi: " .. tostring(meName) .. " (RID: " .. tostring(myRid) .. ")")
Log("Map hiện tại:     " .. tostring(curMap) .. " (Line " .. tostring(curLine) .. ")")
Log("-------------------------------------------------------------------------")
Log("Mục tiêu đến:     Map " .. targetMapId .. " (Quảng Trường Quỷ 15 - Line " .. targetLine .. ")")
Log("Tọa độ Đích:      (" .. targetX .. ", " .. targetY .. ")")
Log("Đội trưởng RID:   " .. tostring(targetLeaderRid))
Log("Team ID:          " .. tostring(teamId))
Log("-------------------------------------------------------------------------")

pcall(function()
    Log(">>> ĐANG GỬI GÓI TIN MapMessage.ReqCallFlag... <<<")
    NetManager.Send(MapMessage.ReqCallFlag, {
        type = 2, -- Call_team (Tổ đội)
        rid = targetLeaderRid,
        mapId = targetMapId,
        line = targetLine,
        x = targetX,
        y = targetY,
        hostId = 0
    })
    Log("  [+] ReqCallFlag Sent OK!")
end)

if FloatingWordUtility and FloatingWordUtility.QuickMsg then
    FloatingWordUtility.QuickMsg(string.format("[REQCALLFLAG] ĐANG BAY VÀO QUẢNG TRƯỜNG QUỶ 15 (%d, %d)!", targetX, targetY))
end

-- Ghi kết quả ra file output.txt trên Android
local resultText = table.concat(logLines, "\n")
pcall(function()
    local outPath = "/storage/emulated/0/Android/data/com.vnyh.gp/files/output.txt"
    local f = io.open(outPath, "w")
    if f then
        f:write(resultText)
        f:close()
    end
end)

return resultText
