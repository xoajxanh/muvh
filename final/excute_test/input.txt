-- =========================================================================
-- [DEBUG SCRIPT]: KIỂM TRA ĐIỀU KIỆN & THỬ NGHIỆM TỰ DÙNG CỰC HẠN + THÁNH HỒN GAI
-- Mục đích:
--   1. Quét CD Cực Hạn (410700), Thánh Hồn Thiên Sứ - Gai (34011800), Buff Elf & Map
--   2. Tìm ID chiêu thức chính xác của nhân vật
--   3. Thử nghiệm thi triển ngay 2 chiêu này để kiểm tra nhân vật có dùng chiêu thành công không
-- Ghi kết quả ra: output.txt (tại persistentDataPath để push_test.ps1 kéo về)
-- =========================================================================

pcall(function()
    local logLines = {}
    local function AddLog(line)
        table.insert(logLines, line)
        if CS.UnityEngine.Debug and CS.UnityEngine.Debug.Log then
            CS.UnityEngine.Debug.Log("[DETECT_DEBUG] " .. tostring(line))
        end
    end

    AddLog("==================================================================")
    AddLog("  DETECT & TEST CAST: CỰC HẠN & THÁNH HỒN THIÊN SỨ - GAI         ")
    AddLog("                 Thời gian: " .. os.date("%Y-%m-%d %H:%M:%S"))
    AddLog("==================================================================")

    local me = _G.RoleManager and _G.RoleManager.me
    if not me then
        AddLog("[LỖI] Không tìm thấy RoleManager.me (Chưa vào game hoặc nhân vật chưa sẵn sàng)!")
        return
    end

    local meId = me.id or (me.data and me.data.id) or 0
    local serverTime = (_G.Time and _G.Time.GetServerTime and _G.Time.GetServerTime()) or 0

    -- =========================================================================
    -- 1. BẢN ĐỒ HIỆN TẠI
    -- =========================================================================
    local mapId = (_G.SceneData and _G.SceneData.mapId) or 0
    local groupId = (_G.SceneData and _G.SceneData.groupId) or 0
    local inCopy = (_G.TranScriptData and _G.TranScriptData.InTranscript) or false
    local isLorencia = (mapId == 1001 or groupId == 1001) and not inCopy
    local isTowerMap = string.match(tostring(mapId), "^1059") or string.match(tostring(groupId), "^1059")

    AddLog(string.format("1. BẢN ĐỒ: mapId=%s | groupId=%s | TrongPB=%s | LàTháp=%s",
        tostring(mapId), tostring(groupId), tostring(inCopy), tostring(isTowerMap ~= nil)))

    -- =========================================================================
    -- 2. HÀM TÌM SKILL ID THEO GROUP ID
    -- =========================================================================
    local function FindSkillId(skillGroupId)
        local meSkills = (me and me.skills) or (_G.ViewData and _G.ViewData.meData and _G.ViewData.meData.skills)
        if meSkills and meSkills[skillGroupId] then
            local sk = meSkills[skillGroupId]
            local sid = (type(sk) == "table" and (sk.sid or sk.id)) or (type(sk) == "number" and sk)
            if sid then return sid end
        end
        if _G.ViewData and _G.ViewData.meData and _G.ViewData.meData.allSkills then
            for _, sk in pairs(_G.ViewData.meData.allSkills) do
                local sid = (type(sk) == "table" and (sk.sid or sk.id)) or (type(sk) == "number" and sk)
                if sid then
                    local cfg = _G.ClientTable and _G.ClientTable.cfg_Skill_skillManager and _G.ClientTable.cfg_Skill_skillManager:TryGetValue(sid)
                    if cfg and cfg.groupId == skillGroupId then
                        return sid
                    end
                end
            end
        end
        return nil
    end

    -- =========================================================================
    -- 3. HÀM THI TRIỂN SKILL (CAST SKILL)
    -- =========================================================================
    local function CastSkill(sId, sGroupId)
        local myCell = me.serverCoord or (me.cellPos and { x = me.cellPos.x, y = me.cellPos.y }) or { x = 0, y = 0 }
        local myId = (me.data and me.data.id) or me.id or 0

        -- 1. QiJiHelper SetPressSkill
        pcall(function()
            if _G.QiJiHelperData and _G.QiJiHelperData.SetPressSkill then
                _G.QiJiHelperData.SetPressSkill(sId)
            end
        end)

        -- 2. Gửi gói tin Server (ReqPlayerUseSkill & ReqBroadcastUseSkill)
        pcall(function()
            if _G.NetManager and _G.NetManager.Send and _G.FightMessage then
                if _G.FightMessage.ReqPlayerUseSkill then
                    _G.NetManager.Send(_G.FightMessage.ReqPlayerUseSkill, {
                        skillId = sId,
                        targetId = myId,
                        x = myCell.x or 0,
                        y = myCell.y or 0,
                        position = 0
                    })
                end
                if _G.FightMessage.ReqBroadcastUseSkill then
                    _G.NetManager.Send(_G.FightMessage.ReqBroadcastUseSkill, {
                        skillId = sId,
                        targetId = myId,
                        x = myCell.x or 0,
                        y = myCell.y or 0,
                        position = 0
                    })
                end
            end
        end)

        -- 3. SkillMgr client
        pcall(function()
            if _G.SkillMgr then
                if _G.SkillMgr.RequestSkillToMe then
                    _G.SkillMgr.RequestSkillToMe(sId)
                elseif _G.SkillMgr.RequestSkillTest then
                    _G.SkillMgr.RequestSkillTest(sId)
                end
            end
        end)

        -- 4. PerformClientSkill & Update CD UI
        pcall(function()
            local attackSpeed = (_G.ViewData and _G.ViewData.meData and _G.ViewData.meData.GetAttribute and _G.ViewData.meData:GetAttribute(_G.EAttributeType.attackSpeedCalculateValue)) or 1000
            local skill_struct = _G.SkillUtility and _G.SkillUtility.ConstructSkillFromClientData and _G.SkillUtility.ConstructSkillFromClientData(sId, myId, myCell, myId, myCell, 0, attackSpeed)
            if skill_struct and _G.SkillController and _G.SkillController.PerformClientSkill then
                _G.SkillController.PerformClientSkill(skill_struct)
            end
            if _G.MeController and _G.MeController.UpdateClientSkillCd then
                _G.MeController.UpdateClientSkillCd(sId)
            end
        end)
    end

    -- =========================================================================
    -- 4. KIỂM TRA VÀ TEST DÙNG SKILL CỰC HẠN (410700)
    -- =========================================================================
    local limitGroup = 410700
    local cdTable = me.cd or {}
    local limitCd = cdTable[limitGroup]
    local isLimitReady = (not limitCd) or (not limitCd.endTime) or (limitCd.endTime <= serverTime)
    local limitSid = FindSkillId(limitGroup) or 41070001
    local limitCdRemain = (limitCd and limitCd.endTime and limitCd.endTime > serverTime) and math.max(0, math.floor((limitCd.endTime - serverTime) / 1000)) or 0

    AddLog(string.format("2. SKILL CỰC HẠN: Group=%d | SkillID=%s | Ready=%s | CD còn=%ds",
        limitGroup, tostring(limitSid), tostring(isLimitReady), limitCdRemain))

    -- =========================================================================
    -- 5. KIỂM TRA VÀ TEST DÙNG THÁNH HỒN THIÊN SỨ - GAI (34011800)
    -- =========================================================================
    local angelGroup = 34011800
    local angelCd = cdTable[angelGroup] or cdTable[34011801]
    local isAngelReady = (not angelCd) or (not angelCd.endTime) or (angelCd.endTime <= serverTime)
    local angelSid = FindSkillId(angelGroup) or 34011801
    local angelCdRemain = (angelCd and angelCd.endTime and angelCd.endTime > serverTime) and math.max(0, math.floor((angelCd.endTime - serverTime) / 1000)) or 0

    AddLog(string.format("3. THÁNH HỒN THIÊN SỨ - GAI: Group=%d | SkillID=%s | Ready=%s | CD còn=%ds",
        angelGroup, tostring(angelSid), tostring(isAngelReady), angelCdRemain))

    -- =========================================================================
    -- 6. THỬ NGHIỆM THI TRIỂN NGAY LẬP TỨC (NẾU SẴN SÀNG)
    -- =========================================================================
    AddLog("4. THỰC HIỆN TEST CAST SKILL:")
    local castLog = {}
    if isLimitReady then
        CastSkill(limitSid, limitGroup)
        table.insert(castLog, "Đã cast Cực Hạn (" .. tostring(limitSid) .. ")")
        AddLog("   [OK] Đã kích hoạt thi triển Cực Hạn (ID: " .. tostring(limitSid) .. ")")
    else
        table.insert(castLog, "Cực Hạn đang hồi (" .. tostring(limitCdRemain) .. "s)")
        AddLog("   [BỎ QUA] Cực Hạn đang trong thời gian hồi chiêu!")
    end

    if isAngelReady then
        CastSkill(angelSid, angelGroup)
        table.insert(castLog, "Đã cast Thánh Hồn Gai (" .. tostring(angelSid) .. ")")
        AddLog("   [OK] Đã kích hoạt thi triển Thánh Hồn Thiên Sứ - Gai (ID: " .. tostring(angelSid) .. ")")
    else
        table.insert(castLog, "Thánh Hồn Gai đang hồi (" .. tostring(angelCdRemain) .. "s)")
        AddLog("   [BỎ QUA] Thánh Hồn Thiên Sứ - Gai đang trong thời gian hồi chiêu!")
    end

    AddLog("==================================================================")
    AddLog(">>> HOÀN THÀNH TEST! Xem nhân vật trong game có vừa dùng chiêu không.")
    AddLog("==================================================================")

    -- Ghi ra output.txt
    pcall(function()
        local fullContent = table.concat(logLines, "\n")
        local outPaths = {
            (CS.UnityEngine.Application and CS.UnityEngine.Application.persistentDataPath) and (CS.UnityEngine.Application.persistentDataPath .. "/output.txt"),
            "/storage/emulated/0/Android/data/com.vnyh.gp/files/output.txt"
        }
        for _, p in ipairs(outPaths) do
            if p and p ~= "" then
                pcall(function()
                    if CS.System.IO.File and CS.System.IO.File.WriteAllText then
                        CS.System.IO.File.WriteAllText(p, fullContent)
                    else
                        local f = io.open(p, "w")
                        if f then f:write(fullContent); f:close() end
                    end
                end)
            end
        end
    end)

    local quickMsg = table.concat(castLog, " | ")
    if _G.FloatingWordUtility and _G.FloatingWordUtility.QuickMsg then
        _G.FloatingWordUtility.QuickMsg(quickMsg)
    end
end)
