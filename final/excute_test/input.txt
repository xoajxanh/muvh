-- =================================================================
-- SCRIPT TEST GẮP HỒI SINH TỰ ĐỘNG (Dùng cho nút Execute)
-- File: input.txt
-- =================================================================

pcall(function()
    local logMsg = ""
    local function Log(msg)
        logMsg = logMsg .. msg .. "\n"
        print("[TestResurgence] " .. msg)
        if _G.WriteLog then _G.WriteLog("[TestResurgence] " .. msg) end
        if _G.FloatingWordUtility then _G.FloatingWordUtility.QuickMsg(msg) end
    end

    Log("=== BẮT ĐẦU TEST HỒI SINH TỰ ĐỘNG ===")

    local isVisible = false
    if _G.UIManager and _G.UIManager.IsVisible and _G.UIID and _G.UIID.Role_ResurgenceUI then
        isVisible = _G.UIManager.IsVisible(_G.UIID.Role_ResurgenceUI)
    end

    Log("Trạng thái UI Role_ResurgenceUI IsVisible: " .. tostring(isVisible))

    local uiObj = nil
    if _G.UIManager and _G.UIManager.GetUiByName then
        uiObj = _G.UIManager.GetUiByName("Role_ResurgenceUI")
    end

    Log("Đối tượng UI Role_ResurgenceUI: " .. (uiObj and "ĐÃ TÌM THẤY" or "NIL"))

    if uiObj then
        Log("Thử kích hoạt uiObj:Button_1OnClick()...")
        local success, err = pcall(function()
            uiObj:Button_1OnClick()
        end)
        if success then
            Log("✅ ĐÃ GỌI Button_1OnClick THÀNH CÔNG!")
        else
            Log("❌ LỖI KHI GỌI Button_1OnClick: " .. tostring(err))
        end
    else
        Log("Bảng Hồi sinh chưa hiển thị hoặc không lấy được UI object. Thử ép gửi gói tin trực tiếp...")
        local mapId = 0
        if _G.SceneData and _G.SceneData.mapId then
            mapId = _G.SceneData.mapId
        elseif _G.RoleManager and _G.RoleManager.me and _G.RoleManager.me.mapId then
            mapId = _G.RoleManager.me.mapId
        end
        Log("Map ID hiện tại: " .. tostring(mapId))

        if _G.MeController and _G.RoleReliveType then
            if mapId == 1077 or (_G.TranScriptData and _G.TranScriptData.IsInRefineKSBattle and _G.TranScriptData.IsInRefineKSBattle()) then
                Log("⚡ Ép phát gói tin MeController.ReqReqRelive(RoleReliveType.KSBattle)...")
                _G.MeController.ReqReqRelive(_G.RoleReliveType.KSBattle)
            else
                Log("⚡ Ép phát gói tin MeController.ReqReqRelive(RoleReliveType.BornPoint)...")
                _G.MeController.ReqReqRelive(_G.RoleReliveType.BornPoint)
            end
        end

        if _G.UIManager and _G.UIManager.Hide and _G.UIID and _G.UIID.Role_ResurgenceUI then
            _G.UIManager.Hide(_G.UIID.Role_ResurgenceUI)
        end
    end

    -- Ghi kết quả test ra output.txt
    pcall(function()
        local f = io.open("/storage/emulated/0/Android/data/com.vnyh.gp/files/output.txt", "w")
        if f then
            f:write(logMsg)
            f:close()
        end
    end)
end)
