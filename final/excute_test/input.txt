-- =========================================================================
-- [SCRIPT]: TRÍCH XUẤT DANH SÁCH 50 NGƯỜI CHƠI BẢNG XẾP HẠNG THỰC LỰC SERVER
-- Mô tả: Lấy danh sách Top 50 BXH Thực Lực (Tổng Hợp) và xuất ra output.txt
-- =========================================================================

local function FormatCareer(careerId)
    local basic = 0
    if _G.RoleUtility and _G.RoleUtility.GetBasicCareer then
        basic = _G.RoleUtility.GetBasicCareer(careerId or 0)
    else
        basic = (careerId or 0) % 10 + 10
    end
    if basic == 11 then return "Kiếm Sĩ"
    elseif basic == 12 then return "Ma Pháp Sư"
    elseif basic == 13 then return "Cung Thủ"
    elseif basic == 14 then return "Ma Kiếm Sĩ"
    elseif basic == 15 then return "Thánh Đạo Sư"
    elseif basic == 16 then return "Triệu Hồi Sư"
    else return "Khác (" .. tostring(careerId) .. ")"
    end
end

local function WriteOutputFile(content)
    local path = ""
    pcall(function()
        path = CS.UnityEngine.Application.persistentDataPath .. "/output.txt"
    end)
    if path == "" then
        path = "/storage/emulated/0/Android/data/com.vnyh.gp/files/output.txt"
    end

    local success = false
    -- Cách 1: Ghi qua C# File UTF-8
    pcall(function()
        if CS.System and CS.System.IO and CS.System.IO.File and CS.System.Text and CS.System.Text.Encoding then
            CS.System.IO.File.WriteAllText(path, content, CS.System.Text.Encoding.UTF8)
            success = true
        end
    end)

    -- Cách 2: Ghi qua io.open chuẩn Lua
    if not success then
        pcall(function()
            local f = io.open(path, "w")
            if f then
                f:write(content)
                f:close()
                success = true
            end
        end)
    end
    return success, path
end

local function ProcessRankList(ranks)
    if not ranks or #ranks == 0 then
        -- Thử duyệt nếu ranks là map key number
        local temp = {}
        for k, v in pairs(ranks) do
            if type(k) == "number" and type(v) == "table" then
                table.insert(temp, v)
            end
        end
        ranks = temp
    end

    local total = #ranks
    local limit = math.min(total, 50)
    if limit == 0 then
        local msg = "Không tìm thấy dữ liệu xếp hạng nào trong bộ nhớ!"
        WriteOutputFile(msg)
        if _G.FloatingWordUtility and _G.FloatingWordUtility.QuickMsg then
            _G.FloatingWordUtility.QuickMsg(msg)
        end
        return
    end

    local out = {}
    local function addLine(fmt, ...)
        local str = select("#", ...) > 0 and string.format(fmt, ...) or tostring(fmt)
        table.insert(out, str)
    end

    addLine("=========================================================================")
    addLine(">>> DANH SÁCH TOP %d BẢNG XẾP HẠNG THỰC LỰC (TỔNG HỢP) TẠI SERVER <<<", limit)
    addLine("Thời gian trích xuất: %s", os.date and os.date("%Y/%m/%d %H:%M:%S") or "now")
    addLine("Tổng số người trong BXH: %d", total)
    addLine("=========================================================================\n")

    -- 1. Chuỗi copy nhanh 1 dòng (cách nhau bằng dấu chấm phẩy ;)
    local nameList = {}
    for i = 1, limit do
        local item = ranks[i]
        local name = item.name or item.roleName or ("Player_" .. tostring(i))
        table.insert(nameList, name)
    end
    local fastCopyStr = table.concat(nameList, ";")

    addLine("-------------------------------------------------------------------------")
    addLine("1. CHUỖI TÊN COPY NHANH (Dùng paste thẳng vào ô Khóa Mục Tiêu / Mod):")
    addLine("-------------------------------------------------------------------------")
    addLine("%s\n", fastCopyStr)

    -- 2. Danh sách tên theo từng dòng
    addLine("-------------------------------------------------------------------------")
    addLine("2. DANH SÁCH TÊN RIÊNG BIỆT (Dễ copy từng người):")
    addLine("-------------------------------------------------------------------------")
    for i = 1, limit do
        local item = ranks[i]
        local name = item.name or item.roleName or ""
        addLine("%2d. %s", i, name)
    end
    addLine("")

    -- 3. Bảng chi tiết đầy đủ thông tin
    addLine("-------------------------------------------------------------------------")
    addLine("3. BẢNG CHI TIẾT THÔNG TIN TOP %d:", limit)
    addLine("-------------------------------------------------------------------------")
    addLine(string.format("%-4s | %-24s | %-12s | %-12s | %-6s | %-18s | %s", "Hạng", "Tên Nhân Vật", "Phái", "Lực Chiến", "Cấp", "Guild", "RoleID"))
    addLine(string.rep("-", 95))

    for i = 1, limit do
        local item = ranks[i]
        local name = item.name or item.roleName or ""
        local careerStr = FormatCareer(item.career)
        local fight = item.fightValue or 0
        local level = item.level or 0
        local guild = (item.unionName and item.unionName ~= "") and item.unionName or "(Trống)"
        local lid = tostring(item.lid or item.id or 0)

        addLine(string.format("%-4d | %-24s | %-12s | %-12s | %-6s | %-18s | %s", i, name, careerStr, tostring(fight), tostring(level), guild, lid))
    end

    addLine(string.rep("-", 95))
    addLine("\n=========================================================================")
    addLine(">>> KẾT THÚC DANH SÁCH TOP %d! <<<", limit)
    addLine("=========================================================================")

    local fullContent = table.concat(out, "\n")
    local ok, filePath = WriteOutputFile(fullContent)

    -- Lưu vào biến toàn cục để tiện dùng trong game
    _G.Mod_LastRankTop50 = nameList
    _G.Mod_LastRankTop50Str = fastCopyStr
    _G.Mod_LastRankTopData = ranks

    local tipMsg = string.format("Đã lấy thành công danh sách %d người chơi BXH vào output.txt!", limit)
    if _G.FloatingWordUtility and _G.FloatingWordUtility.QuickMsg then
        _G.FloatingWordUtility.QuickMsg(tipMsg)
    elseif _G.CommonTipManage and _G.CommonTipManage.ShowTips then
        _G.CommonTipManage.ShowTips(tipMsg)
    end
end

-- =========================================================================
-- LOGIC TÌM VÀ TRUY VẤN DỮ LIỆU BXH THỰC LỰC
-- =========================================================================
local foundRanks = nil

-- Cách 1: Lấy trực tiếp từ UI nếu đang mở
pcall(function()
    local ui = _G.UIManager and _G.UIManager.GetUiByName and _G.UIManager.GetUiByName("Rank_ListInfoUI")
    if ui and ui.GetRankData then
        local d = ui:GetRankData()
        if d and #d > 0 then
            foundRanks = d
        end
    end
end)

-- Cách 2: Lấy từ RankListData.RankListAll[RANKTYPE.fight]
if not foundRanks or #foundRanks == 0 then
    pcall(function()
        if _G.RankListData and _G.RankListData.RankListAll then
            local fightType = (_G.RANKTYPE and _G.RANKTYPE.fight) or 2
            local fightTable = _G.RankListData.RankListAll[fightType]
            if fightTable then
                -- Ưu tiên career All (-1 hoặc 0)
                if fightTable[-1] and #fightTable[-1] > 0 then
                    foundRanks = fightTable[-1]
                elseif fightTable[0] and #fightTable[0] > 0 then
                    foundRanks = fightTable[0]
                else
                    -- Duyệt tất cả các key để tìm bảng nhiều phần tử nhất
                    local maxLen = 0
                    local best = nil
                    for k, v in pairs(fightTable) do
                        if type(v) == "table" and #v > maxLen then
                            maxLen = #v
                            best = v
                        end
                    end
                    if best then foundRanks = best end
                end
            end
        end
    end)
end

-- Nếu đã có dữ liệu trong bộ nhớ -> Xuất ngay lập tức!
if foundRanks and #foundRanks > 0 then
    ProcessRankList(foundRanks)
else
    -- Nếu chưa có dữ liệu (chưa mở BXH bao giờ) -> Gửi yêu cầu lên Server và đợi phản hồi
    if _G.FloatingWordUtility and _G.FloatingWordUtility.QuickMsg then
        _G.FloatingWordUtility.QuickMsg("Đang gửi yêu cầu lấy dữ liệu BXH Thực Lực từ Server...")
    end

    pcall(function()
        local fightType = (_G.RANKTYPE and _G.RANKTYPE.fight) or 2
        local careerAll = (_G.ERoleCareer and _G.ERoleCareer.All) or -1

        if _G.networkRequest and _G.networkRequest.ReqQueryRanks then
            _G.networkRequest.ReqQueryRanks(fightType, careerAll)
        elseif _G.NetManager and _G.RankMessage and _G.RankMessage.ReqQueryRanks then
            _G.NetManager.Send(_G.RankMessage.ReqQueryRanks, { rankType = fightType, career = careerAll })
        end
    end)

    -- Đợi 0.8 giây cho Server trả gói tin về
    local timerObj = _G.Timer or Timer
    if timerObj and timerObj.Start then
        timerObj.Start(0.8, function()
            local delayedRanks = nil
            pcall(function()
                local fightType = (_G.RANKTYPE and _G.RANKTYPE.fight) or 2
                if _G.RankListData and _G.RankListData.RankListAll and _G.RankListData.RankListAll[fightType] then
                    local fTable = _G.RankListData.RankListAll[fightType]
                    delayedRanks = fTable[-1] or fTable[0]
                    if not delayedRanks or #delayedRanks == 0 then
                        for k, v in pairs(fTable) do
                            if type(v) == "table" and #v > 0 then
                                delayedRanks = v
                                break
                            end
                        end
                    end
                end
            end)

            if delayedRanks and #delayedRanks > 0 then
                ProcessRankList(delayedRanks)
            else
                WriteOutputFile("Chưa nhận được phản hồi dữ liệu BXH từ Server. Vui lòng mở bảng BXH trong game lên một lần rồi chạy lại script!")
                if _G.FloatingWordUtility and _G.FloatingWordUtility.QuickMsg then
                    _G.FloatingWordUtility.QuickMsg("Chưa có dữ liệu BXH. Hãy mở bảng BXH trong game rồi chạy lại!")
                end
            end
        end)
    else
        WriteOutputFile("Đã gửi yêu cầu tới server. Hãy mở bảng BXH và chạy lại script để lấy danh sách.")
    end
end
