-- ========================================================
-- SCRIPT TEST DI CHUYỂN TRONG MAP VÀ PHÓ BẢN
-- ========================================================
local me = _G.RoleManager and _G.RoleManager.me
if not me then
    print("[TEST_MOVE] Lỗi: Không tìm thấy nhân vật (RoleManager.me)")
    return
end

local curX = me.serverCoord and me.serverCoord.x or (me.cellPos and me.cellPos.x) or 0
local curY = me.serverCoord and me.serverCoord.y or (me.cellPos and me.cellPos.y) or 0

print(string.format("[TEST_MOVE] Tọa độ hiện tại: X=%d, Y=%d", curX, curY))

-- Thay đổi tọa độ targetX, targetY theo ý bạn muốn test (mặc định lấy tọa độ hiện tại + 10 ô)
local targetX = curX + 10
local targetY = curY + 10
local coordStr = string.format("%d#%d", targetX, targetY)

print(string.format("[TEST_MOVE] Thử di chuyển tới mục tiêu: X=%d, Y=%d ...", targetX, targetY))

-- PHƯƠNG ÁN 1: me:MoveTo (Trực tiếp của nhân vật)
local status1, err1 = pcall(function()
    if me.MoveTo then
        me:MoveTo({ x = targetX, y = targetY }, 0)
        print("[TEST_MOVE] -> Đã gọi me:MoveTo thành công!")
    end
end)
if not status1 then print("[TEST_MOVE] Lỗi me:MoveTo: " .. tostring(err1)) end

-- PHƯƠNG ÁN 2: JumpMapToMoveToPos (Dùng SceneData.groupId)
local status2, err2 = pcall(function()
    if _G.PathFinderManager and _G.PathFinderManager.JumpMapToMoveToPos and _G.SceneData and _G.SceneData.groupId then
        local targetPosData = (_G.PathFinderManager.GetCalcPosData and _G.PathFinderManager.GetCalcPosData(coordStr)) or
            (_G.Vector2 and _G.Vector2(targetX, targetY)) or { x = targetX, y = targetY }
            
        print("[TEST_MOVE] Map GroupID = " .. tostring(_G.SceneData.groupId))
        _G.PathFinderManager.JumpMapToMoveToPos(_G.SceneData.groupId, targetPosData, nil, nil, nil, 0, nil, 1, true)
        print("[TEST_MOVE] -> Đã gọi JumpMapToMoveToPos thành công!")
    end
end)
if not status2 then print("[TEST_MOVE] Lỗi JumpMapToMoveToPos: " .. tostring(err2)) end
