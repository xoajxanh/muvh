local function LogMsg(msg)
    local t = os.date("%H:%M:%S")
    _G.Mod_DebugMsg("[" .. t .. "] " .. msg)
end

if not _G.QiJiHelperController then return end

local skills = _G.ViewData.meData.skills or {}
for _, v in pairs(skills) do
    local skillData = _G.ClientTable.cfg_Skill_skillManager:TryGetValue(v.sid)
    if skillData and skillData.autoSkillType == 8 then
        local selfSelSkill = _G.QiJiHelperData.GetSelfSelSkill(tostring(skillData.groupId))
        local isOpen = selfSelSkill and selfSelSkill.isOpen or false
        LogMsg("Skill: " .. v.sid .. " | isOpen = " .. tostring(isOpen))
        
        _G.QiJiHelperController.SetSelfSelSkill(v.sid, not isOpen)
        
        local afterOpen = _G.QiJiHelperData.GetSelfSelSkill(tostring(skillData.groupId)).isOpen
        LogMsg("--> After Set: " .. tostring(afterOpen))
    end
end

_G.QiJiHelperController.Save()
if _G.UIManager and _G.UIManager.GetUiByName then
    local ui = _G.UIManager.GetUiByName("Preference_QiJiHelperUI")
    if ui and ui.Refresh then
        ui:Refresh()
        LogMsg("Đã Refresh UI nếu đang mở!")
    end
end
LogMsg("Done!")
