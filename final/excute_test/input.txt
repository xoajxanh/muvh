-- Mô tả:
--   1. Fix camera: Tự động gắn MainCamera vào RoleManager.me ngay khi vào map Tháp.
--   2. Hook NetManager.Send: Bắt BẤT KỲ gói ReqRelive nào (kể cả từ EmmyluaDebug auto)
--      và ÉP GHI ĐÈ sang mã reliveType = 7 (RefineTower) trước khi gửi lên Server!
--   3. Ghi nhận toàn bộ phản hồi từ Server (ResRelive, ResChangeMap, PromptMsg).

local logPath = "/storage/emulated/0/Android/data/com.vnyh.gp/files/output.txt"
local TEST_TYPE = 7
local TEST_NAME = "RefineTower (Mã 7)"

local function FileLog(msg)
    local text = os.date("[%H:%M:%S] ") .. tostring(msg)
    pcall(function()
        local f = io.open(logPath, "a")
        if f then
            f:write(text .. "\n")
            f:close()
        end
    end)
    print("[TOWER_TEST]: " .. tostring(msg))
end

local function QuickMsg(msg)
    local text = tostring(msg)
    FileLog(text)
    if _G.FloatingWordUtility and _G.FloatingWordUtility.QuickMsg then
        pcall(function() _G.FloatingWordUtility.QuickMsg("[THÁP]: " .. text) end)
    end
end

-- 1. Khởi tạo log
pcall(function()
    local f = io.open(logPath, "w")
    if f then
        f:write("=== TEST HƯỚNG 1: ÉP GỬI MÃ " .. TEST_NAME .. " ===\n")
        f:close()
    end
end)

QuickMsg("Script v7 đã kích hoạt! Đang theo dõi & ép mã 7...")

local function IsClimbTower()
    local mapId = _G.SceneData and (_G.SceneData.groupId or _G.SceneData.mapId)
    if mapId and string.match(tostring(mapId), "^1059") then
        return true, mapId
    end
    if _G.TranScriptData and _G.TranScriptData.IsInPersonBossBattle and _G.TranScriptData.IsInPersonBossBattle() then
        return true, mapId
    end
    return false, mapId
end

-- 2. Hàm gắn lại Camera vào nhân vật
local function AttachCameraToMe()
    pcall(function()
        local me = _G.RoleManager and _G.RoleManager.me
        if me and _G.MainCamera and _G.MainCamera.AttachRole then
            _G.MainCamera.AttachRole(me)
            FileLog("Đã gắn MainCamera vào RoleManager.me")
        end
    end)
end

-- Gắn ngay lúc chạy script
AttachCameraToMe()

-- 3. Hook Me.RoleDead để triệt tiêu delay 3s trong Tháp
if _G.Me then
    if not _G.Mod_Orig_RoleDead then
        _G.Mod_Orig_RoleDead = _G.Me.RoleDead
    end
    _G.Me.RoleDead = function(self)
        local inTower, mapId = IsClimbTower()
        FileLog("[EVENT] Me:RoleDead gọi! inTower=" .. tostring(inTower) .. " mapId=" .. tostring(mapId))
        if inTower then
            if _G.PlayerSoundManager then pcall(_G.PlayerSoundManager.PlayDieSound, self.data.career) end
            self:StopAutoFight()
            if _G.Player and _G.Player.RoleDead then _G.Player.RoleDead(self) end
            if _G.EventManager and _G.Event then
                _G.EventManager.Dispatch(_G.Event.Me_Dead)
                _G.EventManager.Dispatch(_G.Event.CanNotCollectBox)
            end
            FileLog("-> Đã triệt tiêu Timer 3s của Role_ResurgenceUI trong Tháp!")
            return
        end
        return _G.Mod_Orig_RoleDead(self)
    end
end

-- 4. Hook NetManager.Send: BẮT VÀ GHI ĐÈ BẤT KỲ GÓI ReqRelive NÀO SANG MÃ 7
if _G.NetManager then
    if not _G.Mod_Orig_NetSend then
        _G.Mod_Orig_NetSend = _G.NetManager.Send
    end
    _G.NetManager.Send = function(id, msg)
        local inTower, mapId = IsClimbTower()
        local msgName = (_G.MessageIDToName and _G.MessageIDToName[id]) or tostring(id)

        -- Bắt gói tin hồi sinh
        local isReqRelive = (id == 101004) or (msgName == "FightPackage.ReqRelive")
        if isReqRelive and msg then
            local oldType = msg.reliveType
            if inTower then
                msg.reliveType = TEST_TYPE -- ÉP CHUYỂN SANG MÃ 7!
                QuickMsg("-> ĐÃ GHI ĐÈ ReqRelive từ mã " .. tostring(oldType) .. " sang MÃ " .. tostring(TEST_TYPE) .. " (" .. TEST_NAME .. ")!")
                FileLog("[SEND_RELIVE_OVERRIDE] Ghi đè mã từ " .. tostring(oldType) .. " -> " .. tostring(TEST_TYPE))
            else
                FileLog("[SEND_RELIVE] ReqRelive bình thường: reliveType=" .. tostring(oldType))
            end
        elseif msgName ~= "UserPackage.ReqHeart" and msgName ~= "MapPackage.ReqMove" and msgName ~= "MapPackage.ReqChangeDir" then
            FileLog("[SEND] " .. msgName .. " (ID: " .. tostring(id) .. ")")
        end

        return _G.Mod_Orig_NetSend(id, msg)
    end
    FileLog("Đã hook NetManager.Send (Chặn ghi đè mã hồi sinh)")
end

-- 5. Hook NetManager.Dispatch để theo dõi phản hồi và tự động gắn Camera khi đổi map
if _G.NetManager then
    if not _G.Mod_Orig_NetDispatch then
        _G.Mod_Orig_NetDispatch = _G.NetManager.Dispatch
    end
    _G.NetManager.Dispatch = function(id, msg)
        local inTower, mapId = IsClimbTower()
        local msgName = (_G.MessageIDToName and _G.MessageIDToName[id]) or tostring(id)

        -- Tự gắn lại camera khi đổi map
        if id == 100307 or id == 100302 or msgName == "MapPackage.ResChangeMap" or msgName == "MapPackage.ResLoginMap" then
            if _G.Timer and _G.Timer.Start then
                _G.Timer.Start(0.5, function()
                    AttachCameraToMe()
                end)
            end
        end

        if inTower or string.match(msgName, "Map") or string.match(msgName, "Relive") or string.match(msgName, "Tip") or string.match(msgName, "Prompt") then
            if msgName ~= "UserPackage.ResHeart" and msgName ~= "MapPackage.ResMove" and msgName ~= "MapPackage.ResChangeDir" then
                FileLog("[RECV] " .. msgName .. " (ID: " .. tostring(id) .. ")")

                if id == 101005 or msgName == "FightPackage.ResRelive" then
                    QuickMsg("SERVER PHẢN HỒI: ResRelive thành công! lid=" .. tostring(msg and msg.lid))
                    FileLog("  -> ResRelive: lid=" .. tostring(msg and msg.lid) .. " hp=" .. tostring(msg and msg.hp))
                end

                if id == 100307 or msgName == "MapPackage.ResChangeMap" then
                    local tMap = msg and (msg.mid or msg.mapId)
                    local tX = msg and msg.x
                    local tY = msg and msg.y
                    QuickMsg("SERVER ĐỔI MAP: tới mid=" .. tostring(tMap) .. " (" .. tostring(tX) .. ", " .. tostring(tY) .. ")")
                    FileLog("  -> ResChangeMap: targetMap=" .. tostring(tMap) .. " x=" .. tostring(tX) .. " y=" .. tostring(tY))
                end

                if id == 300001 or msgName == "TipPackage.PromptMsg" then
                    QuickMsg("SERVER BÁO LỖI: " .. tostring(msg and msg.content))
                    FileLog("  -> PromptMsg: " .. tostring(msg and msg.content))
                end
            end
        end

        return _G.Mod_Orig_NetDispatch(id, msg)
    end
end

FileLog("Kịch bản v7 đã sẵn sàng!")
