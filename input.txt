local players = _G.RoleManager.GetRolesByTypeAndRangeAlive({_G.ERoleType.Player}, 20, nil)
local count = 0
local msg = "Quét thấy:\n"

if players and #players > 0 then
    for i, v in ipairs(players) do
        -- Bỏ qua chính bản thân mình
        if v.id ~= _G.RoleManager.me.id then
            local name = v.name or "Unknown"
            local level = v.level or 0
            msg = msg .. "- " .. name .. " (Lv." .. level .. ")\n"
            count = count + 1
        end
    end
end

if count > 0 then
    _G.FloatingTipUtility.QuickMsg(msg)
else
    _G.FloatingTipUtility.QuickMsg("Không có ai ở gần đây!")
end
